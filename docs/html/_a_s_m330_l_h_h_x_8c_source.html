20bool ASM330LHHX_Init(ASM330LHHX_t *dev, SPI_HandleTypeDef *hspi) {
21 if (!dev || !hspi) {
22 return false;
23 }
24
25 dev->hspi = hspi;
26 dev->data_ready = 0;
27 memset(dev->read_buffer, 0, sizeof(dev->read_buffer));
28
29 // Setup ST driver context
30 dev_ctx.write_reg = platform_write;
31 dev_ctx.read_reg = platform_read;
32 dev_ctx.mdelay = platform_delay;
33 dev_ctx.handle = (void*)hspi;
34
35
36 HAL_Delay(BOOT_TIME);
37
38 // Check device ID - initialize to invalid value first
39 whoami = 0x00;
40 if (asm330lhhx_device_id_get(&dev_ctx, &whoami) != 0) {
41 printf("[IMU] Failed to read device ID\r\n");
42 return false;
43 }
44 printf("[IMU] WHO_AM_I = 0x%02X (expected 0x%02X)\r\n", whoami, ASM330LHHX_ID);
45 if (whoami != ASM330LHHX_ID) {
46 printf("[IMU] Device ID mismatch!\r\n");
47 return false;
48 }
49
50 return true;
51}

53bool ASM330LHHX_Configure(ASM330LHHX_t *dev) {
54 if (!dev) {
55 return false;
56 }
57
58
59 // Restore default configuration
60 if (asm330lhhx_reset_set(&dev_ctx, PROPERTY_ENABLE) != 0) {return false;}
61 do {
62 asm330lhhx_reset_get(&dev_ctx, &rst);
63 } while (rst);
64 // Disable I3C interface
65 if (asm330lhhx_i3c_disable_set(&dev_ctx, ASM330LHHX_I3C_DISABLE) != 0) {return false;}
66
67 // Set full speed
68 if (asm330lhhx_xl_data_rate_set(&dev_ctx, ASM330LHHX_XL_ODR_6667Hz) != 0) {return false;}
69 if (asm330lhhx_gy_data_rate_set(&dev_ctx, ASM330LHHX_GY_ODR_6667Hz) != 0) {return false;}
70
71 // Set full scales
72 if (asm330lhhx_xl_full_scale_set(&dev_ctx, ASM330LHHX_2g)) {return false;}
73 if (asm330lhhx_gy_full_scale_set(&dev_ctx, ASM330LHHX_2000dps)) {return false;}
74
75 // Enable data ready interrupt
76 if (asm330lhhx_pin_int1_route_get(&dev_ctx, &int1_route) != 0) {return false;}
77 int1_route.md1_cfg.int1_ff = PROPERTY_ENABLE;
78 if (asm330lhhx_pin_int1_route_set(&dev_ctx, &int1_route) != 0) {return false;}
79
80 return true;
81}

83bool ASM330LHHX_StartReadDMA(ASM330LHHX_t *dev) {
84 if (!dev || !dev->hspi) {
85 return false;
86 }
87
88 // Prepare TX buffer: [COMMAND][DUMMIES]
89 uint8_t tx_buffer[15];
90 tx_buffer[0] = (1 << 7) | (0x20 << 1); // Read from TEMP_OUT_L (0x20)
91 memset(&tx_buffer[1], 0x00, 14);
92
93 // Single DMA TransmitReceive
94 if (HAL_SPI_TransmitReceive_DMA(dev->hspi, tx_buffer, dev->read_buffer, 15) != HAL_OK) {
95 return false;
96 }
97
98 return true;
99}

101void ASM330LHHX_ParseDMABuffer(ASM330LHHX_t *dev) {
102 if (!dev) {
103 return;
104 }
105
106 // Parse received buffer (skip first byte which is command echo)
107 dev->temp_raw = (int16_t)((dev->read_buffer[2] << 8) | dev->read_buffer[1]);
108
109 dev->accel_raw[0] = (int16_t)((dev->read_buffer[4] << 8) | dev->read_buffer[3]);
110 dev->accel_raw[1] = (int16_t)((dev->read_buffer[6] << 8) | dev->read_buffer[5]);
111 dev->accel_raw[2] = (int16_t)((dev->read_buffer[8] << 8) | dev->read_buffer[7]);
112
113 dev->gyro_raw[0] = (int16_t)((dev->read_buffer[10] << 8) | dev->read_buffer[9]);
114 dev->gyro_raw[1] = (int16_t)((dev->read_buffer[12] << 8) | dev->read_buffer[11]);
115 dev->gyro_raw[2] = (int16_t)((dev->read_buffer[14] << 8) | dev->read_buffer[13]);
116
117 dev->data_ready = 1;
118}

120bool ASM330LHHX_ProcessData(ASM330LHHX_t *dev, IMU_t *output) {
121 if (!dev || !output || !dev->data_ready) {
122 return false;
123 }
124
125 // Convert raw values to physical units
126 output->accel_x = asm330lhhx_from_fs2g_to_mg(dev->accel_raw[0]);
127 output->accel_y = asm330lhhx_from_fs2g_to_mg(dev->accel_raw[1]);
128 output->accel_z = asm330lhhx_from_fs2g_to_mg(dev->accel_raw[2]);
129
130 output->gyro_x = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[0]);
131 output->gyro_y = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[1]);
132 output->gyro_z = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[2]);
133
134 output->temperature_c = asm330lhhx_from_lsb_to_celsius(dev->temp_raw);
135 output->timestamp_ms = HAL_GetTick();
136
137 dev->data_ready = 0;
138
139 return true;
140}

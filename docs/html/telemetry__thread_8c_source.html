23void telemetry_thread_function() {
24 printf("[TELEM] Thread started\r\n");
25 fsm_report_thread_started("TELEMETRY");
26 data_packet_t packet;
27
28 TickType_t last_update = xTaskGetTickCount();
29 const TickType_t update_period = pdMS_TO_TICKS(50); // 20Hz update to radio
30
31 while (1) {
32 // Receive sensor data from queues
33 while (xQueueReceive(queue_to_telemetry, &packet, 0) == pdTRUE) {
34 data_packet_lock(&packet);
35
36 switch (packet.type) {
37 case DATA_TYPE_IMU:
38 data_packet_copy_imu(&packet, &latest_imu);
39 have_imu = true;
40 break;
41 case DATA_TYPE_BARO:
42 data_packet_copy_baro(&packet, &latest_baro);
43 have_baro = true;
44 break;
45 case DATA_TYPE_BNO:
46 data_packet_copy_bno(&packet, &latest_bno);
47 have_bno = true;
48 break;
49 case DATA_TYPE_GPS:
50 data_packet_copy_gps(&packet, &latest_gps);
51 have_gps = true;
52 break;
53 default:
54 break;
55 }
56
57 data_packet_unlock(&packet);
58 }
59
60 // Periodically update radio thread with latest data
61 TickType_t now = xTaskGetTickCount();
62 if ((now - last_update) >= update_period) {
63 radio_update_sensor_data(
64 have_imu ? &latest_imu : NULL,
65 have_baro ? &latest_baro : NULL,
66 have_bno ? &latest_bno : NULL,
67 have_gps ? &latest_gps : NULL
68 );
69 last_update = now;
70 }
71
72 vTaskDelay(pdMS_TO_TICKS(10));
73 }

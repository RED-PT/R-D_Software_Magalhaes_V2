2 * FX29.c - TE Connectivity FX29 I2C Load Cell Driver
3 *
4 * Created on: Jan 11, 2026
5 * Author: Tomas Teixeira
6 *
7 * I2C Protocol:
8 * - Simple 3-byte read (no command required)
9 * - Byte 0: [Status(2)] [Bridge Data(6)] (upper 6 bits of 14-bit data)
10 * - Byte 1: [Bridge Data(8)] (lower 8 bits of 14-bit data)
11 * - Byte 2: [Temperature(8)] (optional, for compensation)
12 *
13 * Force Calculation:
14 * OUTPUT = (counts - 1000) / 14000 * FORCE_RANGE
15 */

29bool FX29_Init(FX29_t *dev, I2C_HandleTypeDef *hi2c,
30 uint8_t i2c_addr, FX29_Range_t range) {
31 if (!dev || !hi2c) {
32 return false;
33 }
34
35 dev->hi2c = hi2c;
36 dev->i2c_addr = i2c_addr;
37 dev->force_range_n = range;

44 return false;
45 }
46
47 printf("[FX29] Initialized: addr=0x%02X, range=%dN\r\n",
48 i2c_addr, (int)range);
49
50 return true;

73bool FX29_ReadWithPWM(FX29_t *dev, LOADCELL_t *output, uint16_t pwm_value) {
74 if (!dev || !dev->hi2c || !output) {
75 return false;
76 }
77
78 memset(output, 0, sizeof(LOADCELL_t));
79 output->timestamp_ms = HAL_GetTick();
80 output->pwm_value = pwm_value;
81
82 // Read 2 bytes (status + bridge data)
83 // Some variants have 3rd temperature byte, but we only need force
84 uint8_t rx_buf[2];
85 HAL_StatusTypeDef hal_status = HAL_I2C_Master_Receive(
86 dev->hi2c,
87 (uint16_t)(dev->i2c_addr << 1),
88 rx_buf,
89 2,
90 FX29_I2C_TIMEOUT_MS
91 );
92
93 if (hal_status != HAL_OK) {
94 output->status = FX29_COMM_ERROR;
95 return false;
96 }
97
98 // Parse status bits
99 output->status = FX29_ParseStatus(rx_buf[0]);
100
101 // Parse 14-bit counts
102 output->raw_counts = FX29_ParseCounts(rx_buf);
103
104 // Convert to force (uncorrected)
105 output->force_raw_n = FX29_CountsToForce(dev, output->raw_counts);
106
107 // Apply tare if set
108 if (dev->is_tared) {
109 // Subtract tare offset to get delta from zero point
110 int32_t corrected_counts = (int32_t)output->raw_counts - dev->tare_offset;
111 // Allow negative values for compression, clamp extreme negatives
112 if (corrected_counts < -FX29_COUNT_SPAN) corrected_counts = -FX29_COUNT_SPAN;
113 output->force_n = ((float)corrected_counts / (float)FX29_COUNT_SPAN)
114 * (float)dev->force_range_n;
115 } else {
116 output->force_n = output->force_raw_n;
117 }
118
119 return true;
120}

129bool FX29_Tare(FX29_t *dev) {
130 if (!dev) return false;
131
132 printf("[FX29] Starting tare (%d samples)...\r\n", FX29_TARE_SAMPLES);
133
134 int32_t sum = 0;
135 int valid_samples = 0;
136 LOADCELL_t reading;
137
138 for (int i = 0; i < FX29_TARE_SAMPLES; i++) {
139 HAL_Delay(FX29_TARE_DELAY_MS);
140
141 if (FX29_Read(dev, &reading)) {
142 if (reading.status == FX29_OK || reading.status == FX29_STALE_DATA) {
143 sum += reading.raw_counts;
144 valid_samples++;
145 }
146 }
147 }
148
149 if (valid_samples < (FX29_TARE_SAMPLES / 2)) {
150 printf("[FX29] Tare failed: only %d valid samples\r\n", valid_samples);
151 return false;
152 }
153
154 // Store the average raw counts as the tare offset
155 // This makes the current reading = 0 after tare
156 int32_t avg_counts = sum / valid_samples;
157 dev->tare_offset = avg_counts;
158 dev->is_tared = true;
159
160 printf("[FX29] Tare complete: zero_offset=%ld raw counts\r\n", avg_counts);
161
162 return true;
163}

173float FX29_CountsToForce(FX29_t *dev, uint16_t counts) {
174 if (!dev) return 0.0f;
175
176 // Formula from datasheet:
177 // Force% = (counts - 1000) / 14000 * 100
178 // Force_N = Force% * Range_N / 100
179
180 int32_t offset_counts = (int32_t)counts - FX29_COUNT_MIN;
181 if (offset_counts < 0) offset_counts = 0;
182
183 float force = ((float)offset_counts / (float)FX29_COUNT_SPAN)
184 * (float)dev->force_range_n;
185
186 return force;
187}

203static FX29_Status_t FX29_ParseStatus(uint8_t status_byte) {
204 uint8_t status_bits = status_byte & FX29_STATUS_MASK;
205
206 switch (status_bits) {
207 case FX29_STATUS_NORMAL:
208 return FX29_OK;
209 case FX29_STATUS_STALE:
210 return FX29_STALE_DATA;
211 case FX29_STATUS_FAULT:
212 case FX29_STATUS_COMMAND:
213 return FX29_FAULT;
214 default:
215 return FX29_FAULT;
216 }
217}

219static uint16_t FX29_ParseCounts(uint8_t *data) {
220 // 14-bit data spread across 2 bytes:
221 // Byte 0: [S1 S0 B13 B12 B11 B10 B9 B8] (status + upper 6 bits)
222 // Byte 1: [B7 B6 B5 B4 B3 B2 B1 B0] (lower 8 bits)
223
224 uint16_t counts = ((uint16_t)(data[0] & 0x3F) << 8) | data[1];
225 return counts;
226}

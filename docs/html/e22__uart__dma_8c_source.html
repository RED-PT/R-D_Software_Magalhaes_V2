35void E22_SetMode(uint8_t mode) {
36 switch(mode) {
37 case E22_MODE_NORMAL: M0_LOW(); M1_LOW(); break;
38 case E22_MODE_WOR: M0_HIGH(); M1_LOW(); break;
39 case E22_MODE_CONFIG: M0_LOW(); M1_HIGH(); break;
40 case E22_MODE_SLEEP: M0_HIGH(); M1_HIGH(); break;
41 }
42 HAL_Delay(50); // Mode switch settling time

78E22_Status_t E22_Transmit(uint8_t *data, uint16_t length) {
79 if (!data || length == 0) return E22_ERR_INVALID_PARAM;
80 if (length > E22_MAX_PACKET_SIZE) return E22_ERR_INVALID_PARAM;
81 if (tx_busy) return E22_ERR_BUSY;
82
83 memcpy(tx_buffer, data, length);
84 tx_busy = true;
85
86 HAL_StatusTypeDef status = HAL_UART_Transmit_DMA(e22_uart, tx_buffer, length);
87
88 if (status != HAL_OK) {
89 tx_busy = false;
90 stats.tx_failures++;
91 return E22_ERR_TIMEOUT;
92 }
93
94 stats.tx_packets++;
95 stats.tx_bytes += length;
96 return E22_OK;
97}

116uint16_t E22_Available(void) {
117 // Check for UART errors
118 if (__HAL_UART_GET_FLAG(e22_uart, UART_FLAG_ORE)) {
119 E22_UART_ErrorCallback();
120 return 0;
121 }
122
123 uint16_t dma_remaining = __HAL_DMA_GET_COUNTER(e22_uart->hdmarx);
124 uint16_t dma_pos = RX_BUFFER_SIZE - dma_remaining;
125
126 if (dma_pos >= rx_read_pos) {
127 return dma_pos - rx_read_pos;
128 } else {
129 return RX_BUFFER_SIZE - rx_read_pos + dma_pos;
130 }
131}

133int E22_Receive(uint8_t *buffer, uint16_t max_length) {
134 if (!buffer || max_length == 0) return 0;
135
136 uint16_t available = E22_Available();
137 if (available == 0) return 0;
138
139 uint16_t to_read = (available < max_length) ? available : max_length;
140 uint16_t count = 0;
141
142 uint16_t dma_remaining = __HAL_DMA_GET_COUNTER(e22_uart->hdmarx);
143 uint16_t dma_pos = RX_BUFFER_SIZE - dma_remaining;
144
145 while (rx_read_pos != dma_pos && count < to_read) {
146 buffer[count++] = rx_dma_buffer[rx_read_pos];
147 rx_read_pos = (rx_read_pos + 1) % RX_BUFFER_SIZE;
148 }
149
150 if (count > 0) {
151 stats.rx_packets++;
152 stats.rx_bytes += count;
153 }
154
155 return count;
156}

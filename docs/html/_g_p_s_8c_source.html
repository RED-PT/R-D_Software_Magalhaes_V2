27float GPS_nmea_to_dec(float deg_coord, char nsew) {
28 int degree = (int)(deg_coord / 100);
29 float minutes = deg_coord - degree * 100;
30 float dec_deg = minutes / 60;
31 float decimal = degree + dec_deg;
32 if (nsew == 'S' || nsew == 'W') { // return negative
33 decimal *= -1;
34 }
35 return decimal;
36}

38int GPS_validate(char *nmeastr) {
39 char check[3];
40 char checkcalcstr[3];
41 int i;
42 int calculated_check;
43 i = 0;
44 calculated_check = 0;
45 // check to ensure that the string starts with a $
46 if (nmeastr[i] == '$')
47 i++;
48 else
49 return 0;
50 //No NULL reached, 75 char largest possible NMEA message, no '*' reached
51 while ((nmeastr[i] != 0) && (nmeastr[i] != '*') && (i < strlen(nmeastr))) {
52 calculated_check ^= nmeastr[i]; // calculate the checksum
53 i++;
54 }
55 if (i >= strlen(nmeastr)) {
56 return 0; // the string was too long so return an error
57 }
58 if (nmeastr[i] == '*') {
59 check[0] = nmeastr[i + 1]; //put hex chars in check string
60 check[1] = nmeastr[i + 2];
61 check[2] = 0;
62 } else
63 return 0; // no checksum separator found there for invalid
64 sprintf(checkcalcstr, "%02X", calculated_check);
65 return ((checkcalcstr[0] == check[0]) && (checkcalcstr[1] == check[1])) ?
66 1 : 0;
67}

69void GPS_parse(GPS_t *gps, char *GPSstrParse) {
70 // Temporary variables for NMEA format conversion
71 float nmea_lat, nmea_lon;
72 char ns, ew;
73 char msl_units;
74
75 if (!strncmp(GPSstrParse, "$GNGGA", 6)) {
76 if (sscanf(GPSstrParse, "$GNGGA,%f,%f,%c,%f,%c,%d,%d,%f,%f,%c",
77 &gps->utc_time, &nmea_lat, &ns, &nmea_lon, &ew,
78 (int*)&gps->lock, (int*)&gps->satellites,
79 &gps->hdop, &gps->msl_altitude, &msl_units) >= 1) {
80
81 // Convert NMEA to decimal degrees
82 gps->dec_latitude = GPS_nmea_to_dec(nmea_lat, ns);
83 gps->dec_longitude = GPS_nmea_to_dec(nmea_lon, ew);
84 return;
85 }
86 }
87
88 else if (!strncmp(GPSstrParse, "$GPGGA", 6)) {
89 if (sscanf(GPSstrParse, "$GPGGA,%f,%f,%c,%f,%c,%d,%d,%f,%f,%c",
90 &gps->utc_time, &nmea_lat, &ns, &nmea_lon, &ew,
91 (int*)&gps->lock, (int*)&gps->satellites,
92 &gps->hdop, &gps->msl_altitude, &msl_units) >= 1) {
93
94 // Convert NMEA to decimal degrees
95 gps->dec_latitude = GPS_nmea_to_dec(nmea_lat, ns);
96 gps->dec_longitude = GPS_nmea_to_dec(nmea_lon, ew);
97 return;
98 }
99 }
100
101 else if (!strncmp(GPSstrParse, "$GPRMC", 6)) {
102 char rmc_status;
103 int date;
104 float mag_dev;
105 char mag_dev_unit;
106
107 if (sscanf(GPSstrParse, "$GPRMC,%f,%c,%f,%c,%f,%c,%f,%f,%6d,%f,%c",

144bool UBLOX_GPS_StartDMA(UBLOX_GPS_t *dev) {
145 if (!dev || !dev->huart) {
146 return false;
147 }
148
149 // Start circular DMA receive on UART
150 if (HAL_UART_Receive_DMA(dev->huart, dev->dma_buffer, UBLOX_GPS_UART_BUFFER_SIZE) != HAL_OK) {
151 return false;
152 }
153
154 // DISABLE half-transfer interrupt - only use complete and IDLE
155 __HAL_DMA_DISABLE_IT(dev->huart->hdmarx, DMA_IT_HT);
156
157 // Enable UART IDLE interrupt

160 dev->state = UBLOX_STATE_RECEIVING;
161
162 return true;

174bool UBLOX_GPS_Update(UBLOX_GPS_t *dev) {
175 if (!dev || dev->state != UBLOX_STATE_RECEIVING) {
176 return false;
177 }
178
179 uint16_t dma_pos = UBLOX_GPS_GetDMAPosition(dev);
180 uint16_t bytes_available;
181 uint16_t bytes_to_process;
182 uint8_t byte;
183
184 // Calculate bytes available in buffer
185 if (dma_pos >= dev->parse_tail) {
186 bytes_available = dma_pos - dev->parse_tail;
187 } else {
188 bytes_available = (UBLOX_GPS_UART_BUFFER_SIZE - dev->parse_tail) + dma_pos;
189 }
190
191 if (bytes_available == 0) {
192 return false; // No new data
193 }
194
195 // CRITICAL FIX: Limit bytes processed per call to avoid blocking thread
196 bytes_to_process = (bytes_available > MAX_BYTES_PER_UPDATE) ?
197 MAX_BYTES_PER_UPDATE : bytes_available;
198
199 // Process limited number of bytes
200 while (bytes_to_process > 0) {
201 byte = dev->dma_buffer[dev->parse_tail];
202 dev->parse_tail = (dev->parse_tail + 1) % UBLOX_GPS_UART_BUFFER_SIZE;
203 bytes_to_process--;
204
205 // Look for sentence start ($)
206 if (byte == '$') {
207 dev->message_len = 1;
208 dev->message_buffer[0] = '$';
209 continue;
210 }
211
212 // If we're collecting a sentence
213 if (dev->message_len > 0) {
214 // Check for end of sentence (newline or CR)
215 if (byte == '\n' || byte == '\r') {
216 if (dev->message_len > 6) { // Minimum NMEA length
217 dev->message_buffer[dev->message_len] = '\0';
218 dev->state = UBLOX_STATE_COMPLETE;
219 dev->data_ready = 1;
220 return true; // Complete sentence found
221 }
222 dev->message_len = 0; // Reset on too short
223 continue;
224 }
225
226 // Add byte to message if buffer not full
227 if (dev->message_len < UBLOX_GPS_MAX_SENTENCE_LEN - 1) {
228 dev->message_buffer[dev->message_len] = byte;
229 dev->message_len++;
230 } else {
231 // Sentence too long, discard and look for next $
232 dev->message_len = 0;
233 dev->parse_errors++;
234 }
235 }
236 }
237
238 return false; // No complete sentence yet
239}

241bool UBLOX_GPS_ProcessData(UBLOX_GPS_t *dev, GPS_t *output) {
242 if (!dev || !output || !dev->data_ready || dev->state != UBLOX_STATE_COMPLETE) {
243 return false;
244 }
245
246 // Validate NMEA sentence
247 if (!GPS_validate((char*)dev->message_buffer)) {
248 dev->parse_errors++;
249 dev->data_ready = 0;
250 dev->state = UBLOX_STATE_RECEIVING;
251 return false;
252 }
253
254 // Parse NMEA sentence into GPS struct
255 GPS_parse(output, (char*)dev->message_buffer);
256
257 // Add timestamp
258 output->timestamp_ms = HAL_GetTick();
259
260 dev->sentences_received++;
261 dev->data_ready = 0;
262 dev->state = UBLOX_STATE_RECEIVING;
263
264 return true;
265}

267static void UBX_CalculateChecksum(uint8_t *data, uint16_t len, uint8_t *ck_a, uint8_t *ck_b) {
268 *ck_a = 0;
269 *ck_b = 0;

272 *ck_b += *ck_a;

276// Send UBX command to GPS
277bool UBLOX_GPS_SendUBX(UBLOX_GPS_t *dev, uint8_t msg_class, uint8_t msg_id,
278 uint8_t *payload, uint16_t payload_len) {
279 if (!dev || !dev->huart) {
280 return false;
281 }
282
283 uint8_t buffer[256];
284 uint16_t idx = 0;
285
286 // Header
287 buffer[idx++] = UBX_SYNC1;
288 buffer[idx++] = UBX_SYNC2;
289 buffer[idx++] = msg_class;
290 buffer[idx++] = msg_id;
291 buffer[idx++] = payload_len & 0xFF;
292 buffer[idx++] = (payload_len >> 8) & 0xFF;
293
294 // Payload
295 if (payload && payload_len > 0) {
296 memcpy(&buffer[idx], payload, payload_len);
297 idx += payload_len;
298 }
299
300 // Checksum (over class, id, length, payload)
301 uint8_t ck_a, ck_b;
302 UBX_CalculateChecksum(&buffer[2], idx - 2, &ck_a, &ck_b);
303 buffer[idx++] = ck_a;
304 buffer[idx++] = ck_b;
305
306 // Send via UART
307 HAL_StatusTypeDef status = HAL_UART_Transmit(dev->huart, buffer, idx, 100);
308 HAL_Delay(50); // Give GPS time to process
309
310 return (status == HAL_OK);
311}

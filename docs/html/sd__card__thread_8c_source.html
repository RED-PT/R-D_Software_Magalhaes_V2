89static void flush_write_buffer(void) {
90 if (buffer_pos == 0 || !sd_file_open || sd_paused) {
91 return;
92 }
93
94 UINT bw;
95 FRESULT res = f_write(&file, write_buffer, buffer_pos, &bw);
96
97 if (res != FR_OK) {
98 printf("ERROR: SD write failed (error %d)\r\n", res);
99 buffer_pos = 0;
100 return;
101 }
102
103 if (bw != buffer_pos) {
104 printf("WARNING: Partial write (%u/%lu)\r\n", bw, buffer_pos);
105 }
106
107 res = f_sync(&file);
108 if (res != FR_OK) {
109 printf("ERROR: f_sync failed (error %d)\r\n", res);
110 }
111
112 buffer_pos = 0;
113}

115static void log_data_packet(const data_packet_t *packet) {
116 if (!sd_file_open || sd_paused) {
117 return; // Skip logging when paused (motor test) to prevent buffer overflow
118 }
119
120 int line_len = 0;
121 char line_buffer[512];
122
123 data_packet_lock(packet);
124
125 switch (packet->type) {
126 case DATA_TYPE_IMU: {
127 IMU_t imu;
128 data_packet_copy_imu(packet, &imu);
129 // Columns 1-9: timestamp, type, seq, accel(3), gyro(3), temp
130 // Columns 10-30: empty (21 commas)
131 line_len = snprintf(line_buffer, sizeof(line_buffer),
132 "%lu,IMU,%lu,%.3f,%.3f,%.3f,%.3f,%.3f,%.3f,%.2f,,,,,,,,,,,,,,,,,,,\r\n",
133 packet->timestamp_ms, packet->sequence,
134 imu.accel_x, imu.accel_y, imu.accel_z,
135 imu.gyro_x, imu.gyro_y, imu.gyro_z, imu.temperature_c);
136 break;
137 }
138
139 case DATA_TYPE_BARO: {
140 BARO_t baro;
141 data_packet_copy_baro(packet, &baro);
142 // Columns 1-3: timestamp, type, seq
143 // Columns 4-9: empty (6 commas)
144 // Columns 10-12: temp, pressure, altitude
145 // Columns 13-30: empty (18 commas)
146 line_len = snprintf(line_buffer, sizeof(line_buffer),
147 "%lu,BARO,%lu,,,,,,,%.2f,%.2f,%.2f,,,,,,,,,,,,,,,,\r\n",
148 packet->timestamp_ms, packet->sequence,
149 baro.temperature_c, baro.pressure_mbar, baro.altitude_m);
150 break;
151 }
152
153 case DATA_TYPE_MAG: {
154 MAG_t mag;
155 data_packet_copy_mag(packet, &mag);
156 // Columns 1-3: timestamp, type, seq
157 // Columns 4-12: empty (9 commas)
158 // Columns 13-15: mag_x, mag_y, mag_z
159 // Columns 16-30: empty (15 commas)
160 line_len = snprintf(line_buffer, sizeof(line_buffer),
161 "%lu,MAG,%lu,,,,,,,,,,,%.3f,%.3f,%.3f,,,,,,,,,,,,,\r\n",
162 packet->timestamp_ms, packet->sequence,
163 mag.mag_x, mag.mag_y, mag.mag_z);
164 break;
165 }
166
167 case DATA_TYPE_BNO: {
168 BNO_t bno;
169 data_packet_copy_bno(packet, &bno);
170 // Columns 1-3: timestamp, type, seq
171 // Columns 4-15: empty (12 commas)
172 // Columns 16-18: heading, roll, pitch
173 // Columns 19-26: empty (8 commas)
174 // Columns 27-30: quat_w, quat_x, quat_y, quat_z
175 line_len = snprintf(line_buffer, sizeof(line_buffer),
176 "%lu,BNO,%lu,,,,,,,,,,,,,%.2f,%.2f,%.2f,,,,,,,,,%.4f,%.4f,%.4f,%.4f\r\n",
177 packet->timestamp_ms, packet->sequence,
178 bno.heading_deg, bno.roll_deg, bno.pitch_deg,
179 bno.quat_w, bno.quat_x, bno.quat_y, bno.quat_z);
180 break;
181 }
182
183 case DATA_TYPE_GPS: {
184 GPS_t gps;
185 data_packet_copy_gps(packet, &gps);
186
187 // Only log valid GPS data when lock acquired
188 if (gps.lock > 0 && gps.satellites > 0) {
189 // Columns 1-3: timestamp, type, seq
190 // Columns 4-18: empty (15 commas)
191 // Columns 19-25: lat, lon, alt, lock, sats, hdop, speed, course
192 // Columns 26-30: empty (5 commas)
193 line_len = snprintf(line_buffer, sizeof(line_buffer),
194 "%lu,GPS,%lu,,,,,,,,,,,,,,,,,%.6f,%.6f,%.2f,%d,%d,%.2f,%.2f,%.1f,,,,,\r\n",
195 packet->timestamp_ms, packet->sequence,
196 gps.dec_latitude, gps.dec_longitude, gps.msl_altitude,
197 gps.lock, gps.satellites, gps.hdop, gps.speed_k, gps.course_d);
198 } else {
199 // No GPS lock - log minimal info
200 line_len = snprintf(line_buffer, sizeof(line_buffer),
201 "%lu,GPS,%lu,,,,,,,,,,,,,,,,,0.0,0.0,0.0,0,0,99.9,0.0,0.0,,,,,\r\n",
202 packet->timestamp_ms, packet->sequence);
203 }
204 break;
205 }
206
207 case DATA_TYPE_EVENT:
208 break;
209
210 default:
211 break;
212 }
213
214 data_packet_unlock(packet);
215
216 if (line_len > 0 && line_len < (int)(sizeof(line_buffer) - 1)) {
217 if (buffer_pos + line_len > (SD_WRITE_BUFFER_SIZE - 512)) {
218 flush_write_buffer();
219 }
220
221 memcpy(&write_buffer[buffer_pos], line_buffer, line_len);
222 buffer_pos += line_len;
223 }
224}

226void sd_card_thread_function(void *argument) {
227 printf("SD Logger thread started...\r\n");
228 fsm_report_thread_started("SD_CARD");
229
230 sd_card_init(); // Inside this, report status
231 sd_card_configure();
232
233 TickType_t last_flush = xTaskGetTickCount();
234 const TickType_t flush_timeout = pdMS_TO_TICKS(SD_FLUSH_TIMEOUT_MS);
235
236 data_packet_t packet;
237 uint32_t stats_packets = 0;
238 uint32_t stats_queue_empty = 0;
239 TickType_t last_stats = xTaskGetTickCount();
240
241 while(1) {
242 TickType_t now = xTaskGetTickCount();
243
244 if (xQueueReceive(queue_to_sd, &packet, pdMS_TO_TICKS(50)) == pdTRUE) {
245 log_data_packet(&packet);
246 stats_packets++;
247
248 if (buffer_pos > ((SD_WRITE_BUFFER_SIZE * 3) / 4)) {
249 flush_write_buffer();
250 last_flush = now;
251 }
252 } else {
253 stats_queue_empty++;
254 }
255
256 if ((now - last_flush) >= flush_timeout) {
257 if (buffer_pos > 0) {
258 flush_write_buffer();
259 }
260 last_flush = now;
261 }
262
263 if ((now - last_stats) >= pdMS_TO_TICKS(10000)) {
264 UBaseType_t queue_msgs = uxQueueMessagesWaiting(queue_to_sd);
265 printf("SD: %lu pkts logged, buf:%lu/%u bytes, queue:%lu msgs\r\n",
266 stats_packets, buffer_pos, SD_WRITE_BUFFER_SIZE, queue_msgs);
267
268 if (stats_queue_empty > 80) {
269 printf("SD: queue empty %lu times\r\n", stats_queue_empty);
270 }
271
272 stats_packets = 0;
273 stats_queue_empty = 0;
274 last_stats = now;
275 }
276 }

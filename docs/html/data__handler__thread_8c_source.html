19static void flush_buffer_to_queues(ram_circular_buffer_t *cb, data_type_t type, uint32_t *seq) {
20 if (ram_circular_buffer_is_empty(cb)) {return;}
21
22 while (!ram_circular_buffer_is_empty(cb)) {
23 if (xSemaphoreTake(cb->mutex, pdMS_TO_TICKS(5)) != pdTRUE) {
24 break;
25 }
26
27 if (ram_circular_buffer_is_empty(cb)) {
28 xSemaphoreGive(cb->mutex);
29 break;
30 }
31
32 uint8_t *buffer = (uint8_t*)cb->buffer;
33 uint8_t *read_addr = buffer + (cb->tail * cb->sample_size);
34
35 data_packet_t packet = {
36 .type = type,
37 .data_ptr = read_addr,
38 .timestamp_ms = 0,
39 .sequence = (*seq)++,
40 .source_cb = cb
41 };
42
43 switch (type) {
44 case DATA_TYPE_IMU:
45 packet.timestamp_ms = ((IMU_t*)read_addr)->timestamp_ms;
46 break;
47 case DATA_TYPE_BARO:
48 packet.timestamp_ms = ((BARO_t*)read_addr)->timestamp_ms;
49 break;
50 case DATA_TYPE_MAG:
51 packet.timestamp_ms = ((MAG_t*)read_addr)->timestamp_ms;
52 break;
53 case DATA_TYPE_BNO:
54 packet.timestamp_ms = ((BNO_t*)read_addr)->timestamp_ms;
55 break;
56 case DATA_TYPE_GPS:
57 packet.timestamp_ms = ((GPS_t*)read_addr)->timestamp_ms;
58 break;
59 default:
60 break;
61 }
62
63 // Prioritize SD with blocking send (short timeout)
64 xQueueSend(queue_to_sd, &packet, pdMS_TO_TICKS(5));
65
66 // Telemetry is lower priority
67 xQueueSend(queue_to_telemetry, &packet, 0);
68
69 cb->tail = (cb->tail + 1) % cb->capacity;
70 xSemaphoreGive(cb->mutex);
71 }
72}

74static bool is_buffer_above_threshold(ram_circular_buffer_t *cb) {
75 uint32_t available = ram_circular_buffer_available(cb);
76 uint32_t threshold = (cb->capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;

115void data_handler_thread_function() {
116 printf("Data Handler thread started...\r\n");
117 fsm_report_thread_started("DATA_HANDLER");
118 fsm_report_init_status("DATA_HANDLER", true);
119
120 TickType_t last_safety_flush = xTaskGetTickCount();
121 const TickType_t safety_timeout = pdMS_TO_TICKS(SAFETY_FLUSH_TIMEOUT_MS);
122
123 TickType_t last_monitor = xTaskGetTickCount();
124 const TickType_t monitor_period = pdMS_TO_TICKS(10000);
125
126 uint32_t ulNotificationValue;
127 uint32_t flush_count = 0;
128 uint32_t safety_flush_count = 0;
129 uint32_t opportunistic_flush_count = 0;
130
131 while(1) {
132 TickType_t now = xTaskGetTickCount();
133
134 // Wait for events with 200ms timeout for opportunistic flushing
135 xTaskNotifyWait(0, ULONG_MAX, &ulNotificationValue, pdMS_TO_TICKS(200));
136
137 bool should_flush = false;
138 const char *flush_reason = "";
139 (void)flush_reason;
140
141 // EVENT 1: Buffer threshold crossed (50% full)
142 if (ulNotificationValue & DATA_HANDLER_NOTIFY_THRESHOLD) {
143 if (any_buffer_above_threshold()) {
144 should_flush = true;
145 flush_reason = "THRESHOLD";
146 last_safety_flush = now;
147 }
148 }
149
150 // EVENT 2: Safety timeout (always flush if data exists)
151 if (!should_flush && (now - last_safety_flush) >= safety_timeout) {
152 if (any_buffer_has_data()) {
153 should_flush = true;
154 flush_reason = "SAFETY";
155 safety_flush_count++;
156 last_safety_flush = now;
157 }
158 }
159
160 // EVENT 3: Opportunistic flush (SD queue has space and buffers not empty)
161 if (!should_flush && has_queue_space() && any_buffer_has_data()) {
162 should_flush = true;
163 flush_reason = "OPPORTUNISTIC";
164 opportunistic_flush_count++;
165 }
166
167 if (should_flush) {
168 flush_all_buffers();
169 flush_count++;
170 }
171
172 // MONITORING
173 if ((now - last_monitor) >= monitor_period) {
174 printf("Data Handler Status\r\n");
175
176 // Overflows
177 if (cb_imu.overflow_count > 0) printf("WARNING: IMU overflows: %lu\r\n", cb_imu.overflow_count);
178 if (cb_baro.overflow_count > 0) printf("WARNING: BARO overflows: %lu\r\n", cb_baro.overflow_count);
179 if (cb_mag.overflow_count > 0) printf("WARNING: MAG overflows: %lu\r\n", cb_mag.overflow_count);
180 if (cb_bno.overflow_count > 0) printf("WARNING: BNO overflows: %lu\r\n", cb_bno.overflow_count);
181 if (cb_gps.overflow_count > 0) printf("WARNING: GPS overflows: %lu\r\n", cb_gps.overflow_count);
182
183 // Buffer percentages
184 uint32_t imu_pct = (ram_circular_buffer_available(&cb_imu) * 100) / RAM_IMU_BUFFER_SIZE;
185 uint32_t baro_pct = (ram_circular_buffer_available(&cb_baro) * 100) / RAM_BARO_BUFFER_SIZE;
186 uint32_t mag_pct = (ram_circular_buffer_available(&cb_mag) * 100) / RAM_MAG_BUFFER_SIZE;
187 uint32_t bno_pct = (ram_circular_buffer_available(&cb_bno) * 100) / RAM_BNO_BUFFER_SIZE;
188 uint32_t gps_pct = (ram_circular_buffer_available(&cb_gps) * 100) / RAM_GPS_BUFFER_SIZE;
189
190 printf("Buffers: IMU=%lu%%, BARO=%lu%%, MAG=%lu%%, BNO=%lu%%, GPS=%lu%%\r\n",
191 imu_pct, baro_pct, mag_pct, bno_pct, gps_pct);
192
193 // Queues
194 UBaseType_t est_msgs = uxQueueMessagesWaiting(queue_to_estimator);
195 UBaseType_t tel_msgs = uxQueueMessagesWaiting(queue_to_telemetry);
196 UBaseType_t sd_msgs = uxQueueMessagesWaiting(queue_to_sd);
197 UBaseType_t sd_space = uxQueueSpacesAvailable(queue_to_sd);
198
199 printf("Queues: EST=%lu/%u, TEL=%lu/%u, SD=%lu/%u (free:%lu)\r\n",
200 est_msgs, QUEUE_LENGTH_ESTIMATOR,
201 tel_msgs, QUEUE_LENGTH_TELEMETRY,
202 sd_msgs, QUEUE_LENGTH_SD, sd_space);
203
204 printf("Flushes: total=%lu, threshold=%lu, safety=%lu, opportunistic=%lu\r\n",
205 flush_count, flush_count - safety_flush_count - opportunistic_flush_count,
206 safety_flush_count, opportunistic_flush_count);
207
208 last_monitor = now;
209 }
210 }

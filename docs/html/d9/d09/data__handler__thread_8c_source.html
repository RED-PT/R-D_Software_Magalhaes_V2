45static void flush_buffer_to_queues(ram_circular_buffer_t *cb, data_type_t type, uint32_t *seq) {
46 if (ram_circular_buffer_is_empty(cb)) {return;}
47
48 while (!ram_circular_buffer_is_empty(cb)) {
49 if (xSemaphoreTake(cb->mutex, pdMS_TO_TICKS(5)) != pdTRUE) {
50 break;
51 }
52
53 if (ram_circular_buffer_is_empty(cb)) {
54 xSemaphoreGive(cb->mutex);
55 break;
56 }
57
58 uint8_t *buffer = (uint8_t*)cb->buffer;
59 uint8_t *read_addr = buffer + (cb->tail * cb->sample_size);
60
61 data_packet_t packet = {
62 .type = type,
63 .data_ptr = read_addr,
64 .timestamp_ms = 0,
65 .sequence = (*seq)++,
66 .source_cb = cb
67 };
68
69 switch (type) {
70 case DATA_TYPE_IMU:
71 packet.timestamp_ms = ((IMU_t*)read_addr)->timestamp_ms;
72 break;
73 case DATA_TYPE_BARO:
74 packet.timestamp_ms = ((BARO_t*)read_addr)->timestamp_ms;
75 break;
76 case DATA_TYPE_MAG:
77 packet.timestamp_ms = ((MAG_t*)read_addr)->timestamp_ms;
78 break;
79 case DATA_TYPE_BNO:
80 packet.timestamp_ms = ((BNO_t*)read_addr)->timestamp_ms;
81 break;
82 case DATA_TYPE_GPS:
83 packet.timestamp_ms = ((GPS_t*)read_addr)->timestamp_ms;
84 break;
85 default:
86 break;
87 }
88
89 // Prioritize SD with blocking send (short timeout)
90 xQueueSend(queue_to_sd, &packet, pdMS_TO_TICKS(5));
91
92 // Telemetry is lower priority
93 xQueueSend(queue_to_telemetry, &packet, 0);
94
95 cb->tail = (cb->tail + 1) % cb->capacity;
96 xSemaphoreGive(cb->mutex);
97 }
98}

141void data_handler_thread_function() {
142 printf("Data Handler thread started...\r\n");
143 fsm_report_thread_started("DATA_HANDLER");
144 fsm_report_init_status("DATA_HANDLER", true);
145
146 TickType_t last_safety_flush = xTaskGetTickCount();
147 const TickType_t safety_timeout = pdMS_TO_TICKS(SAFETY_FLUSH_TIMEOUT_MS);
148
149 TickType_t last_monitor = xTaskGetTickCount();
150 const TickType_t monitor_period = pdMS_TO_TICKS(10000);
151
152 uint32_t ulNotificationValue;
153 uint32_t flush_count = 0;
154 uint32_t safety_flush_count = 0;
155 uint32_t opportunistic_flush_count = 0;
156
157 while(1) {
158 TickType_t now = xTaskGetTickCount();
159
160 // Wait for events with 200ms timeout for opportunistic flushing
161 xTaskNotifyWait(0, ULONG_MAX, &ulNotificationValue, pdMS_TO_TICKS(200));
162
163 bool should_flush = false;
164 const char *flush_reason = "";
165 (void)flush_reason;
166
167 // EVENT 1: Buffer threshold crossed (50% full)
168 if (ulNotificationValue & DATA_HANDLER_NOTIFY_THRESHOLD) {
169 if (any_buffer_above_threshold()) {
170 should_flush = true;
171 flush_reason = "THRESHOLD";
172 last_safety_flush = now;
173 }
174 }
175
176 // EVENT 2: Safety timeout (always flush if data exists)
177 if (!should_flush && (now - last_safety_flush) >= safety_timeout) {
178 if (any_buffer_has_data()) {
179 should_flush = true;
180 flush_reason = "SAFETY";
181 safety_flush_count++;
182 last_safety_flush = now;
183 }
184 }
185
186 // EVENT 3: Opportunistic flush (SD queue has space and buffers not empty)
187 if (!should_flush && has_queue_space() && any_buffer_has_data()) {
188 should_flush = true;
189 flush_reason = "OPPORTUNISTIC";
190 opportunistic_flush_count++;
191 }
192
193 if (should_flush) {
194 flush_all_buffers();
195 flush_count++;
196 }
197
198 // MONITORING
199 if ((now - last_monitor) >= monitor_period) {
200 printf("Data Handler Status\r\n");
201
202 // Overflows
203 if (cb_imu.overflow_count > 0) printf("WARNING: IMU overflows: %lu\r\n", cb_imu.overflow_count);
204 if (cb_baro.overflow_count > 0) printf("WARNING: BARO overflows: %lu\r\n", cb_baro.overflow_count);
205 if (cb_mag.overflow_count > 0) printf("WARNING: MAG overflows: %lu\r\n", cb_mag.overflow_count);
206 if (cb_bno.overflow_count > 0) printf("WARNING: BNO overflows: %lu\r\n", cb_bno.overflow_count);
207 if (cb_gps.overflow_count > 0) printf("WARNING: GPS overflows: %lu\r\n", cb_gps.overflow_count);
208
209 // Buffer percentages
210 uint32_t imu_pct = (ram_circular_buffer_available(&cb_imu) * 100) / RAM_IMU_BUFFER_SIZE;
211 uint32_t baro_pct = (ram_circular_buffer_available(&cb_baro) * 100) / RAM_BARO_BUFFER_SIZE;
212 uint32_t mag_pct = (ram_circular_buffer_available(&cb_mag) * 100) / RAM_MAG_BUFFER_SIZE;
213 uint32_t bno_pct = (ram_circular_buffer_available(&cb_bno) * 100) / RAM_BNO_BUFFER_SIZE;
214 uint32_t gps_pct = (ram_circular_buffer_available(&cb_gps) * 100) / RAM_GPS_BUFFER_SIZE;
215
216 printf("Buffers: IMU=%lu%%, BARO=%lu%%, MAG=%lu%%, BNO=%lu%%, GPS=%lu%%\r\n",
217 imu_pct, baro_pct, mag_pct, bno_pct, gps_pct);
218
219 // Queues
220 UBaseType_t est_msgs = uxQueueMessagesWaiting(queue_to_estimator);
221 UBaseType_t tel_msgs = uxQueueMessagesWaiting(queue_to_telemetry);
222 UBaseType_t sd_msgs = uxQueueMessagesWaiting(queue_to_sd);
223 UBaseType_t sd_space = uxQueueSpacesAvailable(queue_to_sd);
224
225 printf("Queues: EST=%lu/%u, TEL=%lu/%u, SD=%lu/%u (free:%lu)\r\n",
226 est_msgs, QUEUE_LENGTH_ESTIMATOR,
227 tel_msgs, QUEUE_LENGTH_TELEMETRY,
228 sd_msgs, QUEUE_LENGTH_SD, sd_space);
229
230 printf("Flushes: total=%lu, threshold=%lu, safety=%lu, opportunistic=%lu\r\n",
231 flush_count, flush_count - safety_flush_count - opportunistic_flush_count,
232 safety_flush_count, opportunistic_flush_count);
233
234 last_monitor = now;
235 }
236 }

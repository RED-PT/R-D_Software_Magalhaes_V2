115void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
116 if (osKernelGetState() != osKernelRunning) return;
117
118 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
119
120 // Sensor DRDY pins
121 if (GPIO_Pin == EXTI_IMU_PIN) {
122 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_IMU_DRDY,
123 eSetBits, &xHigherPriorityTaskWoken);
124 }
125 else if (GPIO_Pin == EXTI_MAG_PIN) {
126 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_MAG_DRDY,
127 eSetBits, &xHigherPriorityTaskWoken);
128 }
129 // SX126x DIO1 interrupt (PB2)
130 else if (GPIO_Pin == EXTI_LORA_PIN) {
131 SX126x_DIO1_IRQ_Handler();
132 }
133
134 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
135}

158void HAL_SPI_TxRxCpltCallback(SPI_HandleTypeDef *hspi) {
159 if (osKernelGetState() != osKernelRunning) return;
160
161 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
162
163 // Parse DMA buffer based on active sensor
164 if (hspi->Instance == SPI_IMU_BARO_INSTANCE) {
165 if (spi1_active_sensor == ACTIVE_SENSOR_IMU) {
166 ASM330LHHX_ParseDMABuffer(&imu_device);
167 }
168 }
169 else if (hspi->Instance == SPI_MAG_INSTANCE) {
170 if (spi_mag_active_sensor == ACTIVE_SENSOR_MAG) {
171 MMC5983MA_ParseDMABuffer(&mag_device);
172 }
173 }
174
175 // Notify sensors thread
176 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_DMA_COMPLETE,
177 eSetBits, &xHigherPriorityTaskWoken);
178
179 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
180}

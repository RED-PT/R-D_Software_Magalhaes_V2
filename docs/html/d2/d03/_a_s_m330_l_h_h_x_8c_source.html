55bool ASM330LHHX_Init(ASM330LHHX_t *dev, SPI_HandleTypeDef *hspi) {
56 if (!dev || !hspi) {
57 return false;
58 }
59
60 dev->hspi = hspi;
61 dev->data_ready = 0;
62 memset(dev->read_buffer, 0, sizeof(dev->read_buffer));
63
64 // Setup ST driver context
65 dev_ctx.write_reg = platform_write;
66 dev_ctx.read_reg = platform_read;
67 dev_ctx.mdelay = platform_delay;
68 dev_ctx.handle = (void*)hspi;
69
70
71 HAL_Delay(BOOT_TIME);
72
73 // Check device ID - initialize to invalid value first
74 whoami = 0x00;
75 if (asm330lhhx_device_id_get(&dev_ctx, &whoami) != 0) {
76 printf("[IMU] Failed to read device ID\r\n");
77 return false;
78 }
79 printf("[IMU] WHO_AM_I = 0x%02X (expected 0x%02X)\r\n", whoami, ASM330LHHX_ID);
80 if (whoami != ASM330LHHX_ID) {
81 printf("[IMU] Device ID mismatch!\r\n");
82 return false;
83 }
84
85 return true;
86}

88bool ASM330LHHX_Configure(ASM330LHHX_t *dev) {
89 if (!dev) {
90 return false;
91 }
92
93
94 // Restore default configuration
95 if (asm330lhhx_reset_set(&dev_ctx, PROPERTY_ENABLE) != 0) {return false;}
96 do {
97 asm330lhhx_reset_get(&dev_ctx, &rst);
98 } while (rst);
99 // Disable I3C interface
100 if (asm330lhhx_i3c_disable_set(&dev_ctx, ASM330LHHX_I3C_DISABLE) != 0) {return false;}
101
102 // Set full speed
103 if (asm330lhhx_xl_data_rate_set(&dev_ctx, ASM330LHHX_XL_ODR_6667Hz) != 0) {return false;}
104 if (asm330lhhx_gy_data_rate_set(&dev_ctx, ASM330LHHX_GY_ODR_6667Hz) != 0) {return false;}
105
106 // Set full scales
107 if (asm330lhhx_xl_full_scale_set(&dev_ctx, ASM330LHHX_2g)) {return false;}
108 if (asm330lhhx_gy_full_scale_set(&dev_ctx, ASM330LHHX_2000dps)) {return false;}
109
110 // Enable data ready interrupt
111 if (asm330lhhx_pin_int1_route_get(&dev_ctx, &int1_route) != 0) {return false;}
112 int1_route.md1_cfg.int1_ff = PROPERTY_ENABLE;
113 if (asm330lhhx_pin_int1_route_set(&dev_ctx, &int1_route) != 0) {return false;}
114
115 return true;
116}

118bool ASM330LHHX_StartReadDMA(ASM330LHHX_t *dev) {
119 if (!dev || !dev->hspi) {
120 return false;
121 }
122
123 // Prepare TX buffer: [COMMAND][DUMMIES]
124 uint8_t tx_buffer[15];
125 tx_buffer[0] = (1 << 7) | (0x20 << 1); // Read from TEMP_OUT_L (0x20)
126 memset(&tx_buffer[1], 0x00, 14);
127
128 // Single DMA TransmitReceive
129 if (HAL_SPI_TransmitReceive_DMA(dev->hspi, tx_buffer, dev->read_buffer, 15) != HAL_OK) {
130 return false;
131 }
132
133 return true;
134}

136void ASM330LHHX_ParseDMABuffer(ASM330LHHX_t *dev) {
137 if (!dev) {
138 return;
139 }
140
141 // Parse received buffer (skip first byte which is command echo)
142 dev->temp_raw = (int16_t)((dev->read_buffer[2] << 8) | dev->read_buffer[1]);
143
144 dev->accel_raw[0] = (int16_t)((dev->read_buffer[4] << 8) | dev->read_buffer[3]);
145 dev->accel_raw[1] = (int16_t)((dev->read_buffer[6] << 8) | dev->read_buffer[5]);
146 dev->accel_raw[2] = (int16_t)((dev->read_buffer[8] << 8) | dev->read_buffer[7]);
147
148 dev->gyro_raw[0] = (int16_t)((dev->read_buffer[10] << 8) | dev->read_buffer[9]);
149 dev->gyro_raw[1] = (int16_t)((dev->read_buffer[12] << 8) | dev->read_buffer[11]);
150 dev->gyro_raw[2] = (int16_t)((dev->read_buffer[14] << 8) | dev->read_buffer[13]);
151
152 dev->data_ready = 1;
153}

155bool ASM330LHHX_ProcessData(ASM330LHHX_t *dev, IMU_t *output) {

163 output->accel_z = asm330lhhx_from_fs2g_to_mg(dev->accel_raw[2]);
164
165 output->gyro_x = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[0]);
166 output->gyro_y = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[1]);
167 output->gyro_z = asm330lhhx_from_fs2000dps_to_mdps(dev->gyro_raw[2]);
168
169 output->temperature_c = asm330lhhx_from_lsb_to_celsius(dev->temp_raw);
170 output->timestamp_ms = HAL_GetTick();
171
172 dev->data_ready = 0;
173

166void fsm_report_init_status(const char *component, bool success) {
167 if (fsm_mutex && xSemaphoreTake(fsm_mutex, pdMS_TO_TICKS(50)) != pdTRUE) {
168 printf("[FSM] WARNING: Could not lock mutex for init report\r\n");
169 return;
170 }
171
172 boot_status_t *bs = &fsm_ctx.boot_status;
173 int8_t status = success ? 1 : 0;
174
175 if (strcmp(component, "IMU_INIT") == 0) bs->imu_init = status;
176 else if (strcmp(component, "IMU_CONFIG") == 0) bs->imu_config = status;
177 else if (strcmp(component, "BARO_INIT") == 0) bs->baro_init = status;
178 else if (strcmp(component, "BARO_CONFIG") == 0) bs->baro_config = status;
179 else if (strcmp(component, "MAG_INIT") == 0) bs->mag_init = status;
180 else if (strcmp(component, "MAG_CONFIG") == 0) bs->mag_config = status;
181 else if (strcmp(component, "BNO_INIT") == 0) bs->bno_init = status;
182 else if (strcmp(component, "BNO_CONFIG") == 0) bs->bno_config = status;
183 else if (strcmp(component, "GPS_INIT") == 0) bs->gps_init = status;
184 else if (strcmp(component, "GPS_CONFIG") == 0) bs->gps_config = status;
185 else if (strcmp(component, "SD_CARD") == 0) bs->sd_card_init = status;
186 else if (strcmp(component, "RADIO") == 0) bs->radio_init = status;
187 else if (strcmp(component, "DATA_HANDLER") == 0) bs->data_handler_init = status;
188
189 bs->total_checks++;
190 if (success) {
191 bs->passed_checks++;
192 } else {
193 bs->failed_checks++;
194 if (strstr(component, "IMU") || strstr(component, "BARO")) {
195 bs->critical_failures++;
196 }
197 }
198
199 if (fsm_mutex) xSemaphoreGive(fsm_mutex);
200 printf("[FSM] Init report: %s = %s\r\n", component, success ? "OK" : "FAIL");
201}

203void fsm_report_thread_started(const char *thread_name) {
204 if (fsm_mutex && xSemaphoreTake(fsm_mutex, pdMS_TO_TICKS(50)) != pdTRUE) {

210 if (strcmp(thread_name, "SENSORS") == 0) bs->sensors_thread_started = 1;
211 else if (strcmp(thread_name, "DATA_HANDLER") == 0) bs->data_handler_thread_started = 1;
212 else if (strcmp(thread_name, "TELEMETRY") == 0) bs->telemetry_thread_started = 1;
213 else if (strcmp(thread_name, "RADIO") == 0) bs->radio_thread_started = 1;
214 else if (strcmp(thread_name, "SD_CARD") == 0) bs->sd_card_thread_started = 1;
215 else if (strcmp(thread_name, "ESTIMATOR") == 0) bs->estimator_thread_started = 1;
216 else if (strcmp(thread_name, "CONTROLLER") == 0) bs->controller_thread_started = 1;
217
218 if (fsm_mutex) xSemaphoreGive(fsm_mutex);
219 printf("[FSM] Thread started: %s\r\n", thread_name);
220}

240bool fsm_is_boot_complete(void) {
241 boot_status_t *bs = &fsm_ctx.boot_status;
242
243 if (bs->sensors_thread_started != 1) return false;
244 if (bs->data_handler_thread_started != 1) return false;
245 if (bs->telemetry_thread_started != 1) return false;
246 if (bs->radio_thread_started != 1) return false;
247 if (bs->imu_init != 1) return false;
248 if (bs->baro_init != 1) return false;
249
250 return true;

264bool fsm_send_command(fsm_command_t cmd, uint8_t cmd_seq, const void *payload, uint16_t size) {
265 if (queue_cmd_to_fsm == NULL) {
266 printf("[FSM] ERROR: queue_cmd_to_fsm is NULL\r\n");
267 return false;
268 }
269
270 fsm_cmd_msg_t msg;
271 memset(&msg, 0, sizeof(msg));
272 msg.cmd = cmd;
273 msg.cmd_seq = cmd_seq;
274
275 if (payload && size > 0) {
276 uint16_t copy_size = (size > sizeof(msg.payload)) ? sizeof(msg.payload) : size;
277 memcpy(&msg.payload, payload, copy_size);
278 }
279
280 if (xQueueSend(queue_cmd_to_fsm, &msg, pdMS_TO_TICKS(10)) != pdTRUE) {
281 printf("[FSM] ERROR: Failed to queue command\r\n");
282 return false;
283 }
284
285 return true;
286}

293

296 msg.type = event_type;

298
299 if (data) {
300 memcpy(&msg.data, data, sizeof(msg.data));
301 }
302
303 if (xQueueSend(queue_event_to_fsm, &msg, pdMS_TO_TICKS(10)) != pdTRUE) {
304 printf("[FSM] ERROR: Failed to queue event\r\n");

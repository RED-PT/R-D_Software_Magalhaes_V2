120static float pid_compute(float error, float *integral, float *prev_error,
121 const pid_gains_t *gains, float dt) {
122 // Proportional
123 float p_term = gains->kp * error;
124
125 // Integral with anti-windup
126 *integral += error * dt;
127 if (*integral > gains->integral_limit) *integral = gains->integral_limit;
128 if (*integral < -gains->integral_limit) *integral = -gains->integral_limit;
129 float i_term = gains->ki * (*integral);
130
131 // Derivative
132 float derivative = (error - *prev_error) / dt;
133 float d_term = gains->kd * derivative;
134 *prev_error = error;
135
136 return p_term + i_term + d_term;
137}

139static float compute_altitude_control(float target_alt, float current_alt, float dt) {
140 float error = target_alt - current_alt;
141
142 float output = pid_compute(error,
143 &ctrl_state.altitude_error_integral,
144 &ctrl_state.altitude_error_prev,
145 &altitude_pid, dt);
146
147 // Add feedforward for hover (gravity compensation)
148 // This is vehicle-specific - adjust for your thrust-to-weight ratio
149 float hover_throttle = 0.5f; // ~50% throttle to hover
150
151 return hover_throttle + output;
152}
153
154static float compute_velocity_control(float target_vel, float current_vel, float dt) {
155 float error = target_vel - current_vel;
156
157 float output = pid_compute(error,
158 &ctrl_state.velocity_error_integral,
159 &ctrl_state.velocity_error_prev,
160 &velocity_pid, dt);
161
162 // Add feedforward
163 float hover_throttle = 0.5f;
164
165 return hover_throttle + output;
166}

172static void run_control_loop(float dt) {
173 if (ctrl_state.emergency_stop) {
174 PWM_EmergencyStop();
175 return;
176 }
177
178 float throttle_cmd = 0.0f;
179
180 switch (ctrl_state.mode) {
181 case CTRL_MODE_IDLE:
182 throttle_cmd = 0.0f;
183 break;
184
185 case CTRL_MODE_MANUAL:
186 throttle_cmd = ctrl_state.manual_throttle;
187 break;
188
189 case CTRL_MODE_ALTITUDE_HOLD:
190 throttle_cmd = compute_altitude_control(
191 ctrl_state.target_altitude_m,
192 ctrl_state.current_altitude_m,
193 dt
194 );
195 break;
196
197 case CTRL_MODE_VELOCITY_CTRL:
198 throttle_cmd = compute_velocity_control(
199 ctrl_state.target_velocity_ms,
200 ctrl_state.current_velocity_ms,
201 dt
202 );
203 break;
204
205 case CTRL_MODE_LANDING:
206 // Landing uses velocity control with decreasing target
207 // Target velocity is negative (descending)
208 throttle_cmd = compute_velocity_control(
209 -fsm_ctx.profile.touchdown_velocity_ms,
210 ctrl_state.current_velocity_ms,
211 dt
212 );
213 break;
214 }
215
216 // Apply limits (normalized 0-1)
217 if (throttle_cmd < ctrl_state.throttle_min) throttle_cmd = ctrl_state.throttle_min;
218 if (throttle_cmd > ctrl_state.throttle_max) throttle_cmd = ctrl_state.throttle_max;
219
220 ctrl_state.throttle_output = throttle_cmd;
221
222 if (ctrl_state.motor_initialized) {
223 // Convert 0.0-1.0 to 0-100%
224 PWM_SetThrottle(throttle_cmd * 100.0f);
225 }
226}

232void controller_start_ramp(float target_throttle, float duration_s) {
233 // Calculate rate based on duration and control loop period
234 float steps = (duration_s * 1000.0f) / CONTROL_LOOP_PERIOD_MS;
235 float rate = (target_throttle * 100.0f) / steps; // Rate in % per step
236
237 printf("[CTRL] Starting ramp to %.1f%% over %.1fs\r\n",
238 target_throttle * 100.0f, duration_s);
239
240 PWM_StartRamp(target_throttle * 100.0f, rate);
241 ctrl_state.mode = CTRL_MODE_MANUAL;
242}

255void controller_thread_function() {
256 printf("[CTRL] Controller Thread started\r\n");
257 fsm_report_thread_started("CONTROLLER");
258
259 // Initialize state
260 memset(&ctrl_state, 0, sizeof(ctrl_state));
261 ctrl_state.mode = CTRL_MODE_IDLE;
262 ctrl_state.throttle_min = 0.0f;
263 ctrl_state.throttle_max = 1.0f;
264
265 TickType_t last_loop_tick = xTaskGetTickCount();
266 TickType_t last_stats_tick = xTaskGetTickCount();
267
268 // Thread starts suspended, waiting for FSM to resume us
269 printf("[CTRL] Suspending until armed...\r\n");
270 vTaskSuspend(NULL);
271
272 printf("[CTRL] Resumed - control active\r\n");
273
274 // Get limits from profile
275 ctrl_state.throttle_min = fsm_ctx.profile.throttle_min;
276 ctrl_state.throttle_max = fsm_ctx.profile.throttle_max;
277
278 while (1) {
279 TickType_t now = xTaskGetTickCount();
280
281 // Calculate dt for PID
282 float dt = (float)(now - last_loop_tick) / 1000.0f;
283 if (dt < 0.001f) dt = 0.001f; // Minimum dt to avoid divide by zero
284 last_loop_tick = now;
285
286 // Update limits from FSM profile (in case they changed)
287 ctrl_state.throttle_min = fsm_ctx.profile.throttle_min;
288 ctrl_state.throttle_max = fsm_ctx.profile.throttle_max;
289
290 // Update ramp if active
291 controller_update_ramp();
292
293 // Run control loop
294 run_control_loop(dt);
295
296 // Periodic stats
297 if ((now - last_stats_tick) >= pdMS_TO_TICKS(5000)) {
298 printf("[CTRL] Mode=%d Alt=%.1f/%.1f Vel=%.2f Thr=%.1f%%\r\n",
299 ctrl_state.mode,
300 ctrl_state.current_altitude_m,
301 ctrl_state.target_altitude_m,
302 ctrl_state.current_velocity_ms,
303 ctrl_state.throttle_output * 100.0f);
304 last_stats_tick = now;
305 }
306
307 vTaskDelay(pdMS_TO_TICKS(CONTROL_LOOP_PERIOD_MS));
308 }

31void HAL_GPIO_EXTI_Callback(uint16_t GPIO_Pin) {
32 if (osKernelGetState() != osKernelRunning) return;
33
34 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
35
36 // Sensor DRDY pins
37 if (GPIO_Pin == EXTI_IMU_PIN) {
38 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_IMU_DRDY,
39 eSetBits, &xHigherPriorityTaskWoken);
40 }
41 else if (GPIO_Pin == EXTI_MAG_PIN) {
42 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_MAG_DRDY,
43 eSetBits, &xHigherPriorityTaskWoken);
44 }
45 // SX126x DIO1 interrupt (PB2)
46 else if (GPIO_Pin == EXTI_LORA_PIN) {
47 SX126x_DIO1_IRQ_Handler();
48 }
49
50 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
51}

57void HAL_SPI_TxRxCpltCallback(SPI_HandleTypeDef *hspi) {
58 if (osKernelGetState() != osKernelRunning) return;
59
60 BaseType_t xHigherPriorityTaskWoken = pdFALSE;
61
62 // Parse DMA buffer based on active sensor
63 if (hspi->Instance == SPI_IMU_BARO_INSTANCE) {
64 if (spi1_active_sensor == ACTIVE_SENSOR_IMU) {
65 ASM330LHHX_ParseDMABuffer(&imu_device);
66 }
67 }
68 else if (hspi->Instance == SPI_MAG_INSTANCE) {
69 if (spi3_active_sensor == ACTIVE_SENSOR_MAG) {
70 MMC5983MA_ParseDMABuffer(&mag_device);
71 }
72 }
73
74 // Notify sensors thread
75 xTaskNotifyFromISR(sensors_thread_id, SENSOR_NOTIFY_DMA_COMPLETE,
76 eSetBits, &xHigherPriorityTaskWoken);
77
78 portYIELD_FROM_ISR(xHigherPriorityTaskWoken);
79}

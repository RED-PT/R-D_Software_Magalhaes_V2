203void DC_to_Period(float percentage_motor) {
204 // Timer configured so 1 count = 1µs
205 // CCR value directly equals pulse width in microseconds
206 // Map 0-100% throttle → PULSE_MIN_US to PULSE_MAX_US
207 uint32_t pulse_us = PULSE_MIN_US + (uint32_t)((percentage_motor / 100.0f) * (PULSE_MAX_US - PULSE_MIN_US));
208
209 // Clamp to valid ESC range
210 if (pulse_us < PULSE_MIN_US) pulse_us = PULSE_MIN_US;
211 if (pulse_us > PULSE_MAX_US) pulse_us = PULSE_MAX_US;
212
213 // Apply to ESC channel (per config.h macro)
214 PWM_ESC_CHANNEL_WRITE = pulse_us;
215}

264void PWM_StartRamp(float target_percentage, float rate) {
265 // Clamp target
266 if (target_percentage < THROTTLE_MIN) target_percentage = THROTTLE_MIN;
267 if (target_percentage > THROTTLE_MAX) target_percentage = THROTTLE_MAX;
268
269 ramp_target = target_percentage;
270 ramp_rate = rate;
271 ramp_direction = (target_percentage > currentThrottle) ? 1.0f : -1.0f;
272 ramp_active = true;
273
274 printf("[PWM] Starting ramp: %.1f%% -> %.1f%% (rate=%.2f)\r\n",
275 currentThrottle, target_percentage, rate);
276}

278bool PWM_RampUpdate(void) {
279 if (!ramp_active) {
280 return true; // No ramp in progress, consider "complete"
281 }
282
283 // Calculate new throttle
284 float new_throttle = currentThrottle + (ramp_direction * ramp_rate);
285
286 // Check if we've reached or passed target
287 if (ramp_direction > 0) {
288 // Ramping up
289 if (new_throttle >= ramp_target) {
290 new_throttle = ramp_target;
291 ramp_active = false;
292 }
293 } else {
294 // Ramping down
295 if (new_throttle <= ramp_target) {
296 new_throttle = ramp_target;
297 ramp_active = false;
298 }
299 }
300
301 PWM_SetThrottle(new_throttle);
302
303 return !ramp_active;
304}

347bool PWM_ESC_StartCalibration(uint32_t phase1_ms) {
348 // Cannot start calibration if already in progress
349 if (esc_cal_ctx.state == ESC_CAL_PHASE1_MAX ||
350 esc_cal_ctx.state == ESC_CAL_PHASE2_MIN) {
351 printf("[PWM] ESC calibration already in progress!\r\n");
352 return false;
353 }
354
355 // Ensure PWM timer is running before calibration
356 PWM_Init();
357
358 // Initialize calibration context
359 esc_cal_ctx.state = ESC_CAL_PHASE1_MAX;
360 esc_cal_ctx.phase_start_tick = HAL_GetTick();
361 esc_cal_ctx.phase1_duration_ms = (phase1_ms > 0) ? phase1_ms : ESC_CAL_PHASE1_DEFAULT_MS;
362 esc_cal_ctx.phase2_duration_ms = ESC_CAL_PHASE2_DEFAULT_MS;
363 esc_cal_ctx.user_acknowledged = false;
364
365 // Cancel any ongoing ramp
366 ramp_active = false;
367
368 // Set throttle to MAXIMUM
369 printf("[PWM] ESC CALIBRATION STARTED\r\n");
370 printf("[PWM] PHASE 1: Sending MAX throttle (100%%)\r\n");
371 printf("[PWM] >>> POWER CYCLE the ESC NOW! <<<\r\n");
372 printf("[PWM] Waiting %lu ms for ESC to beep twice...\r\n", esc_cal_ctx.phase1_duration_ms);
373
374 PWM_SetThrottle(100.0f);
375
376 return true;
377}

379esc_calibration_state_t PWM_ESC_CalibrationUpdate(void) {
380 if (esc_cal_ctx.state == ESC_CAL_IDLE ||
381 esc_cal_ctx.state == ESC_CAL_COMPLETE ||
382 esc_cal_ctx.state == ESC_CAL_FAILED) {
383 return esc_cal_ctx.state;
384 }
385
386 uint32_t elapsed = HAL_GetTick() - esc_cal_ctx.phase_start_tick;
387
388 switch (esc_cal_ctx.state) {
389 case ESC_CAL_PHASE1_MAX:
390 // Check if user acknowledged or timeout elapsed
391 if (esc_cal_ctx.user_acknowledged || elapsed >= esc_cal_ctx.phase1_duration_ms) {
392 // Transition to Phase 2: MIN throttle
393 printf("[PWM] PHASE 2: Sending MIN throttle (0%%)\r\n");
394 printf("[PWM] ESC should beep cell count then long beep...\r\n");
395
396 PWM_SetThrottle(0.0f);
397
398 esc_cal_ctx.state = ESC_CAL_PHASE2_MIN;
399 esc_cal_ctx.phase_start_tick = HAL_GetTick();
400 }
401 break;
402
403 case ESC_CAL_PHASE2_MIN:
404 // Wait for phase 2 duration to complete
405 if (elapsed >= esc_cal_ctx.phase2_duration_ms) {
406 printf("[PWM] ESC CALIBRATION COMPLETE!\r\n");
407 printf("[PWM] ESC should now be calibrated for 0-100%% throttle range.\r\n");
408
409 esc_cal_ctx.state = ESC_CAL_COMPLETE;
410 }
411 break;
412
413 default:
414 break;
415 }
416
417 return esc_cal_ctx.state;
418}

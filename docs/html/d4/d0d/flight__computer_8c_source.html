365void fsm_report_init_status(const char *component, bool success) {
366 if (fsm_mutex && xSemaphoreTake(fsm_mutex, pdMS_TO_TICKS(50)) != pdTRUE) {
367 printf("[FSM] WARNING: Could not lock mutex for init report\r\n");
368 return;
369 }
370
371 boot_status_t *bs = &fsm_ctx.boot_status;
372 int8_t status = success ? 1 : 0;
373
374 if (strcmp(component, "IMU_INIT") == 0) bs->imu_init = status;
375 else if (strcmp(component, "IMU_CONFIG") == 0) bs->imu_config = status;
376 else if (strcmp(component, "BARO_INIT") == 0) bs->baro_init = status;
377 else if (strcmp(component, "BARO_CONFIG") == 0) bs->baro_config = status;
378 else if (strcmp(component, "MAG_INIT") == 0) bs->mag_init = status;
379 else if (strcmp(component, "MAG_CONFIG") == 0) bs->mag_config = status;
380 else if (strcmp(component, "BNO_INIT") == 0) bs->bno_init = status;
381 else if (strcmp(component, "BNO_CONFIG") == 0) bs->bno_config = status;
382 else if (strcmp(component, "GPS_INIT") == 0) bs->gps_init = status;
383 else if (strcmp(component, "GPS_CONFIG") == 0) bs->gps_config = status;
384 else if (strcmp(component, "SD_CARD") == 0) bs->sd_card_init = status;
385 else if (strcmp(component, "RADIO") == 0) bs->radio_init = status;
386 else if (strcmp(component, "DATA_HANDLER") == 0) bs->data_handler_init = status;
387
388 bs->total_checks++;
389 if (success) {
390 bs->passed_checks++;
391 } else {
392 bs->failed_checks++;
393 if (strstr(component, "IMU") || strstr(component, "BARO")) {
394 bs->critical_failures++;
395 }
396 }
397
398 if (fsm_mutex) xSemaphoreGive(fsm_mutex);
399 printf("[FSM] Init report: %s = %s\r\n", component, success ? "OK" : "FAIL");
400}

509bool fsm_is_boot_complete(void) {
510 boot_status_t *bs = &fsm_ctx.boot_status;
511

513 if (bs->data_handler_thread_started != 1) return false;
514 if (bs->telemetry_thread_started != 1) return false;
515 if (bs->radio_thread_started != 1) return false;

578bool fsm_send_command(fsm_command_t cmd, uint8_t cmd_seq, const void *payload, uint16_t size) {
579 if (queue_cmd_to_fsm == NULL) {
580 printf("[FSM] ERROR: queue_cmd_to_fsm is NULL\r\n");
581 return false;
582 }
583
584 fsm_cmd_msg_t msg;
585 memset(&msg, 0, sizeof(msg));
586 msg.cmd = cmd;
587 msg.cmd_seq = cmd_seq;
588
589 if (payload && size > 0) {
590 uint16_t copy_size = (size > sizeof(msg.payload)) ? sizeof(msg.payload) : size;
591 memcpy(&msg.payload, payload, copy_size);
592 }
593
594 if (xQueueSend(queue_cmd_to_fsm, &msg, pdMS_TO_TICKS(10)) != pdTRUE) {
595 printf("[FSM] ERROR: Failed to queue command\r\n");
596 return false;
597 }
598
599 return true;
600}

624bool fsm_send_event(fsm_internal_event_t event_type, const void *data) {

629

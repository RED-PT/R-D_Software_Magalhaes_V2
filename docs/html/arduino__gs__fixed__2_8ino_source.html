1/**
2 * @file arduino_gs_fixed_2.ino
3 * @brief Ground Station firmware for the Magalhaes Flight Computer
4 * @author Tomas Teixeira
5 * @date October 2025
6 *
7 * This Arduino sketch implements the Ground Station (GS) that communicates
8 * with the Magalhaes Flight Computer via E22 LoRa radio and relays data
9 * to a PC dashboard via USB Serial.
10 *
11 * @section gs_overview Overview
12 * The Ground Station performs:
13 * - Receive telemetry from Flight Computer (slots 0-8)
14 * - Forward telemetry to PC dashboard via framed serial protocol
15 * - Send commands from PC to Flight Computer (slot 9)
16 * - Maintain TDMA synchronization with FC
17 * - Command acknowledgment tracking
18 *
19 * @section gs_hardware Hardware
20 * - Arduino Nano/Uno (ATmega328P)
21 * - E22 LoRa module on SoftwareSerial (pins 2,3)
22 * - E22 mode control (M0=pin 4, M1=pin 5)
23 * - USB Serial to PC at 115200 baud
24 *
25 * @section gs_protocol PC Communication Protocol
26 * All messages to PC use framed format:
27 * | Sync1 | Sync2 | LenH | LenL | Type | Payload... | CRCH | CRCL |
28 * |-------|-------|------|------|------|------------|------|------|
29 * | 0xAA | 0x55 | | | | | | |
30 *
31 * @section gs_commands PC Commands (single character)
32 * | Char | Command | Description |
33 * |------|---------|-------------|
34 * | P | PING | Test connectivity |
35 * | C | CALIBRATE | Calibrate barometer |
36 * | M | MOTOR_CAL | ESC calibration mode |
37 * | E | STATIC_TEST | Static thrust test (E20 = 20% max) |
38 * | 1/2/3 | PROFILE | Select flight profile |
39 * | A | ARM | Arm for launch |
40 * | D | DISARM | Disarm |
41 * | T | TEST | Start test sequence |
42 * | L | LAUNCH | Launch command |
43 * | X | ABORT | Abort mission |
44 * | S | SAFE | Force safe mode |
45 * | R | RESET_STATS | Reset statistics |
46 * | ? | DEBUG | Toggle debug output |
47 *
48 * @section gs_tdma TDMA Structure
49 * Synchronized to Flight Computer timing:
50 * - Slots 0-7: Receive fast telemetry
51 * - Slot 8: Receive slow telemetry
52 * - Slot 9: Transmit commands or sync beacon
53 */

414/**
415 * @defgroup CRC CRC-16 Calculation
416 * @{
417 */
418
419/**
420 * @brief Calculate CRC-16 (Modbus polynomial)
421 *
422 * Same algorithm as STM32 for compatibility.
423 *
424 * @param[in] data Pointer to data buffer
425 * @param[in] length Number of bytes
426 *
427 * @return CRC-16 value
428 */
429uint16_t crc16_calc(const uint8_t *data, uint16_t length) {
430 uint16_t crc = 0xFFFF;
431 for (uint16_t i = 0; i < length; i++) {
432 crc ^= (uint16_t)data[i];
433 for (uint8_t j = 0; j < 8; j++) {
434 if (crc & 0x0001) {
435 crc = (crc >> 1) ^ 0xA001;
436 } else {
437 crc >>= 1;
438 }
439 }
440 }
441 return crc;
442}

451/**
452 * @brief Send framed message to PC
453 *
454 * Wraps payload in framed format with sync bytes and CRC.
455 *
456 * @param[in] msgType Message type identifier
457 * @param[in] data Pointer to payload data
458 * @param[in] len Payload length
459 */
460void sendToPC(uint8_t msgType, const void* data, uint16_t len) {
461 uint16_t totalLen = len + 1;
462
463 uint8_t crc_buf[64];
464 crc_buf[0] = msgType;
465 if (len > 0 && data != NULL && len < 63) {
466 memcpy(&crc_buf[1], data, len);
467 }
468 uint16_t crc = crc16_calc(crc_buf, totalLen);
469
470 PC_SERIAL.write(FRAME_SYNC_1);
471 PC_SERIAL.write(FRAME_SYNC_2);
472 PC_SERIAL.write((uint8_t)(totalLen >> 8));
473 PC_SERIAL.write((uint8_t)(totalLen & 0xFF));
474 PC_SERIAL.write(msgType);
475 if (len > 0 && data != NULL) {
476 PC_SERIAL.write((const uint8_t*)data, len);
477 }
478 PC_SERIAL.write((uint8_t)(crc >> 8));
479 PC_SERIAL.write((uint8_t)(crc & 0xFF));
480}

582void sim_generate_fast_telemetry() {
583 TelemetryFast_t pkt = {0};
584
585 pkt.packet_type = TELEM_PACKET_FAST;
586 pkt.frame_id = frame_id;
587 pkt.slot_id = 0;
588 pkt.seq = sim_seq++;
589 pkt.flags = 0x01;
590 pkt.time = millis();
591 pkt.state = sim_state;
592 pkt.substate = sim_substate;
593 pkt.last_cmd_seq_acked = awaiting_ack_seq;
594 pkt.last_cmd_status = 0;
595
596 float t = millis() / 1000.0f;
597
598 sim_altitude = 100.0f + 50.0f * sin(t * 0.5f);
599 pkt.altitude = (int16_t)(sim_altitude * 10);
600 pkt.vario = (int16_t)(25.0f * cos(t * 0.5f) * 100);
601
602 sim_pitch = 5.0f * sin(t * 0.3f);
603 sim_roll = 3.0f * cos(t * 0.4f);
604 float sim_yaw = fmod(t * 10.0f, 360.0f);
605 pkt.pitch = (int16_t)(sim_pitch * 10);
606 pkt.roll = (int16_t)(sim_roll * 10);
607 pkt.yaw = (int16_t)(sim_yaw * 10);
608
609 pkt.accel_x = (int16_t)(50 * sin(t * 2.0f));
610 pkt.accel_y = (int16_t)(30 * cos(t * 1.5f));
611 pkt.accel_z = (int16_t)(1000 + 20 * sin(t));
612
613 pkt.gyro_x = (int16_t)(100 * sin(t * 0.8f));
614 pkt.gyro_y = (int16_t)(80 * cos(t * 0.7f));
615 pkt.gyro_z = (int16_t)(50 * sin(t * 0.5f));
616
617 pkt.crc16 = crc16_calc((uint8_t*)&pkt, sizeof(pkt) - 2);
618
619 if (awaiting_ack) {
620 awaiting_ack = false;
621 cmd_pending = false;
622 stats.ack_ok++;
623
624 uint8_t ack_data[2] = {awaiting_ack_seq, 0};
625 sendToPC(MSG_TYPE_GS_ACK, ack_data, 2);
626 }
627
628 sendToPC(MSG_TYPE_FC_FAST, &pkt, sizeof(pkt));
629 stats.rx_fast++;
630}

678void sim_update() {
679 uint32_t now = millis();
680
681 if ((now - last_sim_fast_ms) >= SIM_FAST_INTERVAL_MS) {
682 sim_generate_fast_telemetry();
683 last_sim_fast_ms = now;
684 sim_handle_ping();
685 }

836/**
837 * @brief Process received FC packet
838 *
839 * Validates CRC, extracts data, forwards to PC, handles ACKs.
840 *
841 * @param[in] data Packet data buffer
842 * @param[in] len Packet length
843 * @param[in] type Packet type
844 */
845void process_fc_packet(uint8_t* data, uint16_t len, uint8_t type) {
846 // STM32 sends CRC as little-endian (low byte first)
847 uint16_t pkt_crc = (uint16_t)data[len-2] | ((uint16_t)data[len-1] << 8);
848
849 uint16_t calc_crc = crc16_calc(data, len - 2);
850
851 if (pkt_crc != calc_crc) {
852 stats.crc_errors++;
853 if (debug_enabled) {
854 PC_SERIAL.print(F("[CRC ERR] type="));
855 PC_SERIAL.print(type, HEX);
856 PC_SERIAL.print(F(" got="));
857 PC_SERIAL.print(pkt_crc, HEX);
858 PC_SERIAL.print(F(" calc="));
859 PC_SERIAL.println(calc_crc, HEX);
860 debugHex(data, len);
861 }
862 return;
863 }
864
865 if (debug_enabled) {
866 PC_SERIAL.print(F("[RX OK] type="));
867 PC_SERIAL.print(type, HEX);
868 PC_SERIAL.print(F(" len="));
869 PC_SERIAL.println(len);
870 }
871
872 uint8_t msgType;
873 switch (type) {
874 case TELEM_PACKET_FAST:
875 msgType = MSG_TYPE_FC_FAST;
876 stats.rx_fast++;
877
878 // Check for command ACK in fast packet
879 if (awaiting_ack && data[11] == awaiting_ack_seq) {
880 awaiting_ack = false;
881 cmd_pending = false;
882 stats.ack_ok++;
883
884 uint8_t ack_data[2] = {data[11], data[12]};
885 sendToPC(MSG_TYPE_GS_ACK, ack_data, 2);
886
887 if (debug_enabled) {
888 PC_SERIAL.print(F("[ACK] seq="));
889 PC_SERIAL.println(data[11]);
890 }
891 }
892 break;
893
894 case TELEM_PACKET_SLOW:
895 msgType = MSG_TYPE_FC_SLOW;
896 stats.rx_slow++;
897 break;
898
899 case TELEM_PACKET_EVENT:
900 msgType = MSG_TYPE_FC_EVENT;
901 stats.rx_event++;
902
903 // Check for pong in event packet
904 if (data[8] == 13 && ping_awaiting_pong) {
905 uint8_t pong_seq = data[11];
906 if (pong_seq == ping_seq_sent) {
907 uint32_t rtt = millis() - ping_sent_time;
908 ping_awaiting_pong = false;
909
910 GSPong_t pong;
911 pong.ping_seq = pong_seq;
912 pong.rtt_ms = rtt;
913 memcpy(&pong.fc_timestamp, &data[12], 4);
914 sendToPC(MSG_TYPE_GS_PONG, &pong, sizeof(pong));
915
916 if (debug_enabled) {
917 PC_SERIAL.print(F("[PONG] RTT="));
918 PC_SERIAL.println(rtt);
919 }
920 }
921 }
922 break;
923
924 default:
925 return;
926 }
927
928 sendToPC(msgType, data, len);
929
930 // Sync TDMA timing from received packet
931 if (!tdma_synced) {
932 tdma_synced = true;
933 frame_start_ms = millis() - (data[2] * TDMA_SLOT_MS);
934 frame_id = data[1];
935 sendLog("TDMA synced!");
936 }
937}

939/**
940 * @brief Process single received byte
941 *
942 * Accumulates bytes into packet, triggers processing when complete.
943 *
944 * @param[in] byte Received byte
945 *
946 * @return true if complete packet processed
947 */
948bool process_rx_byte(uint8_t byte) {
949 uint32_t now = millis();
950
951 // Timeout: reset buffer
952 if (rx_idx > 0 && (now - rx_last_byte_ms) > RX_TIMEOUT_MS) {
953 if (debug_enabled && rx_idx > 0) {
954 PC_SERIAL.print(F("[TIMEOUT] dropped "));
955 PC_SERIAL.print(rx_idx);
956 PC_SERIAL.println(F(" bytes"));
957 }
958 rx_idx = 0;
959 }
960 rx_last_byte_ms = now;
961
962 // First byte: check if valid packet type
963 if (rx_idx == 0) {
964 uint16_t expected = get_packet_size(byte);
965 if (expected == 0) {
966 if (debug_enabled) {
967 PC_SERIAL.print(F("[?] Unknown type: 0x"));
968 PC_SERIAL.println(byte, HEX);
969 }
970 return false;
971 }
972 }
973
974 // Store byte
975 if (rx_idx < RX_BUFFER_SIZE) {
976 rx_buffer[rx_idx++] = byte;
977 } else {
978 rx_idx = 0;
979 return false;
980 }
981
982 // Check for complete packet
983 uint8_t type = rx_buffer[0];
984 uint16_t expected = get_packet_size(type);
985
986 if (rx_idx >= expected) {
987 process_fc_packet(rx_buffer, expected, type);
988 rx_idx = 0;
989 return true;
990 }
991
992 return false;
993}

1261/**
1262 * @brief Arduino main loop
1263 *
1264 * Handles TDMA timing, slot processing, simulation, and PC communication.
1265 */
1266void loop() {
1267 uint32_t now = millis();
1268
1269 // Frame rollover
1270 if ((now - frame_start_ms) >= TDMA_SUPERFRAME_MS) {
1271 advance_frame();
1272 }
1273
1274 // Slot handling
1275 uint8_t slot = get_current_slot();
1276 static uint8_t last_slot = 255;
1277 if (slot != last_slot) {
1278 last_slot = slot;
1279 tx_done_this_slot = false;
1280 }
1281
1282 if (slot == TDMA_TX_SLOT) {
1283 handle_tx_slot();
1284 } else {
1285 handle_rx_slots();
1286 }
1287
1288 #if SIMULATION_MODE
1289 sim_update();
1290 #endif
1291
1292 check_command_timeout();
1293 process_pc_input();
1294
1295 // Periodic status
1296 if ((now - last_status_ms) >= STATUS_INTERVAL_MS) {
1297 send_status();
1298 last_status_ms = now;
1299
1300 if (debug_enabled) {
1301 PC_SERIAL.print(F("[STATS] fast="));
1302 PC_SERIAL.print(stats.rx_fast);
1303 PC_SERIAL.print(F(" slow="));
1304 PC_SERIAL.print(stats.rx_slow);

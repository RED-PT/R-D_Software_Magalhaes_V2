41// Helper: Check if buffer crossed threshold and notify data handler
42static void check_and_notify_threshold(void) {
43 // Only notify if data_handler_thread_id is set (it's running)
44 uint32_t available_imu = ram_circular_buffer_available(&cb_imu);
45 uint32_t available_baro = ram_circular_buffer_available(&cb_baro);
46 uint32_t available_mag = ram_circular_buffer_available(&cb_mag);
47 uint32_t available_bno = ram_circular_buffer_available(&cb_bno);
48 uint32_t available_gps = ram_circular_buffer_available(&cb_gps);
49
50 uint32_t threshold_imu = (cb_imu.capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;
51 uint32_t threshold_baro = (cb_baro.capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;
52 uint32_t threshold_mag = (cb_mag.capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;
53 uint32_t threshold_bno = (cb_bno.capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;
54 uint32_t threshold_gps = (cb_gps.capacity * BUFFER_FULL_THRESHOLD_PCT) / 100;
55
56 if ((available_imu > threshold_imu) ||
57 (available_baro > threshold_baro) ||
58 (available_mag > threshold_mag) ||
59 (available_bno > threshold_bno) ||
60 (available_gps > threshold_gps)) {
61
62 // Notify data handler that a buffer crossed threshold
63 data_handler_notify_threshold();
64 }
65}

97void ram_circular_buffer_init(ram_circular_buffer_t *cb, void *buffer, uint32_t capacity, uint32_t sample_size) {
98 cb->buffer = buffer;
99 cb->capacity = capacity;
100 cb->head = 0;
101 cb->tail = 0;
102 cb->sample_size = sample_size;
103 cb->overflow_count = 0;
104 cb->mutex = xSemaphoreCreateMutex();
105
106 if (cb->mutex == NULL) {
107 printf("ERROR: Failed to create mutex\n");
108 }
109}

111bool ram_circular_buffer_write(ram_circular_buffer_t *cb, const void *data, const void **written_ptr) {
112 // Take mutex with timeout to prevent deadlock
113 if (xSemaphoreTake(cb->mutex, pdMS_TO_TICKS(5)) != pdTRUE) {
114 return false;
115 }
116
117 // Calculate next head position
118 uint32_t next_head = (cb->head + 1) % cb->capacity;
119
120 // Check for overflow
121 if (next_head == cb->tail) {
122 cb->overflow_count++;
123 // Advance tail (overwrite oldest data)
124 cb->tail = (cb->tail + 1) % cb->capacity;
125 }
126
127 // Calculate address in buffer
128 uint8_t *buffer = (uint8_t*)cb->buffer;
129 uint8_t *write_addr = buffer + (cb->head * cb->sample_size);
130
131 // Copy data to buffer
132 memcpy(write_addr, data, cb->sample_size);
133
134 // Return pointer to written data
135 if (written_ptr) {
136 *written_ptr = write_addr;
137 }
138
139 // Update head
140 cb->head = next_head;
141
142 xSemaphoreGive(cb->mutex);
143 return true;
144}

146uint32_t ram_circular_buffer_available(ram_circular_buffer_t *cb) {
147 uint32_t head = cb->head;
148 uint32_t tail = cb->tail;
149
150 if (head >= tail) {
151 return head - tail;
152 } else {
153 return cb->capacity - tail + head;
154 }
155}

157bool ram_circular_buffer_is_empty(ram_circular_buffer_t *cb) {
158 return cb->head == cb->tail;
159}

161static void distribute_data_packet(const data_packet_t *packet) {
162 // Check FSM state for conditional routing
163 // NOTE: Use different names to avoid shadowing the global queue handles!
164 bool should_queue_to_est = fsm_should_queue_to_estimator();
165 bool should_queue_to_sd = fsm_should_queue_to_sd();
166
167 // Send to estimator only if in appropriate state
168 if (should_queue_to_est) {
169 if (packet->type == DATA_TYPE_IMU ||
170 packet->type == DATA_TYPE_BARO ||
171 packet->type == DATA_TYPE_MAG) {
172 xQueueSend(queue_to_estimator, packet, 0);
173 }
174 }
175
176 // Always send to telemetry (for GS monitoring)
177 xQueueSend(queue_to_telemetry, packet, 0);
178
179 // Send to SD only if logging enabled
180 if (should_queue_to_sd) {
181 xQueueSend(queue_to_sd, packet, 0);
182 }
183}

186void data_handler_store_imu(const IMU_t *imu_data) {
187 const void *data_ptr = NULL;
188
189 // Store in RAM circular buffer and get pointer to stored data
190 if (ram_circular_buffer_write(&cb_imu, imu_data, &data_ptr)) {
191 // Create data packet with pointer to data in buffer
192 data_packet_t packet = {
193 .type = DATA_TYPE_IMU,
194 .data_ptr = data_ptr,
195 .timestamp_ms = imu_data->timestamp_ms,
196 .sequence = seq_imu++,
197 .source_cb = &cb_imu
198 };
199
200 // Distribute to consumers
201 distribute_data_packet(&packet);
202
203 // Check if buffer crossed threshold
204 check_and_notify_threshold();
205 }
206}

39void DC_to_Period(float percentage_motor) {
40 // Convert motor percentage (0-100%) to duty cycle
41 // DC_MIN (5%) = 0% throttle, DC_MAX (10%) = 100% throttle
42 float DC = DC_MIN + (percentage_motor / 100.0f) * (DC_MAX - DC_MIN);
43
44 // Clamp to valid range
45 if (DC > DC_MAX) {
46 DC = DC_MAX;
47 } else if (DC < DC_MIN) {
48 DC = DC_MIN;
49 }
50
51 // Calculate CCR value
52 // CCR = (DC / 100) * Period
53 uint32_t ccr = (uint32_t)(DC * htim3.Init.Period / 100.0f);
54
55 // Apply to all channels (TIM3 for ESC)
56 __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_1, ccr);
57 __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_2, ccr);
58 __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_3, ccr);
59 __HAL_TIM_SET_COMPARE(&htim3, TIM_CHANNEL_4, ccr);
60}

79void PWM_MOTOR_ON(float percentage_motor_max) {
80 printf("[PWM] Ramping ON to %.1f%%\r\n", percentage_motor_max);
81
82 for (float pct = 0.0f; pct < percentage_motor_max; pct += RAMP_RATE_NORMAL) {
83 PWM_SetThrottle(pct);
84 HAL_Delay(50);
85 }
86
87 // Ensure we hit the target exactly
88 PWM_SetThrottle(percentage_motor_max);
89 printf("[PWM] Ramp ON complete\r\n");
90}

92void PWM_MOTOR_OFF(float percentage_motor_max, float rate_descida) {
93 printf("[PWM] Ramping OFF from %.1f%% (rate=%.1f)\r\n", percentage_motor_max, rate_descida);
94
95 for (float pct = percentage_motor_max; pct > 0.0f; pct -= rate_descida) {
96 PWM_SetThrottle(pct);
97 HAL_Delay(50);
98 }
99
100 // Ensure we hit zero
101 PWM_SetThrottle(0.0f);
102 printf("[PWM] Ramp OFF complete\r\n");
103}

109void PWM_StartRamp(float target_percentage, float rate) {
110 // Clamp target
111 if (target_percentage < THROTTLE_MIN) target_percentage = THROTTLE_MIN;
112 if (target_percentage > THROTTLE_MAX) target_percentage = THROTTLE_MAX;
113
114 ramp_target = target_percentage;
115 ramp_rate = rate;
116 ramp_direction = (target_percentage > currentThrottle) ? 1.0f : -1.0f;
117 ramp_active = true;
118
119 printf("[PWM] Starting ramp: %.1f%% -> %.1f%% (rate=%.2f)\r\n",
120 currentThrottle, target_percentage, rate);
121}

123bool PWM_RampUpdate(void) {
124 if (!ramp_active) {
125 return true; // No ramp in progress, consider "complete"
126 }
127
128 // Calculate new throttle
129 float new_throttle = currentThrottle + (ramp_direction * ramp_rate);
130
131 // Check if we've reached or passed target
132 if (ramp_direction > 0) {
133 // Ramping up
134 if (new_throttle >= ramp_target) {
135 new_throttle = ramp_target;
136 ramp_active = false;
137 }
138 } else {
139 // Ramping down
140 if (new_throttle <= ramp_target) {
141 new_throttle = ramp_target;
142 ramp_active = false;
143 }
144 }
145
146 PWM_SetThrottle(new_throttle);
147
148 return !ramp_active;
149}

192bool PWM_ESC_StartCalibration(uint32_t phase1_ms) {
193 // Cannot start calibration if already in progress
194 if (esc_cal_ctx.state == ESC_CAL_PHASE1_MAX ||
195 esc_cal_ctx.state == ESC_CAL_PHASE2_MIN) {
196 printf("[PWM] ESC calibration already in progress!\r\n");
197 return false;
198 }
199
200 // Initialize calibration context
201 esc_cal_ctx.state = ESC_CAL_PHASE1_MAX;
202 esc_cal_ctx.phase_start_tick = HAL_GetTick();
203 esc_cal_ctx.phase1_duration_ms = (phase1_ms > 0) ? phase1_ms : ESC_CAL_PHASE1_DEFAULT_MS;
204 esc_cal_ctx.phase2_duration_ms = ESC_CAL_PHASE2_DEFAULT_MS;
205 esc_cal_ctx.user_acknowledged = false;
206
207 // Cancel any ongoing ramp
208 ramp_active = false;
209
210 // Set throttle to MAXIMUM
211 printf("[PWM] ESC CALIBRATION STARTED\r\n");
212 printf("[PWM] PHASE 1: Sending MAX throttle (100%%)\r\n");
213 printf("[PWM] >>> POWER CYCLE the ESC NOW! <<<\r\n");
214 printf("[PWM] Waiting %lu ms for ESC to beep twice...\r\n", esc_cal_ctx.phase1_duration_ms);
215
216 PWM_SetThrottle(100.0f);
217
218 return true;
219}

221esc_calibration_state_t PWM_ESC_CalibrationUpdate(void) {
222 if (esc_cal_ctx.state == ESC_CAL_IDLE ||
223 esc_cal_ctx.state == ESC_CAL_COMPLETE ||
224 esc_cal_ctx.state == ESC_CAL_FAILED) {
225 return esc_cal_ctx.state;
226 }
227
228 uint32_t elapsed = HAL_GetTick() - esc_cal_ctx.phase_start_tick;
229
230 switch (esc_cal_ctx.state) {
231 case ESC_CAL_PHASE1_MAX:
232 // Check if user acknowledged or timeout elapsed
233 if (esc_cal_ctx.user_acknowledged || elapsed >= esc_cal_ctx.phase1_duration_ms) {
234 // Transition to Phase 2: MIN throttle
235 printf("[PWM] PHASE 2: Sending MIN throttle (0%%)\r\n");
236 printf("[PWM] ESC should beep cell count then long beep...\r\n");
237
238 PWM_SetThrottle(0.0f);
239
240 esc_cal_ctx.state = ESC_CAL_PHASE2_MIN;
241 esc_cal_ctx.phase_start_tick = HAL_GetTick();
242 }
243 break;
244
245 case ESC_CAL_PHASE2_MIN:
246 // Wait for phase 2 duration to complete
247 if (elapsed >= esc_cal_ctx.phase2_duration_ms) {
248 printf("[PWM] ESC CALIBRATION COMPLETE!\r\n");
249 printf("[PWM] ESC should now be calibrated for 0-100%% throttle range.\r\n");
250
251 esc_cal_ctx.state = ESC_CAL_COMPLETE;
252 }
253 break;
254
255 default:
256 break;
257 }
258
259 return esc_cal_ctx.state;
260}

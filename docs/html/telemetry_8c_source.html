18 memset(pkt, 0, sizeof(telemetry_fast_t));

20 // Header

57

62 pkt->yaw = (int16_t)(bno->heading_deg * 10.0f);
63 }
64
65 // CRC
66 pkt->crc16 = crc16_calculate((uint8_t*)pkt, sizeof(telemetry_fast_t) - 2);
67
68 return sizeof(telemetry_fast_t);

68// ============================================================================
69// Flight Event Detection
70// ============================================================================
71static void check_flight_events(float altitude_m, float velocity_ms) {
72 // Update FSM context with current nav data
73 fsm_ctx.altitude_agl_m = altitude_m;
74 fsm_ctx.velocity_vertical_ms = velocity_ms;
75
76 // Track max altitude
77 if (altitude_m > max_altitude_m) {
78 max_altitude_m = altitude_m;
79 fsm_ctx.max_altitude_reached_m = max_altitude_m;
80 }
81
82 // Only detect events during flight
83 if (fsm_ctx.state != STATE_FLIGHT) {
84 return;
85 }
86
87 // Liftoff detection
88 if (!liftoff_detected && velocity_ms > LIFTOFF_VELOCITY_THRESHOLD_MS) {
89 liftoff_detected = true;
90 printf("[EST] LIFTOFF detected! vel=%.2f m/s\r\n", velocity_ms);
91 fsm_send_event(FSM_EVT_LIFTOFF, &velocity_ms);
92 }
93
94 // Apogee detection (velocity crosses zero going negative)
95 if (liftoff_detected && !apogee_detected) {
96 if (velocity_ms < 0 && fabsf(velocity_ms) > APOGEE_VELOCITY_THRESHOLD_MS) {
97 apogee_detected = true;
98 printf("[EST] APOGEE detected! alt=%.2f m\r\n", altitude_m);
99 fsm_send_event(FSM_EVT_APOGEE, &altitude_m);
100 }
101 }
102
103 // Flare altitude detection
104 if (apogee_detected && !flare_triggered) {
105 if (altitude_m <= fsm_ctx.profile.flare_altitude_m) {
106 flare_triggered = true;
107 printf("[EST] FLARE altitude reached! alt=%.2f m\r\n", altitude_m);
108 fsm_send_event(FSM_EVT_FLARE_ALT, &altitude_m);
109 }
110 }
111
112 // Touchdown detection
113 if (flare_triggered && !touchdown_detected) {
114 if (altitude_m <= TOUCHDOWN_ALT_THRESHOLD_M) {
115 touchdown_detected = true;
116 touchdown_time = HAL_GetTick();
117 printf("[EST] TOUCHDOWN detected! alt=%.2f m\r\n", altitude_m);
118 fsm_send_event(FSM_EVT_TOUCHDOWN, &altitude_m);
119 }
120 }
121
122 // Landed confirmation (stable on ground for 3 seconds)
123 if (touchdown_detected) {
124 if (fabsf(velocity_ms) < LANDED_VELOCITY_THRESHOLD_MS) {
125 if ((HAL_GetTick() - touchdown_time) > LANDED_TIME_MS) {
126 printf("[EST] LANDED confirmed!\r\n");
127 fsm_send_event(FSM_EVT_LANDED, NULL);
128 }
129 } else {
130 // Reset timer if still moving
131 touchdown_time = HAL_GetTick();
132 }
133 }
134
135 // Safety: Altitude limit exceeded
136 if (altitude_m > fsm_ctx.profile.max_altitude_m) {
137 printf("[EST] WARNING: Max altitude exceeded!\r\n");
138 fsm_send_event(FSM_EVT_ALTITUDE_LIMIT, &altitude_m);
139 }
140
141 // Safety: Velocity limit exceeded
142 if (fabsf(velocity_ms) > fsm_ctx.profile.max_velocity_ms) {
143 printf("[EST] WARNING: Max velocity exceeded!\r\n");
144 fsm_send_event(FSM_EVT_VELOCITY_LIMIT, &velocity_ms);
145 }
146}

160// ============================================================================
161// Main Thread
162// ============================================================================
163void estimator_thread_function() {
164
165 printf("[EST] Estimator Thread started...\r\n");
166 fsm_report_thread_started("ESTIMATOR");
167
168 data_packet_t packet;
169 float current_altitude_m = 0.0f;
170 float current_velocity_ms = 0.0f;
171
172 // Simple velocity estimation from altitude changes
173 float prev_altitude_m = 0.0f;
174 uint32_t prev_time_ms = 0;
175
176 TickType_t last_stats = xTaskGetTickCount();
177
178 while(1) {
179 // Receive sensor data
180 if (xQueueReceive(queue_to_estimator, &packet, pdMS_TO_TICKS(100)) == pdTRUE) {
181 data_packet_lock(&packet);
182
183 switch(packet.type) {
184 case DATA_TYPE_BARO: {
185 BARO_t baro;
186 data_packet_copy_baro(&packet, &baro);
187 current_altitude_m = baro.altitude_m;
188
189 // Simple velocity estimation (derivative of altitude)
190 uint32_t now = HAL_GetTick();
191 if (prev_time_ms > 0) {
192 float dt_s = (now - prev_time_ms) / 1000.0f;
193 if (dt_s > 0.001f) {
194 current_velocity_ms = (current_altitude_m - prev_altitude_m) / dt_s;
195 }
196 }
197 prev_altitude_m = current_altitude_m;
198 prev_time_ms = now;
199 break;
200 }
201
202 case DATA_TYPE_IMU: {
203 // TODO: Use IMU for better state estimation
204 break;
205 }
206
207 default:
208 break;
209 }
210
211 data_packet_unlock(&packet);
212
213 // Check for flight events
214 check_flight_events(current_altitude_m, current_velocity_ms);
215 }
216
217 // Reset detection when entering flight
218 static fsm_state_t prev_state = STATE_BOOT;
219 if (fsm_ctx.state == STATE_FLIGHT && prev_state != STATE_FLIGHT) {
220 reset_flight_detection();
221 }
222 prev_state = fsm_ctx.state;
223
224 // Periodic stats
225 TickType_t now = xTaskGetTickCount();
226 if ((now - last_stats) >= pdMS_TO_TICKS(10000)) {
227 printf("[EST] Alt=%.2f m, Vel=%.2f m/s, MaxAlt=%.2f m\r\n",
228 current_altitude_m, current_velocity_ms, max_altitude_m);
229 last_stats = now;
230 }
231 }
232}

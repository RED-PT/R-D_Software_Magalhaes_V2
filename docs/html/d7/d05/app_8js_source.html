432/**
433 * Pushes new data points to a chart and maintains rolling window
434 *
435 * @param {Chart} chart - Chart.js instance to update
436 * @param {number[]} values - Array of values (one per dataset)
437 *
438 * @description
439 * Adds new data points to all datasets in the chart simultaneously.
440 * Removes oldest points when maxPoints limit is reached to maintain
441 * a rolling window display. Uses 'none' animation for real-time performance.
442 *
443 * @example
444 * // Update gyroscope chart with X, Y, Z values
445 * pushData(charts.gyro, [gyro_x, gyro_y, gyro_z]);
446 */
447function pushData(chart, values) {
448 const label = '';
449 chart.data.labels.push(label);
450 chart.data.datasets.forEach((ds, i) => {
451 ds.data.push(values[i]);
452 });
453 if (chart.data.labels.length > maxPoints) {
454 chart.data.labels.shift();
455 chart.data.datasets.forEach((ds) => ds.data.shift());
456 }
457 chart.update('none');
458}

464/**
465 * Establishes WebSocket connection to the backend server
466 *
467 * @description
468 * Connects to the WebSocket endpoint at /ws on the current host.
469 * Handles incoming messages by routing to appropriate update functions
470 * based on message type. Automatically reconnects on connection close.
471 *
472 * ## Message Types
473 * | Type | Handler | Description |
474 * |------|---------|-------------|
475 * | fast | updateFast() | High-rate telemetry (8 Hz) |
476 * | slow | updateSlow() | Low-rate telemetry (1 Hz) |
477 * | gs_stats | updateStats() | Ground station statistics |
478 * | gs_status | updateStatus() | TDMA sync status |
479 * | event | handleStaticTestEvent() | Flight events |
480 * | pong | - | RTT measurement response |
481 * | log | appendLog() | Server log messages |
482 */
483function connectWebSocket() {
484 ws = new WebSocket(`ws://${location.host}/ws`);
485 ws.addEventListener('open', () => {
486 // Reset rate tracking on connect
487 lastRxTotal = 0;
488 lastRateTime = Date.now();
489 rateHistory = [];
490 });
491
492 ws.addEventListener('close', () => {
493 setTimeout(connectWebSocket, 1000);
494 });
495
496 ws.addEventListener('message', (event) => {
497 const pkt = JSON.parse(event.data);
498 if (pkt.type === 'fast') {
499 updateFast(pkt);
500 } else if (pkt.type === 'slow') {
501 updateSlow(pkt);
502 } else if (pkt.type === 'gs_stats') {
503 updateStats(pkt);
504 } else if (pkt.type === 'gs_status') {
505 updateStatus(pkt);
506 } else if (pkt.type === 'pong') {
507 rttMs = pkt.rtt_ms;
508 document.getElementById('rttValue').textContent = `${pkt.rtt_ms} ms`;
509 appendLog(`[PONG] RTT=${pkt.rtt_ms}ms`);
510 } else if (pkt.type === 'event') {
511 const name = EVENT_NAMES[pkt.event_type] || `EVT_${pkt.event_type}`;
512 console.log('[DEBUG] Received event:', pkt.event_type, name, 'payload length:', pkt.payload ? pkt.payload.length : 0);
513 appendLog(`[EVENT] ${name}`);
514 // Parse boot report from CHECKS_GREEN (4) or CHECKS_RED (5)
515 if (pkt.event_type === 4 && pkt.payload) {
516 parseBootReport(pkt.payload, true); // All checks passed
517 } else if (pkt.event_type === 5 && pkt.payload) {
518 parseBootReport(pkt.payload, false); // Some checks failed
519 }
520 // Handle static test events
521 handleStaticTestEvent(pkt.event_type, pkt.payload);
522 // Handle motor calibration events
523 handleMotorCalEvent(pkt.event_type, pkt.payload);
524 } else if (pkt.type === 'log') {
525 appendLog(`[GS] ${pkt.message}`);
526 }
527 });
528}

664 */
665function updateStats(pkt) {
666 const rxTotal = pkt.rx_fast + pkt.rx_slow + pkt.rx_event;
667 const txTotal = pkt.tx_cmd + pkt.tx_sync;
668 document.getElementById('rxValue').textContent = rxTotal;
669 document.getElementById('txValue').textContent = txTotal;
670 document.getElementById('crcValue').textContent = pkt.crc_errors;
671 document.getElementById('ackOkValue').textContent = pkt.ack_ok;
672 document.getElementById('ackTimeoutValue').textContent = pkt.ack_timeout;
673
674 // Calculate actual data rate with rolling average
675 const now = Date.now();
676 const dt = (now - lastRateTime) / 1000.0; // seconds
677 if (dt >= 0.5) {
678 const delta = rxTotal - lastRxTotal;
679 const bytesPerPkt = 32;
680 const instantRate = Math.max(0, (delta * bytesPerPkt * 8) / (dt * 1000.0)); // kbps
681 console.log(`Rate: delta=${delta}, dt=${dt.toFixed(2)}, instant=${instantRate.toFixed(2)}, history=${rateHistory.length}`);
682 rateHistory.push(instantRate);
683 if (rateHistory.length > RATE_HISTORY_SIZE) {
684 rateHistory.shift();
685 }
686 lastRxTotal = rxTotal;
687 lastRateTime = now;
688 }
689 const kbps = rateHistory.length > 0 ? rateHistory.reduce((a, b) => a + b, 0) / rateHistory.length : 0;
690 const rttText = rttMs ? `${rttMs} ms` : '-- ms';
691 document.getElementById('signalValue').textContent = `${kbps.toFixed(1)} kb/s | ${rttText}`;
692 const signal_icon = document.getElementById('signalIcon');
693 if (kbps > 0.5) {
694 signal_icon.style.color = '#22c55e';
695 } else {
696 signal_icon.style.color = '#ef4444';
697 }
698}

748/**
749 * Parses boot report payload and updates sensor status indicators
750 *
751 * @param {string} payloadHex - Hex-encoded boot report payload
752 * @param {boolean} checksOk - Whether all critical checks passed
753 *
754 * @description
755 * Boot report payload format:
756 * - Bytes 0: total_checks
757 * - Byte 1: passed count
758 * - Byte 2: failed count
759 * - Byte 3: critical failures
760 * - Bytes 4-7: boot_time_ms (uint32)
761 * - Bytes 8+: error strings (8 x 16 bytes)
762 * - Byte 136: error_count
763 *
764 * Parses error strings to identify which sensors failed initialization.
765 */
766function parseBootReport(payloadHex, checksOk) {
767 // Boot report payload format:
768 // Bytes 0: total_checks, 1: passed, 2: failed, 3: critical
769 // Bytes 4-7: boot_time_ms (uint32)
770 // Bytes 8+: error strings (8 x 16 bytes)
771 // Byte 136: error_count
772 if (payloadHex.length < 10) return;
773
774 const total = parseInt(payloadHex.substring(0, 2), 16);
775 const passed = parseInt(payloadHex.substring(2, 4), 16);
776 const failed = parseInt(payloadHex.substring(4, 6), 16);
777 const critical = parseInt(payloadHex.substring(6, 8), 16);
778
779 // Parse error strings to determine which sensors failed
780 const errorCount = payloadHex.length >= 274 ? parseInt(payloadHex.substring(272, 274), 16) : 0;
781
782 // Always parse error messages - even CHECKS_GREEN can have non-critical failures
783 const errors = [];
784 for (let i = 0; i < errorCount && i < 8; i++) {
785 const start = 16 + i * 32; // Each error is 16 bytes = 32 hex chars
786 if (start + 32 <= payloadHex.length) {
787 const errHex = payloadHex.substring(start, start + 32);
788 let errStr = '';
789 for (let j = 0; j < 32; j += 2) {
790 const charCode = parseInt(errHex.substring(j, j + 2), 16);
791 if (charCode > 0 && charCode < 128) errStr += String.fromCharCode(charCode);
792 }
793 if (errStr) errors.push(errStr.trim());
794 }
795 }
796
797 // Update status based on errors found (if no error for sensor, it's OK)
798 updateBootStatus('imu', !errors.some(e => e.includes('IMU')));
799 updateBootStatus('mag', !errors.some(e => e.includes('MAG')));
800 updateBootStatus('baro', !errors.some(e => e.includes('BARO')));
801 updateBootStatus('bno', !errors.some(e => e.includes('BNO')));
802 updateBootStatus('gps', !errors.some(e => e.includes('GPS')));
803
804 if (errors.length > 0) {
805 appendLog(`[BOOT] Errors: ${errors.join(', ')}`);
806 }
807 appendLog(`[BOOT] ${passed}/${total} checks passed, ${critical} critical, boot ${checksOk ? 'OK' : 'FAILED'}`);
808}

849/**
850 * Handles static thrust test events from the flight computer
851 *
852 * @param {number} eventType - Event type ID (19-22 for test events)
853 * @param {string} payload - Hex-encoded event payload
854 *
855 * @description
856 * Processes static thrust test events:
857 * - 19: EVT_STATIC_TEST_STARTED - Initialize test state
858 * - 20: EVT_STATIC_TEST_PROGRESS - Update with new data point
859 * - 21: EVT_STATIC_TEST_COMPLETE - Finalize test
860 * - 22: EVT_STATIC_TEST_FAILED - Handle test failure
861 *
862 * Progress payload format:
863 * - Byte 0: pwm_percent (uint8)
864 * - Bytes 1-4: thrust_n (float32, little-endian)
865 */
866function handleStaticTestEvent(eventType, payload) {
867 const statusEl = document.getElementById('staticTestStatus');
868 const testStatusEl = document.getElementById('testStatus');
869 const testStateEl = document.getElementById('testState');
870
871 switch (eventType) {
872 case 19: // EVT_STATIC_TEST_STARTED
873 console.log('[DEBUG] Static test STARTED event received');
874 testRunning = true;
875 testStartTime = Date.now(); // Track when test started for time calculation
876 testData = [];
877 testMaxThrust = 0;
878 testMaxPWM = 0;
879 if (statusEl) {
880 statusEl.textContent = 'Test running...';
881 statusEl.style.color = '#f59e0b';
882 }
883 if (testStatusEl) {
884 testStatusEl.textContent = 'Test in progress...';
885 testStatusEl.style.color = '#f59e0b';
886 }
887 if (testStateEl) testStateEl.textContent = 'RUNNING';
888 // Clear avg thrust display
889 const avgThrustResetEl = document.getElementById('testAvgThrust');
890 if (avgThrustResetEl) avgThrustResetEl.textContent = '-- N';
891 // Clear chart
892 if (charts.thrust) {
893 charts.thrust.data.datasets[0].data = [];
894 charts.thrust.update('none');
895 }
896 break;
897
898 case 20: // EVT_STATIC_TEST_PROGRESS
899 // Payload: byte 0 = pwm_percent, bytes 1-4 = thrust_n (float, little-endian)
900 console.log('[DEBUG] Progress event payload:', payload, 'length:', payload ? payload.length : 0);
901 if (payload && payload.length >= 10) { // 1 + 4 bytes = 5 bytes = 10 hex chars
902 const pwmPercent = parseInt(payload.substring(0, 2), 16);
903
904 // Parse float from bytes 1-4 (little-endian IEEE 754)
905 const floatBytes = new Uint8Array(4);
906 for (let i = 0; i < 4; i++) {
907 floatBytes[i] = parseInt(payload.substring(2 + i*2, 4 + i*2), 16);
908 }
909 const thrustN = new Float32Array(floatBytes.buffer)[0];
910
911 // Calculate time since test started (client-side)
912 const timestampMs = Date.now() - testStartTime;
913
914 console.log('[DEBUG] Parsed: Time=', timestampMs, 'ms PWM=', pwmPercent, '% Thrust=', thrustN, 'N');
915
916 // Update displays
917 updateAllPWMGauges(pwmPercent);
918 const currentPWMEl = document.getElementById('currentPWM');
919 const currentThrustEl = document.getElementById('currentThrust');
920 const currentTimeEl = document.getElementById('currentTime');
921 if (currentPWMEl) currentPWMEl.textContent = `${pwmPercent}%`;
922 if (currentThrustEl) currentThrustEl.textContent = `${thrustN.toFixed(2)} N`;
923 if (currentTimeEl) currentTimeEl.textContent = `${(timestampMs / 1000).toFixed(1)} s`;
924
925 if (statusEl) {
926 statusEl.textContent = `Test: ${(timestampMs/1000).toFixed(1)}s | ${pwmPercent}% | ${thrustN.toFixed(1)}N`;
927 statusEl.style.color = '#38bdf8';
928 }
929
930 // Add data point with time
931 const dataPoint = { time_ms: timestampMs, pwm: pwmPercent, thrust: thrustN };
932 testData.push(dataPoint);
933
934 // Track max
935 if (thrustN > testMaxThrust) {
936 testMaxThrust = thrustN;
937 testMaxPWM = pwmPercent;
938 }
939
940 // Update chart based on current view mode
941 if (charts.thrust) {
942 const chartPoint = getChartPointForMode(dataPoint);
943 charts.thrust.data.datasets[0].data.push(chartPoint);
944 charts.thrust.update('none');
945 }
946
947 // Update stats
948 const samplesEl = document.getElementById('testSamples');
949 const maxThrustEl = document.getElementById('testMaxThrust');
950 const maxPWMEl = document.getElementById('testMaxPWM');
951 const avgThrustEl = document.getElementById('testAvgThrust');
952 if (samplesEl) samplesEl.textContent = testData.length;
953 if (maxThrustEl) maxThrustEl.textContent = `${testMaxThrust.toFixed(2)} N`;
954 if (maxPWMEl) maxPWMEl.textContent = `${testMaxPWM}%`;
955
956 // Calculate average thrust excluding first 5 seconds
957 if (avgThrustEl) {
958 const samplesAfter5s = testData.filter(d => d.time_ms >= 5000);
959 if (samplesAfter5s.length > 0) {
960 const avgThrust = samplesAfter5s.reduce((sum, d) => sum + d.thrust, 0) / samplesAfter5s.length;
961 avgThrustEl.textContent = `${avgThrust.toFixed(2)} N`;
962 } else {
963 avgThrustEl.textContent = '-- N';
964 }
965 }
966 }
967 break;
968
969 case 21: { // EVT_STATIC_TEST_COMPLETE
970 testRunning = false;
971 // Final average thrust (excluding first 5 seconds)
972 const samplesAfter5s = testData.filter(d => d.time_ms >= 5000);
973 const avgThrust = samplesAfter5s.length > 0
974 ? samplesAfter5s.reduce((sum, d) => sum + d.thrust, 0) / samplesAfter5s.length
975 : 0;
976 const avgThrustFinalEl = document.getElementById('testAvgThrust');
977 if (avgThrustFinalEl) avgThrustFinalEl.textContent = `${avgThrust.toFixed(2)} N`;
978
979 if (statusEl) {
980 statusEl.textContent = `Test complete! ${testData.length} samples | Avg: ${avgThrust.toFixed(2)}N`;
981 statusEl.style.color = '#22c55e';
982 }
983 if (testStatusEl) {
984 testStatusEl.textContent = `Complete! ${testData.length} samples. Max: ${testMaxThrust.toFixed(2)}N | Avg(>5s): ${avgThrust.toFixed(2)}N`;
985 testStatusEl.style.color = '#22c55e';
986 }
987 if (testStateEl) testStateEl.textContent = 'COMPLETE';
988 updateAllPWMGauges(0);
989 break;
990 }
991
992 case 22: // EVT_STATIC_TEST_FAILED
993 testRunning = false;
994 if (statusEl) {
995 statusEl.textContent = 'Test FAILED - check console';
996 statusEl.style.color = '#ef4444';
997 }
998 if (testStatusEl) {
999 testStatusEl.textContent = 'Test FAILED!';
1000 testStatusEl.style.color = '#ef4444';
1001 }
1002 if (testStateEl) testStateEl.textContent = 'FAILED';
1003 updateAllPWMGauges(0);
1004 break;
1005 }
1006}

1393/**

1395 *
1396 * @description
1397 * Creates a Three.js scene with:
1398 * - Perspective camera with 60° FOV
1399 * - Ambient and directional lighting
1400 * - Grid helper for reference
1401 * - Rocket model (cylinder body, cone nose, fins)
1402 *
1403 * Animation loop smoothly interpolates rocket rotation towards
1404 * target values received from telemetry.
1405 */
1406function initThree() {
1407 const container = document.getElementById('threeContainer');
1408 if (!container) return;
1409 scene = new THREE.Scene();
1410 camera = new THREE.PerspectiveCamera(60, container.clientWidth / container.clientHeight, 0.1, 1000);
1411 renderer = new THREE.WebGLRenderer({ antialias: true, alpha: true });
1412 renderer.setSize(container.clientWidth, container.clientHeight);
1413 renderer.setPixelRatio(window.devicePixelRatio || 1);
1414 container.appendChild(renderer.domElement);
1415
1416 const ambient = new THREE.AmbientLight(0xffffff, 0.6);
1417 scene.add(ambient);
1418
1419 const keyLight = new THREE.DirectionalLight(0xffffff, 0.9);
1420 keyLight.position.set(6, 8, 6);
1421 scene.add(keyLight);
1422
1423 const fillLight = new THREE.DirectionalLight(0x88ccff, 0.4);
1424 fillLight.position.set(-6, -4, 4);
1425 scene.add(fillLight);
1426
1427 const grid = new THREE.GridHelper(30, 30, 0x243447, 0x182230);
1428 grid.position.y = -3;
1429 scene.add(grid);
1430
1431 const body = new THREE.CylinderGeometry(0.5, 0.5, 4.5, 16);
1432 const cone = new THREE.ConeGeometry(0.6, 1.2, 16);
1433 const mat = new THREE.MeshStandardMaterial({ color: 0x22c55e, metalness: 0.25, roughness: 0.5 });
1434 const bodyMesh = new THREE.Mesh(body, mat);
1435 const coneMesh = new THREE.Mesh(cone, mat);
1436 coneMesh.position.y = 2.9;
1437
1438 rocket = new THREE.Group();
1439 rocket.add(bodyMesh);
1440 rocket.add(coneMesh);
1441
1442 const finMat = new THREE.MeshStandardMaterial({ color: 0x0ea5e9, metalness: 0.2, roughness: 0.6 });
1443 const fin = new THREE.BoxGeometry(0.15, 1.0, 0.8);
1444 const finPositions = [
1445 { x: 0.55, z: 0 },
1446 { x: -0.55, z: 0 },
1447 { x: 0, z: 0.55 },
1448 { x: 0, z: -0.55 }
1449 ];
1450 finPositions.forEach((pos) => {
1451 const finMesh = new THREE.Mesh(fin, finMat);
1452 finMesh.position.set(pos.x, -1.8, pos.z);
1453 finMesh.rotation.y = pos.x === 0 ? Math.PI / 2 : 0;
1454 rocket.add(finMesh);
1455 });

1457 scene.add(rocket);
1458
1459 camera.position.set(0, 1.5, 8);
1460 camera.lookAt(0, 0, 0);

1462 /**
1463 * Animation loop - smoothly interpolates rocket rotation
1464 */
1465 function animate() {
1466 requestAnimationFrame(animate);
1467 if (rocket) {
1468 rocket.rotation.x += (targetRotation.x - rocket.rotation.x) * 0.08;
1469 rocket.rotation.y += (targetRotation.y - rocket.rotation.y) * 0.08;
1470 rocket.rotation.z += (targetRotation.z - rocket.rotation.z) * 0.08;
1471 }
1472 renderer.render(scene, camera);
1473 }

1481 camera.aspect = clientWidth / clientHeight;
1482 camera.updateProjectionMatrix();

1484 });

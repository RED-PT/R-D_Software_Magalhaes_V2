17bool BNO055_Init(BNO055_t *dev, I2C_HandleTypeDef *hi2c) {
18 uint8_t chip_id;
19
20 if (!dev || !hi2c) {
21 return false;
22 }
23
24 dev->hi2c = hi2c;
25 dev->i2c_addr = BNO055_I2C_ADDR << 1; // I2C addresses are shifted left by 1
26 dev->data_ready = 0;
27 memset(dev->read_buffer, 0, sizeof(dev->read_buffer));
28
29 HAL_Delay(50); // Power-up time
30
31 // Read and verify chip ID
32 if (!BNO055_ReadRegister(dev, BNO055_REG_CHIP_ID, &chip_id)) {
33 return false;
34 }
35
36 if (chip_id != BNO055_CHIP_ID_VALUE) {
37 return false;
38 }
39
40 return true;
41}

43bool BNO055_Configure(BNO055_t *dev) {
44 if (!dev) {
45 return false;
46 }
47
48 // Switch to CONFIG mode first
49 if (!BNO055_SetOpMode(dev, BNO055_MODE_CONFIGMODE)) {
50 return false;
51 }
52 HAL_Delay(25);
53
54 // Set to NDOF mode (full 9-DOF sensor fusion)
55 if (!BNO055_SetOpMode(dev, BNO055_MODE_NDOF)) {
56 return false;
57 }
58 HAL_Delay(100); // Mode change takes time
59
60 // Set power mode to normal
61 if (!BNO055_WriteRegister(dev, BNO055_REG_PWR_MODE, BNO055_POWER_NORMAL)) {
62 return false;
63 }
64 HAL_Delay(10);
65
66 return true;
67}

69bool BNO055_StartReadDMA(BNO055_t *dev) {
70 if (!dev || !dev->hi2c) {
71 return false;
72 }
73
74 // Read euler angles + quaternion + linear accel
75 // Starting from EULER_H_LSB (0x1A) for 20 bytes total
76 // This covers: Euler(6) + Quat(8) + LinAccel(6)
77
78 uint8_t start_reg = BNO055_REG_EULER_H_LSB;
79 uint8_t num_bytes = 20; // Euler(6) + Quat(8) + LinAccel(6)
80
81 // Prepare command: I2C address write, then register address, then read
82 if (HAL_I2C_Mem_Read_DMA(dev->hi2c, dev->i2c_addr, start_reg, I2C_MEMADD_SIZE_8BIT, dev->read_buffer, num_bytes) != HAL_OK) {
83 printf("ERROR123\r\n");
84 return false;
85 }
86
87 return true;

90void BNO055_ParseDMABuffer(BNO055_t *dev) {
91 if (!dev) {
92 return;
93 }
94
95 // Buffer layout (starting from EULER_H_LSB):
96 // [0:1] Heading, [2:3] Roll, [4:5] Pitch
97 // [6:7] Quat W, [8:9] Quat X, [10:11] Quat Y, [12:13] Quat Z
98 // [14:15] LinAccel X, [16:17] LinAccel Y, [18:19] LinAccel Z
99
100 dev->euler_raw[0] = (int16_t)((dev->read_buffer[1] << 8) | dev->read_buffer[0]); // Heading
101 dev->euler_raw[1] = (int16_t)((dev->read_buffer[3] << 8) | dev->read_buffer[2]); // Roll
102 dev->euler_raw[2] = (int16_t)((dev->read_buffer[5] << 8) | dev->read_buffer[4]); // Pitch
103
104 dev->quat_raw[0] = (int16_t)((dev->read_buffer[7] << 8) | dev->read_buffer[6]); // W
105 dev->quat_raw[1] = (int16_t)((dev->read_buffer[9] << 8) | dev->read_buffer[8]); // X
106 dev->quat_raw[2] = (int16_t)((dev->read_buffer[11] << 8) | dev->read_buffer[10]); // Y
107 dev->quat_raw[3] = (int16_t)((dev->read_buffer[13] << 8) | dev->read_buffer[12]); // Z
108
109 dev->lia_raw[0] = (int16_t)((dev->read_buffer[15] << 8) | dev->read_buffer[14]); // X
110 dev->lia_raw[1] = (int16_t)((dev->read_buffer[17] << 8) | dev->read_buffer[16]); // Y
111 dev->lia_raw[2] = (int16_t)((dev->read_buffer[19] << 8) | dev->read_buffer[18]); // Z
112
113 dev->data_ready = 1;
114}

116bool BNO055_ProcessData(BNO055_t *dev, BNO_t *output) {
117 if (!dev || !output || !dev->data_ready) {
118 return false;
119 }
120
121 // Convert Euler angles (16-bit signed, 1/16 degree per LSB)
122 output->heading_deg = (float)dev->euler_raw[0] / 16.0f;
123 output->roll_deg = (float)dev->euler_raw[1] / 16.0f;
124 output->pitch_deg = (float)dev->euler_raw[2] / 16.0f;
125
126 // Normalize heading to 0-360
127 if (output->heading_deg < 0) {
128 output->heading_deg += 360.0f;
129 }
130
131 // Convert Quaternion (16-bit signed, 1/16384 per LSB)
132 output->quat_w = (float)dev->quat_raw[0] / 16384.0f;
133 output->quat_x = (float)dev->quat_raw[1] / 16384.0f;
134 output->quat_y = (float)dev->quat_raw[2] / 16384.0f;
135 output->quat_z = (float)dev->quat_raw[3] / 16384.0f;
136
137 // Linear Acceleration (16-bit signed, 1/100 m/s^2 per LSB)
138 output->accel_x_mg = (float)dev->lia_raw[0] / 100.0f * 1000.0f; // Convert to mg
139 output->accel_y_mg = (float)dev->lia_raw[1] / 100.0f * 1000.0f;
140 output->accel_z_mg = (float)dev->lia_raw[2] / 100.0f * 1000.0f;
141
142 // Gyro, Mag, Raw accel would need separate reads for full data
143 // For now, setting placeholders
144 output->gyro_x_dps = 0.0f;
145 output->gyro_y_dps = 0.0f;
146 output->gyro_z_dps = 0.0f;
147
148 output->mag_x_uT = 0.0f;
149 output->mag_y_uT = 0.0f;
150 output->mag_z_uT = 0.0f;
151
152 // Get calibration status - **REMOVIDO** para evitar bloqueio do I2C
153 // BNO055_GetCalibStatus(dev, &dev->calib_status);
154 // output->calibration_status = (dev->calib_status >> 6) & 0x03;
155 output->calibration_status = 0; // Placeholder
156
157 // Temperature (1 byte, 1°C per LSB)
158 output->timestamp_ms = HAL_GetTick();
159
160 dev->data_ready = 0;
161
162 // **BLOCO DE CÓDIGO INCORRETO REMOVIDO DAQUI**
163
164 return true;
165}

30void SX126x_WriteCommand(uint8_t cmd, uint8_t *data, uint8_t length) {
31 wait_not_busy();
32 SX126X_CS_LOW();
33 HAL_SPI_Transmit(SPI_SX126X, &cmd, 1, 100);

148bool SX126x_Init(SX126x_LoRaConfig_t *config) {
149 memset(&sx126x_state, 0, sizeof(sx126x_state));
150
151 if (!SX126x_Reset()) return false;
152
153 SX126x_SetStandby(SX126X_STANDBY_RC);
154
155 uint8_t reg_mode = SX126X_REGULATOR_DC_DC;
156 SX126x_WriteCommand(SX126X_CMD_SET_REGULATOR_MODE, &reg_mode, 1);
157
158 uint8_t calib = 0x7F;
159 SX126x_WriteCommand(SX126X_CMD_CALIBRATE, &calib, 1);
160 HAL_Delay(10);
161
162 uint8_t pkt_type = SX126X_PACKET_TYPE_LORA;
163 SX126x_WriteCommand(SX126X_CMD_SET_PKT_TYPE, &pkt_type, 1);
164
165 uint32_t freq = (uint32_t)((double)config->frequency_hz / 32000000.0 * 33554432.0);
166 uint8_t freq_buf[4] = {
167 (freq >> 24) & 0xFF, (freq >> 16) & 0xFF,
168 (freq >> 8) & 0xFF, freq & 0xFF
169 };
170 SX126x_WriteCommand(SX126X_CMD_SET_RF_FREQUENCY, freq_buf, 4);
171
172 uint8_t mod_params[4] = {
173 config->spreading_factor,
174 config->bandwidth,
175 config->coding_rate,
176 0x00
177 };
178 SX126x_WriteCommand(SX126X_CMD_SET_MODULATION_PARAMS, mod_params, 4);
179
180 uint8_t pkt_params[6] = {
181 (config->preamble_length >> 8) & 0xFF,
182 config->preamble_length & 0xFF,
183 0x00,
184 config->enable_crc ? 0x01 : 0x00,
185 config->invert_iq ? 0x01 : 0x00,
186 0x00
187 };
188 SX126x_WriteCommand(SX126X_CMD_SET_PKT_PARAMS, pkt_params, 6);
189
190 uint8_t pa_config[4] = {0x04, 0x07, 0x00, 0x01};
191 SX126x_WriteCommand(SX126X_CMD_SET_PA_CONFIG, pa_config, 4);
192
193 uint8_t tx_params[2] = {config->tx_power_dbm, 0x04};
194 SX126x_WriteCommand(SX126X_CMD_SET_TX_PARAMS, tx_params, 2);
195
196 uint8_t base_addr[2] = {0x00, 0x00};
197 SX126x_WriteCommand(SX126X_CMD_SET_BUFFER_BASE_ADDRESS, base_addr, 2);
198
199 uint8_t sync_word[2] = {
200 (config->sync_word >> 8) & 0xFF,
201 config->sync_word & 0xFF
202 };
203 SX126x_WriteRegister(SX126X_REG_LORA_SYNC_WORD_MSB, sync_word, 2);
204
205#ifdef RADIO_INTERFACE_SPI
206 // E22-900M22S has external RF switch (RXEN/TXEN GPIO) — do NOT use DIO2
207 uint8_t dio2_rf = 0x00;
208 SX126x_WriteCommand(SX126X_CMD_SET_DIO2_AS_RF_SWITCH, &dio2_rf, 1);
209 RADIO_RF_SWITCH_OFF();
210#else
211 // Direct SX1262 modules use DIO2 as internal RF switch
212 uint8_t dio2_rf = 0x01;
213 SX126x_WriteCommand(SX126X_CMD_SET_DIO2_AS_RF_SWITCH, &dio2_rf, 1);
214#endif
215
216 uint16_t irq_mask = SX126X_IRQ_TX_DONE | SX126X_IRQ_RX_DONE |
217 SX126X_IRQ_TIMEOUT | SX126X_IRQ_CRC_ERROR;
218 uint8_t irq_params[8] = {
219 (irq_mask >> 8) & 0xFF, irq_mask & 0xFF,
220 (irq_mask >> 8) & 0xFF, irq_mask & 0xFF,
221 0x00, 0x00, 0x00, 0x00
222 };
223 SX126x_WriteCommand(SX126X_CMD_SET_DIO_IRQ_PARAMS, irq_params, 8);
224
225 SX126x_ClearIrqStatus(SX126X_IRQ_ALL);
226
227 return true;
228}

234bool SX126x_TransmitDMA(uint8_t *data, uint8_t length) {
235 if (sx126x_state.spi_state != SX126X_DMA_IDLE) return false;
236 if (length == 0 || length > 255) return false;
237
238 sx126x_state.tx_done = false;
239 sx126x_state.spi_state = SX126X_DMA_TX_BUSY;
240
241 SX126x_SetStandby(SX126X_STANDBY_RC);
242
243 uint8_t pkt_params[6];
244 SX126x_ReadCommand(SX126X_CMD_SET_PKT_PARAMS, pkt_params, 6);
245 pkt_params[0] = 0x00;
246 pkt_params[1] = 0x08;
247 pkt_params[2] = length;
248 SX126x_WriteCommand(SX126X_CMD_SET_PKT_PARAMS, pkt_params, 6);
249
250 wait_not_busy();
251 SX126X_CS_LOW();
252 uint8_t cmd = SX126X_CMD_WRITE_BUFFER;
253 HAL_SPI_Transmit(SPI_SX126X, &cmd, 1, 100);
254 uint8_t offset = 0x00;
255 HAL_SPI_Transmit(SPI_SX126X, &offset, 1, 100);
256
257 memcpy(sx126x_state.tx_buffer, data, length);
258 HAL_StatusTypeDef status = HAL_SPI_Transmit_DMA(SPI_SX126X, sx126x_state.tx_buffer, length);
259
260 if (status != HAL_OK) {
261 SX126X_CS_HIGH();
262 sx126x_state.spi_state = SX126X_DMA_IDLE;
263 return false;
264 }
265
266 return true;
267}

306int SX126x_Receive(uint8_t *buffer, uint16_t max_length, SX126x_RxInfo_t *rx_info) {
307 if (!sx126x_state.rx_done) return 0;
308
309 SX126x_SetStandby(SX126X_STANDBY_RC);
310
311 uint8_t status[2];
312 SX126x_ReadCommand(SX126X_CMD_GET_RX_BUFFER_STATUS, status, 2);
313 uint8_t rx_len = status[0];
314 uint8_t rx_offset = status[1];
315
316 if (rx_len > max_length) rx_len = max_length;
317
318 wait_not_busy();
319 SX126X_CS_LOW();
320 uint8_t cmd = SX126X_CMD_READ_BUFFER;
321 HAL_SPI_Transmit(SPI_SX126X, &cmd, 1, 100);
322 HAL_SPI_Transmit(SPI_SX126X, &rx_offset, 1, 100);
323 uint8_t dummy = 0x00;
324 HAL_SPI_Transmit(SPI_SX126X, &dummy, 1, 100);
325 HAL_SPI_Receive(SPI_SX126X, buffer, rx_len, 100);
326 SX126X_CS_HIGH();
327
328 if (rx_info) {
329 uint8_t pkt_status[3];
330 SX126x_ReadCommand(SX126X_CMD_GET_PKT_STATUS, pkt_status, 3);
331 rx_info->rssi = -pkt_status[0] / 2;
332 rx_info->snr = (int8_t)pkt_status[1] / 4;
333 rx_info->signal_rssi = -pkt_status[2] / 2;
334 rx_info->length = rx_len;
335
336 sx126x_state.last_rssi = rx_info->rssi;
337 sx126x_state.last_snr = rx_info->snr;
338 }
339
340 sx126x_state.rx_done = false;
341 SX126x_ClearIrqStatus(SX126X_IRQ_ALL);
342 SX126x_SetRx(0xFFFFFF);
343
344 return rx_len;
345}

359void SX126x_DIO1_IRQ_Handler(void) {
360 uint16_t irq_status = SX126x_GetIrqStatus();
361
362 if (irq_status & SX126X_IRQ_TX_DONE) {
363 sx126x_state.tx_done = true;
364 SX126x_ClearIrqStatus(SX126X_IRQ_TX_DONE);
365 }
366
367 if (irq_status & SX126X_IRQ_RX_DONE) {
368 sx126x_state.rx_done = true;
369 SX126x_ClearIrqStatus(SX126X_IRQ_RX_DONE);
370 }
371
372 if (irq_status & SX126X_IRQ_TIMEOUT) {
373 SX126x_ClearIrqStatus(SX126X_IRQ_TIMEOUT);
374 }
375
376 if (irq_status & SX126X_IRQ_CRC_ERROR) {
377 SX126x_ClearIrqStatus(SX126X_IRQ_CRC_ERROR);
378 SX126x_SetRx(0xFFFFFF);
379 }

136static bool can_transmit(void) {
137 // Don't TX if we recently transmitted (E22 turnaround)
138 uint32_t since_tx = HAL_GetTick() - last_tx_tick;
139 if (since_tx < E22_TURNAROUND_MS) return false;
140
141 // Don't TX if E22 is busy
142 if (E22_IsBusy()) return false;
143
144 // Already transmitted this slot
145 if (tx_done_this_slot) return false;
146
147 return true;
148}

150static void sync_to_gs(uint8_t gs_frame_id, uint32_t gs_time) {
151 uint32_t now = HAL_GetTick();
152
153 // When we receive a sync, we know GS sent it at the START of slot 9
154 // Align our frame so slot 9 starts at this moment
155 uint32_t new_frame_start = now - (TDMA_RX_SLOT * SLOT_DURATION_MS);
156
157 if (tdma_ctx.state == TDMA_SYNCED) {
158 // Already synced - check if we need correction
159 int32_t error = (int32_t)(new_frame_start - tdma_ctx.frame_start_tick);
160
161 if (error > SYNC_CORRECTION_THRESH || error < -SYNC_CORRECTION_THRESH) {
162 tdma_ctx.frame_start_tick = new_frame_start;
163 radio_stats.sync_corrections++;
164 }
165 } else {
166 // First sync - hard align
167 tdma_ctx.frame_start_tick = new_frame_start;
168 tdma_ctx.state = TDMA_SYNCED;
169 printf("[RADIO] SYNCED! frame=%u\r\n", gs_frame_id);
170 }
171
172 tdma_ctx.frame_id = gs_frame_id;
173 tdma_ctx.last_sync_tick = now;
174 tdma_ctx.missed_sync_count = 0;
175}

264static void process_command(command_packet_t *cmd) {
265 uint16_t calc_crc = crc16_calculate((uint8_t*)cmd, sizeof(command_packet_t) - 2);
266 if (calc_crc != cmd->crc16) {
267 radio_stats.crc_errors++;
268 return;
269 }
270
271 radio_stats.rx_cmd++;
272
273 // Store for ACK piggyback
274 tdma_ctx.last_cmd_seq_received = cmd->cmd_seq;
275
276 // Forward command to FSM thread via queue
277 fsm_cmd_msg_t fsm_msg = {0};
278 fsm_msg.cmd = (fsm_command_t)cmd->cmd_id;
279 fsm_msg.cmd_seq = cmd->cmd_seq;
280 memcpy(&fsm_msg.payload, cmd->params, sizeof(cmd->params));
281
282 if (fsm_send_command(fsm_msg.cmd, fsm_msg.cmd_seq, &fsm_msg.payload, sizeof(fsm_msg.payload))) {
283 tdma_ctx.last_cmd_status = 0; // Queued OK
284 printf("[RADIO] CMD %d -> FSM queue\r\n", cmd->cmd_id);
285 } else {
286 tdma_ctx.last_cmd_status = 2; // Queue full
287 printf("[RADIO] CMD queue FULL!\r\n");
288 }
289
290 // Sync timing
291 sync_to_gs(cmd->frame_id, cmd->time);
292}
293static void process_sync(sync_packet_t *sync) {
294 uint16_t calc_crc = crc16_calculate((uint8_t*)sync, sizeof(sync_packet_t) - 2);
295 if (calc_crc != sync->crc16) {
296 radio_stats.crc_errors++;
297 return;
298 }
299
300 radio_stats.rx_sync++;
301 sync_to_gs(sync->frame_id, sync->gs_time);
302}

304static bool process_rx_byte(uint8_t byte) {
305 uint32_t now = HAL_GetTick();
306
307 // Timeout: reset buffer if no bytes for a while
308 if (rx_idx > 0 && (now - rx_last_byte_tick) > RX_TIMEOUT_MS) {
309 rx_idx = 0;
310 }
311 rx_last_byte_tick = now;
312
313 // First byte: validate packet type
314 if (rx_idx == 0) {
315 uint16_t expected = get_packet_size(byte);
316 if (expected == 0) {
317 return false; // Not a packet type we're looking for
318 }
319 }
320
321 // Store byte
322 if (rx_idx < RX_BUFFER_SIZE) {
323 rx_buffer[rx_idx++] = byte;
324 } else {
325 rx_idx = 0;
326 return false;
327 }
328
329 // Check for complete packet
330 uint8_t type = rx_buffer[0];
331 uint16_t expected = get_packet_size(type);
332
333 if (rx_idx >= expected) {
334 if (type == TELEM_PACKET_COMMAND) {
335 process_command((command_packet_t*)rx_buffer);
336 } else if (type == TELEM_PACKET_SYNC) {
337 process_sync((sync_packet_t*)rx_buffer);
338 }
339 rx_idx = 0;
340 return true;
341 }
342
343 return false;
344}

358static void handle_synced_mode(void) {
359 uint32_t now = HAL_GetTick();
360
361 // Check for frame rollover
362 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
363 advance_frame();
364
365 // Check sync timeout
366 tdma_ctx.missed_sync_count++;
367 if (tdma_ctx.missed_sync_count >= SYNC_TIMEOUT_FRAMES) {
368 tdma_ctx.state = TDMA_UNSYNCED;
369 printf("[RADIO] Lost sync!\r\n");
370 return;
371 }
372 }
373
374 // Get current slot and time within slot
375 uint8_t slot = get_current_slot();
376 uint32_t time_in_slot = get_time_in_slot();
377
378 // Update global slot_index for other code that needs it
379 tdma_ctx.slot_index = slot;
380
381 // Detect slot change -> reset tx_done flag
382 static uint8_t last_slot = 255;
383 if (slot != last_slot) {
384 last_slot = slot;
385 tx_done_this_slot = false;
386 }
387
388 // Slot behavior based on current slot
389 if (slot < TDMA_TX_SLOTS) {
390 // Slots 0-7: Fast telemetry
391 // TX window: 5ms to 70ms into slot
392 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < TX_END_OFFSET_MS) {
393 // First check for pending events (high priority - PONG, etc.)
394 tx_pending_events();
395 // Then send regular fast telemetry if we can still transmit
396 tx_fast_telemetry(slot);
397 }
398 // RX in second half of slot (after TX done)
399 if (time_in_slot >= TX_END_OFFSET_MS || tx_done_this_slot) {
400 process_rx_data();
401 }
402 }
403 else if (slot == TDMA_SLOW_SLOT) {
404 // Slot 8: Slow telemetry
405 // TX early, but stop before slot 9 to avoid collision with GS
406 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < (SLOT_DURATION_MS - RX_SLOT_GUARD_MS)) {
407 // First check for pending events
408 tx_pending_events();
409 tx_slow_telemetry(slot);
410 }
411 // RX after TX
412 if (tx_done_this_slot) {
413 process_rx_data();
414 }
415 }
416 else {
417 // Slot 9: RX only - listen for GS commands/sync
418 process_rx_data();
419 }
420}

422static void handle_unsynced_mode(void) {
423 uint32_t now = HAL_GetTick();
424
425 // In UNSYNCED mode: mostly listen, occasionally beacon
426
427 // Always try to send pending events (like PONG) even when unsynced
428 tx_pending_events();
429
430 // Send a beacon every UNSYNC_BEACON_INTERVAL_MS so GS knows we're alive
431 if ((now - last_unsync_beacon_tick) >= UNSYNC_BEACON_INTERVAL_MS) {
432 if (!E22_IsBusy()) {
433 // Send with slot=0 since we don't know real slot
434 telemetry_fast_t pkt;
435 if (sensor_mutex && xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
436 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
437 tdma_ctx.fast_seq++, &fsm_ctx,
438 &latest_imu, &latest_baro, &latest_bno,
439 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
440 xSemaphoreGive(sensor_mutex);
441 } else {
442 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
443 tdma_ctx.fast_seq++, &fsm_ctx,
444 NULL, NULL, NULL,
445 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
446 }
447
448 if (E22_Transmit((uint8_t*)&pkt, sizeof(pkt)) == E22_OK) {
449 radio_stats.tx_fast++;
450 last_tx_tick = HAL_GetTick();
451 }
452 last_unsync_beacon_tick = now;
453 }
454 }
455
456 // Continuously process RX data
457 process_rx_data();
458
459 // Keep frame counter advancing (free-running when unsynced)
460 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
461 tdma_ctx.frame_id++;
462 tdma_ctx.frame_start_tick = now;
463 }
464}

469void radio_thread_function() {
470 printf("[RADIO] Thread starting...\r\n");
471 fsm_report_thread_started("RADIO");
472
473 // Initialize
474 sensor_mutex = xSemaphoreCreateMutex();
475 if (!sensor_mutex) {
476 printf("[RADIO] Mutex create failed!\r\n");
477 }
478
479 if (!E22_Init(UART_RADIO)) {
480 printf("[RADIO] Init FAIL\r\n");
481 fsm_report_init_status("RADIO", false);
482 vTaskSuspend(NULL);
483 }
484 fsm_report_init_status("RADIO", true);
485 printf("[RADIO] E22 initialized\r\n");
486 printf("[RADIO] Packet sizes: FAST=%u SLOW=%u CMD=%u SYNC=%u\r\n",
487 sizeof(telemetry_fast_t), sizeof(telemetry_slow_t),
488 sizeof(command_packet_t), sizeof(sync_packet_t));
489
490 // Start in UNSYNCED state
491 memset(&tdma_ctx, 0, sizeof(tdma_ctx));
492 tdma_ctx.state = TDMA_UNSYNCED;
493 tdma_ctx.frame_start_tick = HAL_GetTick();
494 last_tx_tick = 0;
495 last_unsync_beacon_tick = 0;
496 tx_done_this_slot = false;
497
498 uint32_t last_stats_tick = HAL_GetTick();
499
500 printf("[RADIO] Waiting for GS sync...\r\n");
501
502 while (1) {
503 uint32_t now = HAL_GetTick();
504
505 if (tdma_ctx.state == TDMA_SYNCED) {
506 handle_synced_mode();
507 } else {
508 handle_unsynced_mode();
509 }
510
511 // Stats every 10 seconds
512 if ((now - last_stats_tick) >= 10000) {
513 (void)E22_GetStats();
514
515 printf("[RADIO] %s F=%u slot=%u | TX: f=%lu s=%lu | RX: cmd=%lu sync=%lu | CRC=%lu | ack_seq=%u\r\n",
516 tdma_ctx.state == TDMA_SYNCED ? "SYNC" : "UNSYNC",
517 tdma_ctx.frame_id, tdma_ctx.slot_index,
518 radio_stats.tx_fast, radio_stats.tx_slow,
519 radio_stats.rx_cmd, radio_stats.rx_sync,
520 radio_stats.crc_errors,
521 tdma_ctx.last_cmd_seq_received);
522
523 last_stats_tick = now;
524 }
525
526 // Fast loop for responsive operation
527 vTaskDelay(pdMS_TO_TICKS(2));
528 }

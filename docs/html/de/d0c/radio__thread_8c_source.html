165static void sync_to_gs(uint8_t gs_frame_id, uint32_t gs_time) {
166 uint32_t now = HAL_GetTick();
167
168 // When we receive a sync, we know GS sent it at the START of slot 9
169 // Align our frame so slot 9 starts at this moment
170 uint32_t new_frame_start = now - (TDMA_RX_SLOT * SLOT_DURATION_MS);
171
172 if (tdma_ctx.state == TDMA_SYNCED) {
173 // Already synced - check if we need correction
174 int32_t error = (int32_t)(new_frame_start - tdma_ctx.frame_start_tick);
175
176 if (error > SYNC_CORRECTION_THRESH || error < -SYNC_CORRECTION_THRESH) {
177 tdma_ctx.frame_start_tick = new_frame_start;
178 radio_stats.sync_corrections++;
179 }
180 } else {
181 // First sync - hard align
182 tdma_ctx.frame_start_tick = new_frame_start;
183 tdma_ctx.state = TDMA_SYNCED;
184 printf("[RADIO] SYNCED! frame=%u\r\n", gs_frame_id);
185 }
186
187 tdma_ctx.frame_id = gs_frame_id;
188 tdma_ctx.last_sync_tick = now;
189 tdma_ctx.missed_sync_count = 0;
190}

245 if (!can_transmit()) return;

251 if (xQueueReceive(queue_fsm_events, &evt, 0) == pdTRUE) {

279static void process_command(command_packet_t *cmd) {
280 uint16_t calc_crc = crc16_calculate((uint8_t*)cmd, sizeof(command_packet_t) - 2);
281 if (calc_crc != cmd->crc16) {
282 radio_stats.crc_errors++;
283 return;
284 }
285
286 radio_stats.rx_cmd++;
287
288 // Store for ACK piggyback
289 tdma_ctx.last_cmd_seq_received = cmd->cmd_seq;
290
291 // Forward command to FSM thread via queue
292 fsm_cmd_msg_t fsm_msg = {0};
293 fsm_msg.cmd = (fsm_command_t)cmd->cmd_id;
294 fsm_msg.cmd_seq = cmd->cmd_seq;
295 memcpy(&fsm_msg.payload, cmd->params, sizeof(cmd->params));
296
297 if (fsm_send_command(fsm_msg.cmd, fsm_msg.cmd_seq, &fsm_msg.payload, sizeof(fsm_msg.payload))) {
298 tdma_ctx.last_cmd_status = 0; // Queued OK
299 printf("[RADIO] CMD %d -> FSM queue\r\n", cmd->cmd_id);
300 } else {
301 tdma_ctx.last_cmd_status = 2; // Queue full
302 printf("[RADIO] CMD queue FULL!\r\n");
303 }
304
305 // Sync timing
306 sync_to_gs(cmd->frame_id, cmd->time);
307}
308static void process_sync(sync_packet_t *sync) {
309 uint16_t calc_crc = crc16_calculate((uint8_t*)sync, sizeof(sync_packet_t) - 2);
310 if (calc_crc != sync->crc16) {
311 radio_stats.crc_errors++;
312 return;
313 }
314
315 radio_stats.rx_sync++;
316 sync_to_gs(sync->frame_id, sync->gs_time);
317}

319static bool process_rx_byte(uint8_t byte) {
320 uint32_t now = HAL_GetTick();
321
322 // Timeout: reset buffer if no bytes for a while
323 if (rx_idx > 0 && (now - rx_last_byte_tick) > RX_TIMEOUT_MS) {
324 rx_idx = 0;
325 }
326 rx_last_byte_tick = now;
327
328 // First byte: validate packet type
329 if (rx_idx == 0) {
330 uint16_t expected = get_packet_size(byte);
331 if (expected == 0) {
332 return false; // Not a packet type we're looking for
333 }
334 }
335
336 // Store byte
337 if (rx_idx < RX_BUFFER_SIZE) {
338 rx_buffer[rx_idx++] = byte;
339 } else {
340 rx_idx = 0;
341 return false;
342 }
343
344 // Check for complete packet
345 uint8_t type = rx_buffer[0];
346 uint16_t expected = get_packet_size(type);
347
348 if (rx_idx >= expected) {
349 if (type == TELEM_PACKET_COMMAND) {
350 process_command((command_packet_t*)rx_buffer);
351 } else if (type == TELEM_PACKET_SYNC) {
352 process_sync((sync_packet_t*)rx_buffer);
353 }
354 rx_idx = 0;
355 return true;
356 }
357
358 return false;
359}

373static void handle_synced_mode(void) {
374 uint32_t now = HAL_GetTick();
375
376 // Check for frame rollover
377 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
378 advance_frame();
379
380 // Check sync timeout
381 tdma_ctx.missed_sync_count++;
382 if (tdma_ctx.missed_sync_count >= SYNC_TIMEOUT_FRAMES) {
383 tdma_ctx.state = TDMA_UNSYNCED;
384 printf("[RADIO] Lost sync!\r\n");
385 return;
386 }
387 }
388
389 // Get current slot and time within slot
390 uint8_t slot = get_current_slot();
391 uint32_t time_in_slot = get_time_in_slot();
392
393 // Update global slot_index for other code that needs it
394 tdma_ctx.slot_index = slot;
395
396 // Detect slot change -> reset tx_done flag
397 static uint8_t last_slot = 255;
398 if (slot != last_slot) {
399 last_slot = slot;
400 tx_done_this_slot = false;
401 }
402
403 // Slot behavior based on current slot
404 if (slot < TDMA_TX_SLOTS) {
405 // Slots 0-7: Fast telemetry
406 // TX window: 5ms to 70ms into slot
407 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < TX_END_OFFSET_MS) {
408 // First check for pending events (high priority - PONG, etc.)
409 tx_pending_events();
410 // Then send regular fast telemetry if we can still transmit
411 tx_fast_telemetry(slot);
412 }
413 // RX in second half of slot (after TX done)
414 if (time_in_slot >= TX_END_OFFSET_MS || tx_done_this_slot) {
415 process_rx_data();
416 }
417 }
418 else if (slot == TDMA_SLOW_SLOT) {
419 // Slot 8: Slow telemetry
420 // TX early, but stop before slot 9 to avoid collision with GS
421 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < (SLOT_DURATION_MS - RX_SLOT_GUARD_MS)) {
422 // First check for pending events
423 tx_pending_events();
424 tx_slow_telemetry(slot);
425 }
426 // RX after TX
427 if (tx_done_this_slot) {
428 process_rx_data();
429 }
430 }
431 else {
432 // Slot 9: RX only - listen for GS commands/sync
433 process_rx_data();
434 }
435}

437static void handle_unsynced_mode(void) {
438 uint32_t now = HAL_GetTick();
439
440 // In UNSYNCED mode: mostly listen, occasionally beacon
441
442 // Always try to send pending events (like PONG) even when unsynced
443 tx_pending_events();
444
445 // Send a beacon every UNSYNC_BEACON_INTERVAL_MS so GS knows we're alive
446 if ((now - last_unsync_beacon_tick) >= UNSYNC_BEACON_INTERVAL_MS) {
447 if (!E22_IsBusy()) {
448 // Send with slot=0 since we don't know real slot
449 telemetry_fast_t pkt;
450 if (sensor_mutex && xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
451 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
452 tdma_ctx.fast_seq++, &fsm_ctx,
453 &latest_imu, &latest_baro, &latest_bno,
454 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
455 xSemaphoreGive(sensor_mutex);
456 } else {
457 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
458 tdma_ctx.fast_seq++, &fsm_ctx,
459 NULL, NULL, NULL,
460 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
461 }
462
463 if (E22_Transmit((uint8_t*)&pkt, sizeof(pkt)) == E22_OK) {
464 radio_stats.tx_fast++;
465 last_tx_tick = HAL_GetTick();
466 }
467 last_unsync_beacon_tick = now;
468 }
469 }
470
471 // Continuously process RX data
472 process_rx_data();
473
474 // Keep frame counter advancing (free-running when unsynced)
475 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
476 tdma_ctx.frame_id++;
477 tdma_ctx.frame_start_tick = now;
478 }
479}

482// Main Thread

484void radio_thread_function() {
485 printf("[RADIO] Thread starting...\r\n");
486 fsm_report_thread_started("RADIO");
487
488 // Initialize
489 sensor_mutex = xSemaphoreCreateMutex();
490 if (!sensor_mutex) {
491 printf("[RADIO] Mutex create failed!\r\n");
492 }
493
494 if (!E22_Init(UART_RADIO)) {
495 printf("[RADIO] Init FAIL\r\n");
496 fsm_report_init_status("RADIO", false);
497 vTaskSuspend(NULL);
498 }
499 fsm_report_init_status("RADIO", true);
500 printf("[RADIO] E22 initialized\r\n");
501 printf("[RADIO] Packet sizes: FAST=%u SLOW=%u CMD=%u SYNC=%u\r\n",
502 sizeof(telemetry_fast_t), sizeof(telemetry_slow_t),
503 sizeof(command_packet_t), sizeof(sync_packet_t));
504
505 // Start in UNSYNCED state
506 memset(&tdma_ctx, 0, sizeof(tdma_ctx));
507 tdma_ctx.state = TDMA_UNSYNCED;
508 tdma_ctx.frame_start_tick = HAL_GetTick();
509 last_tx_tick = 0;
510 last_unsync_beacon_tick = 0;
511 tx_done_this_slot = false;
512
513 uint32_t last_stats_tick = HAL_GetTick();
514
515 printf("[RADIO] Waiting for GS sync...\r\n");
516
517 while (1) {
518 uint32_t now = HAL_GetTick();
519
520 if (tdma_ctx.state == TDMA_SYNCED) {
521 handle_synced_mode();
522 } else {
523 handle_unsynced_mode();
524 }
525
526 // Stats every 10 seconds
527 if ((now - last_stats_tick) >= 10000) {
528 (void)E22_GetStats();
529
530 printf("[RADIO] %s F=%u slot=%u | TX: f=%lu s=%lu | RX: cmd=%lu sync=%lu | CRC=%lu | ack_seq=%u\r\n",
531 tdma_ctx.state == TDMA_SYNCED ? "SYNC" : "UNSYNC",
532 tdma_ctx.frame_id, tdma_ctx.slot_index,
533 radio_stats.tx_fast, radio_stats.tx_slow,
534 radio_stats.rx_cmd, radio_stats.rx_sync,
535 radio_stats.crc_errors,
536 tdma_ctx.last_cmd_seq_received);
537
538 last_stats_tick = now;
539 }
540
541 // Fast loop for responsive operation
542 vTaskDelay(pdMS_TO_TICKS(2));
543 }

816static void process_command(command_packet_t *cmd) {
817 uint16_t calc_crc = crc16_calculate((uint8_t*)cmd, sizeof(command_packet_t) - 2);
818 if (calc_crc != cmd->crc16) {
819 radio_stats.crc_errors++;
820 return;
821 }
822
823 radio_stats.rx_cmd++;
824 tdma_ctx.last_cmd_seq_received = cmd->cmd_seq;
825
826 fsm_cmd_msg_t fsm_msg = {0};
827 fsm_msg.cmd = (fsm_command_t)cmd->cmd_id;
828 fsm_msg.cmd_seq = cmd->cmd_seq;
829 memcpy(&fsm_msg.payload, cmd->params, sizeof(cmd->params));
830
831 if (fsm_send_command(fsm_msg.cmd, fsm_msg.cmd_seq, &fsm_msg.payload, sizeof(fsm_msg.payload))) {
832 tdma_ctx.last_cmd_status = 0;
833 printf("[RADIO] CMD %d -> FSM queue\r\n", cmd->cmd_id);
834 } else {
835 tdma_ctx.last_cmd_status = 2;
836 printf("[RADIO] CMD queue FULL!\r\n");
837 }
838
839 sync_to_gs(cmd->frame_id, cmd->time);
840}

849static void process_sync(sync_packet_t *sync) {

851 if (calc_crc != sync->crc16) {
852 radio_stats.crc_errors++;
853 return;
854 }

856 radio_stats.rx_sync++;
857 sync_to_gs(sync->frame_id, sync->gs_time);

868static void process_rx_data(void) {
869 if (!SX126x_Available()) return;
870
871 uint8_t rx_buf[RADIO_MAX_PACKET_SIZE];
872 SX126x_RxInfo_t rx_info;
873 int len = SX126x_Receive(rx_buf, sizeof(rx_buf), &rx_info);
874
875 if (len <= 0) return;
876
877 uint8_t type = rx_buf[0];
878 uint16_t expected = get_packet_size(type);
879
880 if (expected == 0 || len < expected) return;
881
882 if (type == TELEM_PACKET_COMMAND) {
883 process_command((command_packet_t*)rx_buf);
884 } else if (type == TELEM_PACKET_SYNC) {
885 process_sync((sync_packet_t*)rx_buf);
886 }

902static void handle_synced_mode(void) {
903 uint32_t now = HAL_GetTick();
904
905 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
906 advance_frame();
907 tdma_ctx.missed_sync_count++;
908 if (tdma_ctx.missed_sync_count >= SYNC_TIMEOUT_FRAMES) {
909 tdma_ctx.state = TDMA_UNSYNCED;
910 printf("[RADIO] Lost sync!\r\n");
911 return;
912 }
913 }
914
915 uint8_t slot = get_current_slot();
916 uint32_t time_in_slot = get_time_in_slot();
917 tdma_ctx.slot_index = slot;
918
919 static uint8_t last_slot = 255;
920 if (slot != last_slot) {
921 last_slot = slot;
922 tx_done_this_slot = false;
923 }
924
925 if (slot < TDMA_TX_SLOTS) {
926 // Slots 0-7: Fast telemetry
927 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < TX_END_OFFSET_MS) {
928 tx_pending_events();
929 tx_fast_telemetry(slot);
930 }
931 if (time_in_slot >= TX_END_OFFSET_MS || tx_done_this_slot) {
932 process_rx_data();
933 }
934 }
935 else if (slot == TDMA_SLOW_SLOT) {
936 // Slot 8: Slow telemetry
937 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < (SLOT_DURATION_MS - RX_SLOT_GUARD_MS)) {
938 tx_pending_events();
939 tx_slow_telemetry(slot);
940 }
941 if (tx_done_this_slot) {
942 process_rx_data();
943 }
944 }
945 else {
946 // Slot 9: RX only
947 process_rx_data();
948 }
949}
950
959static void handle_unsynced_mode(void) {
960 uint32_t now = HAL_GetTick();
961
962 tx_pending_events();
963
964 // Send beacon periodically
965 if ((now - last_unsync_beacon_tick) >= UNSYNC_BEACON_INTERVAL_MS) {
966 if (!SX126x_IsTxBusy()) {
967 telemetry_fast_t pkt;
968 if (sensor_mutex && xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
969 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
970 tdma_ctx.fast_seq++, &fsm_ctx,
971 &latest_imu, &latest_baro, &latest_bno,
972 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
973 xSemaphoreGive(sensor_mutex);
974 } else {
975 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
976 tdma_ctx.fast_seq++, &fsm_ctx,
977 NULL, NULL, NULL,
978 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
979 }
980
981 if (SX126x_TransmitDMA((uint8_t*)&pkt, sizeof(pkt))) {
982 radio_stats.tx_fast++;
983 last_tx_tick = HAL_GetTick();
984 }
985 last_unsync_beacon_tick = now;
986 }
987 }
988
989 process_rx_data();
990
991 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
992 tdma_ctx.frame_id++;
993 tdma_ctx.frame_start_tick = now;
994 }
995}

1002void radio_thread_function() {
1003 printf("[RADIO] SPI thread starting...\r\n");
1004 fsm_report_thread_started("RADIO");
1005
1006 sensor_mutex = xSemaphoreCreateMutex();
1007 if (!sensor_mutex) {
1008 printf("[RADIO] Mutex create failed!\r\n");
1009 }
1010
1011 // Initialize SX126x with LoRa config
1012 if (!SX126x_Init(&lora_config)) {
1013 printf("[RADIO] SX126x Init FAIL\r\n");
1014 fsm_report_init_status("RADIO", false);
1015 vTaskSuspend(NULL);
1016 }
1017 fsm_report_init_status("RADIO", true);
1018
1019 printf("[RADIO] SX126x initialized — freq=%luHz SF=%u BW=%u\r\n",
1020 lora_config.frequency_hz,
1021 lora_config.spreading_factor,
1022 lora_config.bandwidth);
1023 printf("[RADIO] Packet sizes: FAST=%u SLOW=%u CMD=%u SYNC=%u\r\n",
1024 sizeof(telemetry_fast_t), sizeof(telemetry_slow_t),
1025 sizeof(command_packet_t), sizeof(sync_packet_t));
1026
1027 // Enter continuous RX mode
1028 SX126x_SetRx(0xFFFFFF);
1029
1030 // Start in UNSYNCED state
1031 memset(&tdma_ctx, 0, sizeof(tdma_ctx));
1032 tdma_ctx.state = TDMA_UNSYNCED;
1033 tdma_ctx.frame_start_tick = HAL_GetTick();
1034 last_tx_tick = 0;
1035 last_unsync_beacon_tick = 0;
1036 tx_done_this_slot = false;
1037
1038 uint32_t last_stats_tick = HAL_GetTick();
1039
1040 printf("[RADIO] Waiting for GS sync...\r\n");
1041
1042 while (1) {
1043 uint32_t now = HAL_GetTick();
1044
1045 if (tdma_ctx.state == TDMA_SYNCED) {
1046 handle_synced_mode();
1047 } else {
1048 handle_unsynced_mode();
1049 }
1050
1051 // Stats every 10 seconds
1052 if ((now - last_stats_tick) >= 10000) {
1053 printf("[RADIO] %s F=%u slot=%u | TX: f=%lu s=%lu | RX: cmd=%lu sync=%lu | CRC=%lu | RSSI=%d SNR=%d\r\n",
1054 tdma_ctx.state == TDMA_SYNCED ? "SYNC" : "UNSYNC",
1055 tdma_ctx.frame_id, tdma_ctx.slot_index,
1056 radio_stats.tx_fast, radio_stats.tx_slow,
1057 radio_stats.rx_cmd, radio_stats.rx_sync,
1058 radio_stats.crc_errors,
1059 SX126x_GetRssi(), SX126x_GetSnr());
1060 last_stats_tick = now;
1061 }
1062
1063 vTaskDelay(pdMS_TO_TICKS(2));
1064 }

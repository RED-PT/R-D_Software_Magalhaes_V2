175static void send_telem_event(telemetry_event_type_t type, const void *payload, uint16_t size) {
176 telemetry_event_t evt;
177 memset(&evt, 0, sizeof(evt));
178
179 evt.packet_type = TELEM_PACKET_EVENT;
180 evt.time = HAL_GetTick();
181 evt.event_type = (uint8_t)type;
182 evt.state = (uint8_t)fsm_ctx.state;
183 evt.substate = (uint8_t)fsm_ctx.substate;
184
185 if (payload && size > 0) {
186 uint16_t copy_size = (size > sizeof(evt.payload)) ? sizeof(evt.payload) : size;
187 memcpy(&evt.payload, payload, copy_size);
188 }
189
190 evt.crc16 = crc16_calculate((uint8_t*)&evt, sizeof(telemetry_event_t) - 2);
191
192 // Queue event directly to radio for immediate transmission
193 // (queue_fsm_events holds the full struct, not a pointer)
194 if (queue_fsm_events != NULL) {
195 if (xQueueSend(queue_fsm_events, &evt, pdMS_TO_TICKS(10)) != pdTRUE) {
196 printf("[FSM] WARNING: Event queue full!\r\n");
197 }
198 }

204static void transition_to(fsm_state_t new_state, fsm_substate_t new_sub, telemetry_event_type_t reason) {
205 fsm_state_t old_state = fsm_ctx.state;
206 fsm_substate_t old_sub = fsm_ctx.substate;
207
208 if (old_state == new_state && old_sub == new_sub) return;
209
210 fsm_ctx.prev_state = old_state;
211 fsm_ctx.prev_substate = old_sub;
212 fsm_ctx.state = new_state;
213 fsm_ctx.substate = new_sub;
214 fsm_ctx.state_entry_tick = HAL_GetTick();
215
216 printf("[FSM] %s.%s -> %s.%s (reason=%u)\r\n",
217 fsm_state_to_str(old_state), fsm_substate_to_str(old_sub),
218 fsm_state_to_str(new_state), fsm_substate_to_str(new_sub),
219 (uint8_t)reason);
220
221 // Send state change event
222 event_state_change_t change = {
223 .old_state = (uint8_t)old_state,
224 .old_substate = (uint8_t)old_sub,
225 .new_state = (uint8_t)new_state,
226 .new_substate = (uint8_t)new_sub,
227 .reason = (uint8_t)reason
228 };
229 send_telem_event(EVT_STATE_CHANGE, &change, sizeof(change));
230
231 // State-specific actions on entry
232 switch (new_state) {
233 case STATE_IDLE:
234 // Suspend estimator and controller
235 if (estimator_thread_id != NULL) {
236 vTaskSuspend(estimator_thread_id);
237 fsm_ctx.flags.estimator_running = 0;
238 printf("[FSM] Estimator suspended\r\n");
239 }
240 if (controller_thread_id != NULL) {
241 vTaskSuspend(controller_thread_id);
242 fsm_ctx.flags.controller_running = 0;
243 printf("[FSM] Controller suspended\r\n");
244 }
245 fsm_ctx.flags.sd_logging_enabled = 0;
246 break;
247
248 case STATE_CONFIGED:
249 // Profile loaded, waiting for ARM
250 // Keep estimator/controller suspended
251 break;
252
253 case STATE_ARMED:
254 // Start motor arm sequence
255 fsm_ctx.substate = SUB_ARM_MOTOR_INIT;
256 motor_arm_start_tick = HAL_GetTick();
257
258 // Resume estimator for sensor monitoring
259 if (estimator_thread_id != NULL) {
260 vTaskResume(estimator_thread_id);
261 fsm_ctx.flags.estimator_running = 1;
262 printf("[FSM] Estimator resumed\r\n");
263 }
264
265 // Enable SD logging
266 fsm_ctx.flags.sd_logging_enabled = 1;
267 break;
268
269 case STATE_TEST_STAND:
270 case STATE_FLIGHT:
271 // Resume controller for active control
272 if (controller_thread_id != NULL) {
273 vTaskResume(controller_thread_id);
274 fsm_ctx.flags.controller_running = 1;
275 printf("[FSM] Controller resumed\r\n");
276 }
277 // Estimator should already be running from ARMED
278 if (estimator_thread_id != NULL && !fsm_ctx.flags.estimator_running) {
279 vTaskResume(estimator_thread_id);
280 fsm_ctx.flags.estimator_running = 1;
281 }
282 fsm_ctx.flags.sd_logging_enabled = 1;
283 break;
284
285 case STATE_SAFE:
286 case STATE_ABORT:
287 // Emergency: cut throttle, suspend controller
288 controller_set_throttle(0.0f);
289 controller_emergency_stop();
290
291 if (controller_thread_id != NULL) {
292 vTaskSuspend(controller_thread_id);
293 fsm_ctx.flags.controller_running = 0;
294 }
295 // Keep estimator for post-flight analysis
296 break;
297
298 default:
299 break;
300 }

306static void handle_state_boot(void) {
307 uint32_t elapsed = HAL_GetTick() - fsm_ctx.boot_start_tick;
308 boot_status_t *bs = &fsm_ctx.boot_status;
309
310 // Wait minimum time for threads to start
311 if (elapsed < BOOT_MIN_WAIT_MS) {
312 return;
313 }
314
315 // Check if boot complete
316 bool boot_ok = fsm_is_boot_complete();
317
318 // Check for timeout
319 if (elapsed >= BOOT_TIMEOUT_MS) {
320 printf("[FSM] BOOT TIMEOUT after %lu ms\r\n", elapsed);
321
322 build_boot_report();
323 printf("[FSM] Boot Report: %u/%u passed, %u critical failures\r\n",
324 boot_report.passed, boot_report.total_checks, boot_report.critical);
325
326 for (int i = 0; i < boot_report.error_count; i++) {
327 printf("[FSM] - %s\r\n", boot_report.errors[i]);
328 }
329
330 send_telem_event(EVT_CHECKS_RED, &boot_report, sizeof(boot_report));
331
332 if (bs->critical_failures > 0) {
333 printf("[FSM] CRITICAL FAILURES - going to SAFE\r\n");
334 transition_to(STATE_SAFE, SUB_NONE, EVT_CHECKS_RED);
335 } else {
336 printf("[FSM] Non-critical failures, proceeding to IDLE\r\n");
337 transition_to(STATE_IDLE, SUB_NONE, EVT_CHECKS_RED);
338 }
339 return;
340 }
341
342 // Boot complete successfully?
343 if (boot_ok) {
344 printf("[FSM] Boot complete in %lu ms\r\n", elapsed);
345
346 build_boot_report();
347 printf("[FSM] Boot Report: %u/%u passed\r\n",
348 boot_report.passed, boot_report.total_checks);
349
350 send_telem_event(EVT_CHECKS_GREEN, &boot_report, sizeof(boot_report));
351 transition_to(STATE_IDLE, SUB_NONE, EVT_CHECKS_GREEN);
352 }

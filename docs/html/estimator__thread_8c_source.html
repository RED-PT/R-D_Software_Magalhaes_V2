19#define LANDED_TIME_MS 3000

23static bool liftoff_detected = false;
24static bool apogee_detected = false;
25static bool flare_triggered = false;
26static bool touchdown_detected = false;
27static uint32_t touchdown_time = 0;

29// ============================================================================
30// Flight Event Detection
31// ============================================================================
32static void check_flight_events(float altitude_m, float velocity_ms) {
33 // Update FSM context with current nav data
34 fsm_ctx.altitude_agl_m = altitude_m;
35 fsm_ctx.velocity_vertical_ms = velocity_ms;
36
37 // Track max altitude
38 if (altitude_m > max_altitude_m) {
39 max_altitude_m = altitude_m;
40 fsm_ctx.max_altitude_reached_m = max_altitude_m;
41 }
42
43 // Only detect events during flight
44 if (fsm_ctx.state != STATE_FLIGHT) {
45 return;
46 }
47
48 // Liftoff detection
49 if (!liftoff_detected && velocity_ms > LIFTOFF_VELOCITY_THRESHOLD_MS) {
50 liftoff_detected = true;
51 printf("[EST] LIFTOFF detected! vel=%.2f m/s\r\n", velocity_ms);
52 fsm_send_event(FSM_EVT_LIFTOFF, &velocity_ms);
53 }
54
55 // Apogee detection (velocity crosses zero going negative)
56 if (liftoff_detected && !apogee_detected) {
57 if (velocity_ms < 0 && fabsf(velocity_ms) > APOGEE_VELOCITY_THRESHOLD_MS) {
58 apogee_detected = true;
59 printf("[EST] APOGEE detected! alt=%.2f m\r\n", altitude_m);
60 fsm_send_event(FSM_EVT_APOGEE, &altitude_m);
61 }
62 }
63
64 // Flare altitude detection
65 if (apogee_detected && !flare_triggered) {
66 if (altitude_m <= fsm_ctx.profile.flare_altitude_m) {
67 flare_triggered = true;
68 printf("[EST] FLARE altitude reached! alt=%.2f m\r\n", altitude_m);
69 fsm_send_event(FSM_EVT_FLARE_ALT, &altitude_m);
70 }
71 }
72
73 // Touchdown detection
74 if (flare_triggered && !touchdown_detected) {
75 if (altitude_m <= TOUCHDOWN_ALT_THRESHOLD_M) {
76 touchdown_detected = true;
77 touchdown_time = HAL_GetTick();
78 printf("[EST] TOUCHDOWN detected! alt=%.2f m\r\n", altitude_m);
79 fsm_send_event(FSM_EVT_TOUCHDOWN, &altitude_m);
80 }
81 }
82
83 // Landed confirmation (stable on ground for 3 seconds)
84 if (touchdown_detected) {
85 if (fabsf(velocity_ms) < LANDED_VELOCITY_THRESHOLD_MS) {
86 if ((HAL_GetTick() - touchdown_time) > LANDED_TIME_MS) {
87 printf("[EST] LANDED confirmed!\r\n");
88 fsm_send_event(FSM_EVT_LANDED, NULL);
89 }
90 } else {
91 // Reset timer if still moving
92 touchdown_time = HAL_GetTick();
93 }
94 }
95
96 // Safety: Altitude limit exceeded
97 if (altitude_m > fsm_ctx.profile.max_altitude_m) {
98 printf("[EST] WARNING: Max altitude exceeded!\r\n");
99 fsm_send_event(FSM_EVT_ALTITUDE_LIMIT, &altitude_m);
100 }
101
102 // Safety: Velocity limit exceeded
103 if (fabsf(velocity_ms) > fsm_ctx.profile.max_velocity_ms) {
104 printf("[EST] WARNING: Max velocity exceeded!\r\n");
105 fsm_send_event(FSM_EVT_VELOCITY_LIMIT, &velocity_ms);
106 }
107}

121// ============================================================================
122// Main Thread
123// ============================================================================
124void estimator_thread_function() {
125
126 printf("[EST] Estimator Thread started...\r\n");
127 fsm_report_thread_started("ESTIMATOR");
128
129 data_packet_t packet;
130 float current_altitude_m = 0.0f;
131 float current_velocity_ms = 0.0f;
132
133 // Simple velocity estimation from altitude changes
134 float prev_altitude_m = 0.0f;
135 uint32_t prev_time_ms = 0;
136
137 TickType_t last_stats = xTaskGetTickCount();
138
139 while(1) {
140 // Receive sensor data
141 if (xQueueReceive(queue_to_estimator, &packet, pdMS_TO_TICKS(100)) == pdTRUE) {
142 data_packet_lock(&packet);
143
144 switch(packet.type) {
145 case DATA_TYPE_BARO: {
146 BARO_t baro;
147 data_packet_copy_baro(&packet, &baro);
148 current_altitude_m = baro.altitude_m;
149
150 // Simple velocity estimation (derivative of altitude)
151 uint32_t now = HAL_GetTick();
152 if (prev_time_ms > 0) {
153 float dt_s = (now - prev_time_ms) / 1000.0f;
154 if (dt_s > 0.001f) {
155 current_velocity_ms = (current_altitude_m - prev_altitude_m) / dt_s;
156 }
157 }
158 prev_altitude_m = current_altitude_m;
159 prev_time_ms = now;
160 break;
161 }
162
163 case DATA_TYPE_IMU: {
164 // TODO: Use IMU for better state estimation
165 break;
166 }
167
168 default:
169 break;
170 }
171
172 data_packet_unlock(&packet);
173
174 // Check for flight events
175 check_flight_events(current_altitude_m, current_velocity_ms);
176 }
177
178 // Reset detection when entering flight
179 static fsm_state_t prev_state = STATE_BOOT;
180 if (fsm_ctx.state == STATE_FLIGHT && prev_state != STATE_FLIGHT) {
181 reset_flight_detection();
182 }
183 prev_state = fsm_ctx.state;
184
185 // Periodic stats
186 TickType_t now = xTaskGetTickCount();
187 if ((now - last_stats) >= pdMS_TO_TICKS(10000)) {
188 printf("[EST] Alt=%.2f m, Vel=%.2f m/s, MaxAlt=%.2f m\r\n",
189 current_altitude_m, current_velocity_ms, max_altitude_m);
190 last_stats = now;
191 }
192 }
193}

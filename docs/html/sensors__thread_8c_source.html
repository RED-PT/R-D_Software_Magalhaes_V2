142 fsm_report_init_status("GPS_INIT", gps_ok);
143
144 if (!gps_ok) {

153

156 printf("GPS config saved to flash\r\n");

158 #else

160 #endif
161
162 bool gps_dma = UBLOX_GPS_StartDMA(&gps_device);
163 fsm_report_init_status("GPS_CONFIG", gps_dma);
164 if (!gps_dma) {
165 printf("ERROR: GPS DMA start failed!\r\n");
166 }
167 }
168
169 printf("Sensors initialization complete!\r\n");
170}

173void sensors_thread_function(void *argument) {
174 data_handler_init();
175 sensors_thread_init();
176
177 printf("Sensors thread started...\r\n");
178 fsm_report_thread_started("SENSORS");
179
180 uint32_t ulNotificationValue;
181 const TickType_t xMaxBlockTime = pdMS_TO_TICKS(10);
182
183 ulTaskNotifyTake(pdTRUE, 0);
184
185 xTimerStart(xBaroTimer, 0);
186 xTimerStart(xBnoTimer, 0);
187
188 static uint32_t gps_update_calls = 0;
189 static uint32_t gps_sentences_found = 0;
190 static uint32_t gps_last_sat_count = 0;
191
192 TickType_t last_stats_time = xTaskGetTickCount();
193
194 while(1) {
195 xTaskNotifyWait(0, ULONG_MAX, &ulNotificationValue, xMaxBlockTime);
196
197 // [REST OF YOUR ORIGINAL LOOP CODE - UNCHANGED]
198
199 if (ulNotificationValue & SENSOR_NOTIFY_IMU_DRDY) {
200 if (spi1_active_sensor == ACTIVE_SENSOR_NONE) {
201 spi1_active_sensor = ACTIVE_SENSOR_IMU;
202 if (!ASM330LHHX_StartReadDMA(&imu_device)) {
203 spi1_active_sensor = ACTIVE_SENSOR_NONE;
204 sensor_stats.imu_errors++;
205 }
206 }
207 }
208
209 if (ulNotificationValue & SENSOR_NOTIFY_MAG_DRDY) {
210 if (spi3_active_sensor == ACTIVE_SENSOR_NONE) {
211 spi3_active_sensor = ACTIVE_SENSOR_MAG;
212 if (!MMC5983MA_StartReadDMA(&mag_device)) {
213 spi3_active_sensor = ACTIVE_SENSOR_NONE;
214 sensor_stats.mag_errors++;
215 }
216 }
217 }
218
219 if (ulNotificationValue & SENSOR_NOTIFY_BARO_TIMER) {
220 BARO_t baro_data;
221 // Use calibrated reading to get AGL altitude (0 at launch pad)
222 if (MS5607_ReadWithCalibration(&baro_device, &baro_data, fsm_get_baro_calibration())) {
223 sensor_stats.baro_samples++;
224 data_handler_store_baro(&baro_data);
225 } else {
226 sensor_stats.baro_errors++;
227 }
228 }
229
230 if (ulNotificationValue & SENSOR_NOTIFY_BNO_TIMER) {
231 if (i2c1_active_sensor == ACTIVE_SENSOR_NONE) {
232 i2c1_active_sensor = ACTIVE_SENSOR_BNO;
233 i2c1_dma_start_tick = HAL_GetTick(); // Track DMA start time
234 if (!BNO055_StartReadDMA(&bno_device)) {
235 i2c1_active_sensor = ACTIVE_SENSOR_NONE;
236 sensor_stats.bno_errors++;
237 }
238 }
239 }
240
241 // Check for I2C DMA timeout (BNO055)
242 if (i2c1_active_sensor != ACTIVE_SENSOR_NONE) {
243 uint32_t elapsed = HAL_GetTick() - i2c1_dma_start_tick;
244 if (elapsed > DMA_TIMEOUT_MS) {
245 printf("[SENSORS] I2C1 DMA timeout! sensor=%d, elapsed=%lu ms\r\n",
246 i2c1_active_sensor, elapsed);
247 sensor_stats.i2c_timeouts++;
248 sensor_stats.bno_errors++;
249
250 // Recover the I2C bus
251 I2C_BusRecovery(bno_device.hi2c);
252
253 // Reset active sensor flag
254 i2c1_active_sensor = ACTIVE_SENSOR_NONE;
255 }
256 }
257
258 if (ulNotificationValue & SENSOR_NOTIFY_GPS_DATA) {
259 if (UBLOX_GPS_Update(&gps_device)) {
260 gps_sentences_found++;
261 ulNotificationValue |= SENSOR_NOTIFY_GPS_DR;
262 }
263 }
264
265 if (ulNotificationValue & SENSOR_NOTIFY_DMA_COMPLETE) {
266 if (spi1_active_sensor == ACTIVE_SENSOR_IMU) {
267 IMU_t imu_data;
268 if (ASM330LHHX_ProcessData(&imu_device, &imu_data)) {
269 sensor_stats.imu_samples++;
270 data_handler_store_imu(&imu_data);
271 } else {
272 sensor_stats.imu_errors++;
273 }
274 spi1_active_sensor = ACTIVE_SENSOR_NONE;
275 }
276 else if (spi3_active_sensor == ACTIVE_SENSOR_MAG) {
277 MAG_t mag_data;
278 if (MMC5983MA_ProcessData(&mag_device, &mag_data)) {
279 sensor_stats.mag_samples++;
280 data_handler_store_mag(&mag_data);
281 } else {
282 sensor_stats.mag_errors++;
283 }
284 spi3_active_sensor = ACTIVE_SENSOR_NONE;
285 }
286 else if (i2c1_active_sensor == ACTIVE_SENSOR_BNO) {
287 BNO_t bno_data;
288 if (BNO055_ProcessData(&bno_device, &bno_data)) {
289 sensor_stats.bno_samples++;
290 data_handler_store_bno(&bno_data);
291 } else {
292 sensor_stats.bno_errors++;
293 }
294 i2c1_active_sensor = ACTIVE_SENSOR_NONE;
295 }
296 }
297
298 if (ulNotificationValue & SENSOR_NOTIFY_GPS_DR) {
299 GPS_t gps_data;
300 if (UBLOX_GPS_ProcessData(&gps_device, &gps_data)) {
301 sensor_stats.gps_samples++;
302 gps_last_sat_count = gps_data.satellites;
303 data_handler_store_gps(&gps_data);
304 } else {
305 sensor_stats.gps_errors++;
306 }
307 }
308
309 if (ulNotificationValue & SENSOR_NOTIFY_DMA_ERROR) {
310 printf("ERROR: DMA error occurred\r\n");
311 CS_IMU_HIGH();
312 CS_BARO_HIGH();
313 CS_MAG_HIGH();
314 spi1_active_sensor = ACTIVE_SENSOR_NONE;
315 spi3_active_sensor = ACTIVE_SENSOR_NONE;
316 i2c1_active_sensor = ACTIVE_SENSOR_NONE;
317 }
318
319 if ((xTaskGetTickCount() - last_stats_time) > pdMS_TO_TICKS(10000)) {
320 printf("Sensor Statistics\r\n");
321 printf("IMU: %lu samples, %lu errors\r\n", sensor_stats.imu_samples, sensor_stats.imu_errors);
322 printf("MAG: %lu samples, %lu errors\r\n", sensor_stats.mag_samples, sensor_stats.mag_errors);
323 printf("BARO: %lu samples, %lu errors\r\n", sensor_stats.baro_samples, sensor_stats.baro_errors);
324 printf("BNO: %lu samples, %lu errors\r\n", sensor_stats.bno_samples, sensor_stats.bno_errors);
325 printf("GPS: %lu samples, %lu errors, %lu sats, updates=%lu, sentences=%lu\r\n",
326 sensor_stats.gps_samples, sensor_stats.gps_errors,
327 gps_last_sat_count, gps_update_calls, gps_sentences_found);
328 printf("DMA errors: %lu, I2C timeouts: %lu, I2C recoveries: %lu\r\n",
329 sensor_stats.dma_errors, sensor_stats.i2c_timeouts, sensor_stats.i2c_recoveries);
330 last_stats_time = xTaskGetTickCount();
331 }
332 }

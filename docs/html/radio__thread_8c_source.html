93static bool can_transmit(void) {
94 // Don't TX if we recently transmitted (E22 turnaround)
95 uint32_t since_tx = HAL_GetTick() - last_tx_tick;
96 if (since_tx < E22_TURNAROUND_MS) return false;
97
98 // Don't TX if E22 is busy
99 if (E22_IsBusy()) return false;
100
101 // Already transmitted this slot
102 if (tx_done_this_slot) return false;
103
104 return true;
105}

107static void sync_to_gs(uint8_t gs_frame_id, uint32_t gs_time) {
108 uint32_t now = HAL_GetTick();
109
110 // When we receive a sync, we know GS sent it at the START of slot 9
111 // Align our frame so slot 9 starts at this moment
112 uint32_t new_frame_start = now - (TDMA_RX_SLOT * SLOT_DURATION_MS);
113
114 if (tdma_ctx.state == TDMA_SYNCED) {
115 // Already synced - check if we need correction
116 int32_t error = (int32_t)(new_frame_start - tdma_ctx.frame_start_tick);
117
118 if (error > SYNC_CORRECTION_THRESH || error < -SYNC_CORRECTION_THRESH) {
119 tdma_ctx.frame_start_tick = new_frame_start;
120 radio_stats.sync_corrections++;
121 }
122 } else {
123 // First sync - hard align
124 tdma_ctx.frame_start_tick = new_frame_start;
125 tdma_ctx.state = TDMA_SYNCED;
126 printf("[RADIO] SYNCED! frame=%u\r\n", gs_frame_id);
127 }
128
129 tdma_ctx.frame_id = gs_frame_id;
130 tdma_ctx.last_sync_tick = now;
131 tdma_ctx.missed_sync_count = 0;
132}

186static void tx_pending_events(void) {
187 if (!can_transmit()) return;
188 if (queue_fsm_events == NULL) return;
189
190 telemetry_event_t evt;
191
192 // Check for pending events (non-blocking)
193 if (xQueueReceive(queue_fsm_events, &evt, 0) == pdTRUE) {
194 // Update frame/slot info
195 evt.frame_id = tdma_ctx.frame_id;
196 evt.slot_id = tdma_ctx.slot_index;
197 evt.seq = tdma_ctx.event_seq++;
198
199 // Recalculate CRC after updating header
200 evt.crc16 = crc16_calculate((uint8_t*)&evt, sizeof(telemetry_event_t) - 2);
201
202 if (E22_Transmit((uint8_t*)&evt, sizeof(evt)) == E22_OK) {
203 radio_stats.tx_event++;
204 last_tx_tick = HAL_GetTick();
205 printf("[RADIO] Event TX: type=%u\r\n", evt.event_type);
206 }
207 }

221static void process_command(command_packet_t *cmd) {
222 uint16_t calc_crc = crc16_calculate((uint8_t*)cmd, sizeof(command_packet_t) - 2);
223 if (calc_crc != cmd->crc16) {
224 radio_stats.crc_errors++;
225 return;
226 }
227
228 radio_stats.rx_cmd++;
229
230 // Store for ACK piggyback
231 tdma_ctx.last_cmd_seq_received = cmd->cmd_seq;
232
233 // Forward command to FSM thread via queue
234 fsm_cmd_msg_t fsm_msg = {0};
235 fsm_msg.cmd = (fsm_command_t)cmd->cmd_id;
236 fsm_msg.cmd_seq = cmd->cmd_seq;
237 memcpy(&fsm_msg.payload, cmd->params, sizeof(cmd->params));
238
239 if (fsm_send_command(fsm_msg.cmd, fsm_msg.cmd_seq, &fsm_msg.payload, sizeof(fsm_msg.payload))) {
240 tdma_ctx.last_cmd_status = 0; // Queued OK
241 printf("[RADIO] CMD %d -> FSM queue\r\n", cmd->cmd_id);
242 } else {
243 tdma_ctx.last_cmd_status = 2; // Queue full
244 printf("[RADIO] CMD queue FULL!\r\n");
245 }
246
247 // Sync timing
248 sync_to_gs(cmd->frame_id, cmd->time);
249}
250static void process_sync(sync_packet_t *sync) {
251 uint16_t calc_crc = crc16_calculate((uint8_t*)sync, sizeof(sync_packet_t) - 2);
252 if (calc_crc != sync->crc16) {
253 radio_stats.crc_errors++;
254 return;
255 }
256
257 radio_stats.rx_sync++;
258 sync_to_gs(sync->frame_id, sync->gs_time);
259}

261static bool process_rx_byte(uint8_t byte) {
262 uint32_t now = HAL_GetTick();
263
264 // Timeout: reset buffer if no bytes for a while
265 if (rx_idx > 0 && (now - rx_last_byte_tick) > RX_TIMEOUT_MS) {
266 rx_idx = 0;
267 }
268 rx_last_byte_tick = now;
269
270 // First byte: validate packet type
271 if (rx_idx == 0) {
272 uint16_t expected = get_packet_size(byte);
273 if (expected == 0) {
274 return false; // Not a packet type we're looking for
275 }
276 }
277
278 // Store byte
279 if (rx_idx < RX_BUFFER_SIZE) {
280 rx_buffer[rx_idx++] = byte;
281 } else {
282 rx_idx = 0;
283 return false;
284 }
285
286 // Check for complete packet
287 uint8_t type = rx_buffer[0];
288 uint16_t expected = get_packet_size(type);
289
290 if (rx_idx >= expected) {
291 if (type == TELEM_PACKET_COMMAND) {
292 process_command((command_packet_t*)rx_buffer);
293 } else if (type == TELEM_PACKET_SYNC) {
294 process_sync((sync_packet_t*)rx_buffer);
295 }
296 rx_idx = 0;
297 return true;
298 }
299
300 return false;
301}

315static void handle_synced_mode(void) {
316 uint32_t now = HAL_GetTick();
317
318 // Check for frame rollover
319 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
320 advance_frame();
321
322 // Check sync timeout
323 tdma_ctx.missed_sync_count++;
324 if (tdma_ctx.missed_sync_count >= SYNC_TIMEOUT_FRAMES) {
325 tdma_ctx.state = TDMA_UNSYNCED;
326 printf("[RADIO] Lost sync!\r\n");
327 return;
328 }
329 }
330
331 // Get current slot and time within slot
332 uint8_t slot = get_current_slot();
333 uint32_t time_in_slot = get_time_in_slot();
334
335 // Update global slot_index for other code that needs it
336 tdma_ctx.slot_index = slot;
337
338 // Detect slot change -> reset tx_done flag
339 static uint8_t last_slot = 255;
340 if (slot != last_slot) {
341 last_slot = slot;
342 tx_done_this_slot = false;
343 }
344
345 // Slot behavior based on current slot
346 if (slot < TDMA_TX_SLOTS) {
347 // Slots 0-7: Fast telemetry
348 // TX window: 5ms to 70ms into slot
349 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < TX_END_OFFSET_MS) {
350 // First check for pending events (high priority - PONG, etc.)
351 tx_pending_events();
352 // Then send regular fast telemetry if we can still transmit
353 tx_fast_telemetry(slot);
354 }
355 // RX in second half of slot (after TX done)
356 if (time_in_slot >= TX_END_OFFSET_MS || tx_done_this_slot) {
357 process_rx_data();
358 }
359 }
360 else if (slot == TDMA_SLOW_SLOT) {
361 // Slot 8: Slow telemetry
362 // TX early, but stop before slot 9 to avoid collision with GS
363 if (time_in_slot >= TX_START_OFFSET_MS && time_in_slot < (SLOT_DURATION_MS - RX_SLOT_GUARD_MS)) {
364 // First check for pending events
365 tx_pending_events();
366 tx_slow_telemetry(slot);
367 }
368 // RX after TX
369 if (tx_done_this_slot) {
370 process_rx_data();
371 }
372 }
373 else {
374 // Slot 9: RX only - listen for GS commands/sync
375 process_rx_data();
376 }
377}

379static void handle_unsynced_mode(void) {
380 uint32_t now = HAL_GetTick();
381
382 // In UNSYNCED mode: mostly listen, occasionally beacon
383
384 // Always try to send pending events (like PONG) even when unsynced
385 tx_pending_events();
386
387 // Send a beacon every UNSYNC_BEACON_INTERVAL_MS so GS knows we're alive
388 if ((now - last_unsync_beacon_tick) >= UNSYNC_BEACON_INTERVAL_MS) {
389 if (!E22_IsBusy()) {
390 // Send with slot=0 since we don't know real slot
391 telemetry_fast_t pkt;
392 if (sensor_mutex && xSemaphoreTake(sensor_mutex, pdMS_TO_TICKS(2)) == pdTRUE) {
393 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
394 tdma_ctx.fast_seq++, &fsm_ctx,
395 &latest_imu, &latest_baro, &latest_bno,
396 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
397 xSemaphoreGive(sensor_mutex);
398 } else {
399 telemetry_build_fast(&pkt, tdma_ctx.frame_id, 0,
400 tdma_ctx.fast_seq++, &fsm_ctx,
401 NULL, NULL, NULL,
402 tdma_ctx.last_cmd_seq_received, tdma_ctx.last_cmd_status);
403 }
404
405 if (E22_Transmit((uint8_t*)&pkt, sizeof(pkt)) == E22_OK) {
406 radio_stats.tx_fast++;
407 last_tx_tick = HAL_GetTick();
408 }
409 last_unsync_beacon_tick = now;
410 }
411 }
412
413 // Continuously process RX data
414 process_rx_data();
415
416 // Keep frame counter advancing (free-running when unsynced)
417 if ((now - tdma_ctx.frame_start_tick) >= FRAME_DURATION_MS) {
418 tdma_ctx.frame_id++;
419 tdma_ctx.frame_start_tick = now;
420 }
421}

426void radio_thread_function() {
427 printf("[RADIO] Thread starting...\r\n");
428 fsm_report_thread_started("RADIO");
429
430 // Initialize
431 sensor_mutex = xSemaphoreCreateMutex();
432 if (!sensor_mutex) {
433 printf("[RADIO] Mutex create failed!\r\n");
434 }
435
436 if (!E22_Init(UART_RADIO)) {
437 printf("[RADIO] Init FAIL\r\n");
438 fsm_report_init_status("RADIO", false);
439 vTaskSuspend(NULL);
440 }
441 fsm_report_init_status("RADIO", true);
442 printf("[RADIO] E22 initialized\r\n");
443 printf("[RADIO] Packet sizes: FAST=%u SLOW=%u CMD=%u SYNC=%u\r\n",
444 sizeof(telemetry_fast_t), sizeof(telemetry_slow_t),
445 sizeof(command_packet_t), sizeof(sync_packet_t));
446
447 // Start in UNSYNCED state
448 memset(&tdma_ctx, 0, sizeof(tdma_ctx));
449 tdma_ctx.state = TDMA_UNSYNCED;
450 tdma_ctx.frame_start_tick = HAL_GetTick();
451 last_tx_tick = 0;
452 last_unsync_beacon_tick = 0;
453 tx_done_this_slot = false;
454
455 uint32_t last_stats_tick = HAL_GetTick();
456
457 printf("[RADIO] Waiting for GS sync...\r\n");
458
459 while (1) {
460 uint32_t now = HAL_GetTick();
461
462 if (tdma_ctx.state == TDMA_SYNCED) {
463 handle_synced_mode();
464 } else {
465 handle_unsynced_mode();
466 }
467
468 // Stats every 10 seconds
469 if ((now - last_stats_tick) >= 10000) {
470 (void)E22_GetStats();
471
472 printf("[RADIO] %s F=%u slot=%u | TX: f=%lu s=%lu | RX: cmd=%lu sync=%lu | CRC=%lu | ack_seq=%u\r\n",
473 tdma_ctx.state == TDMA_SYNCED ? "SYNC" : "UNSYNC",
474 tdma_ctx.frame_id, tdma_ctx.slot_index,
475 radio_stats.tx_fast, radio_stats.tx_slow,
476 radio_stats.rx_cmd, radio_stats.rx_sync,
477 radio_stats.crc_errors,
478 tdma_ctx.last_cmd_seq_received);
479
480 last_stats_tick = now;
481 }
482
483 // Fast loop for responsive operation
484 vTaskDelay(pdMS_TO_TICKS(2));
485 }

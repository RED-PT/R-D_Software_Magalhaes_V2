341 BNO_t bno_data;
342 if (BNO055_ProcessData(&bno_device, &bno_data)) {
343 sensor_stats.bno_samples++;
344 data_handler_store_bno(&bno_data);
345 } else {
346 sensor_stats.bno_errors++;

353 GPS_t gps_data;
354 if (UBLOX_GPS_ProcessData(&gps_device, &gps_data)) {
355 sensor_stats.gps_samples++;
356 gps_last_sat_count = gps_data.satellites;
357 data_handler_store_gps(&gps_data);
358 } else {
359 sensor_stats.gps_errors++;
360 }
361 }
362
363 if (ulNotificationValue & SENSOR_NOTIFY_DMA_ERROR) {
364 printf("ERROR: DMA error occurred\r\n");
365 CS_IMU_HIGH();
366 CS_BARO_HIGH();
367 CS_MAG_HIGH();
368 spi1_active_sensor = ACTIVE_SENSOR_NONE;
369 spi3_active_sensor = ACTIVE_SENSOR_NONE;
370 i2c1_active_sensor = ACTIVE_SENSOR_NONE;
371 }
372
373 if ((xTaskGetTickCount() - last_stats_time) > pdMS_TO_TICKS(10000)) {
374 printf("Sensor Statistics\r\n");
375 printf("IMU: %lu samples, %lu errors\r\n", sensor_stats.imu_samples, sensor_stats.imu_errors);
376 printf("MAG: %lu samples, %lu errors\r\n", sensor_stats.mag_samples, sensor_stats.mag_errors);
377 printf("BARO: %lu samples, %lu errors\r\n", sensor_stats.baro_samples, sensor_stats.baro_errors);
378 printf("BNO: %lu samples, %lu errors\r\n", sensor_stats.bno_samples, sensor_stats.bno_errors);
379 printf("GPS: %lu samples, %lu errors, %lu sats, updates=%lu, sentences=%lu\r\n",
380 sensor_stats.gps_samples, sensor_stats.gps_errors,
381 gps_last_sat_count, gps_update_calls, gps_sentences_found);
382 printf("DMA errors: %lu, I2C timeouts: %lu, I2C recoveries: %lu\r\n",
383 sensor_stats.dma_errors, sensor_stats.i2c_timeouts, sensor_stats.i2c_recoveries);
384 last_stats_time = xTaskGetTickCount();
385 }

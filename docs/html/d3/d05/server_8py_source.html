9FastAPI-based web server for the Magalhães Flight Computer Ground Station.
10Provides real-time telemetry visualization, command interface, and data
11analysis through a modern web dashboard.
12
13## Features
14- Real-time WebSocket telemetry streaming
15- Serial communication with Arduino ground station
16- Binary protocol parsing (Fast/Slow/Event packets)
17- Command relay to flight computer
18- Demo mode for testing without hardware
19
20## Architecture
21```
22┌─────────────┐ Serial ┌─────────────┐ LoRa ┌──────────────┐
23│ Dashboard │◄───(USB)────────│ Arduino │◄──────────────│ Flight │
24│ Server │ 115200 baud │ GS │ 433 MHz │ Computer │
25└──────┬──────┘ └─────────────┘ └──────────────┘
26 │
27 │ WebSocket
28 ▼
29┌─────────────┐
30│ Browser │
31│ Dashboard │
32└─────────────┘
33```
34
35## Binary Protocol
36All messages use framed format:
37| Field | Size | Description |
38|-------|------|-------------|
39| Sync1 | 1 | 0xAA |
40| Sync2 | 1 | 0x55 |
41| Length | 2 | Payload length (big-endian) |
42| Type | 1 | Message type ID |
43| Payload | N | Message-specific data |
44| CRC16 | 2 | CRC-16 checksum |
45
46## Usage
47```bash
48python server.py # Normal mode
49python server.py --demo # Demo mode (no hardware)
50python server.py --port COM3 # Specify serial port
51```

158def parse_fast_packet(data):
159 """
160 Parse a fast telemetry packet from the flight computer.
161
162 Fast packets are transmitted at 8 Hz (slots 0-7) and contain:
163 - IMU data (accelerometer, gyroscope)
164 - Altitude and vertical speed (vario)
165 - Attitude (pitch, roll, yaw)
166 - FSM state information
167 - ACK piggyback for command confirmation
168
169 @param data: Raw packet bytes (minimum 37 bytes)
170 @return: Dictionary with parsed fields, or None if invalid
171
172 Packet format (37 bytes):
173 | Offset | Type | Field |
174 |--------|------|-------|
175 | 0 | u8 | packet_type (0x01) |
176 | 1 | u8 | frame_id |
177 | 2 | u8 | slot_id |
178 | 3 | u8 | sequence |
179 | 4 | u8 | flags |
180 | 5 | u32 | time_ms |
181 | 9 | u8 | state |
182 | 10 | u8 | substate |
183 | 11 | u8 | ack_seq |
184 | 12 | u8 | ack_status |
185 | 13 | i16[3] | accel_x/y/z (mG) |
186 | 19 | i16[3] | gyro_x/y/z (0.01 dps) |
187 | 25 | i16 | altitude (0.1 m) |
188 | 27 | i16 | vario (0.01 m/s) |
189 | 29 | i16[3] | pitch/roll/yaw (0.1 deg) |
190 | 35 | u16 | CRC16 |
191 """
192 if len(data) < 37:
193 return None
194 fmt = '<BBBBBIBBBB3h3h5hH'
195 u = struct.unpack(fmt, data[:37])
196 return {
197 'type': 'fast',
198 'frame_id': u[1], 'slot_id': u[2], 'seq': u[3], 'flags': u[4],
199 'time_ms': u[5], 'state': u[6], 'substate': u[7],
200 'ack_seq': u[8], 'ack_status': u[9],
201 'accel_x': u[10] / 1000.0, 'accel_y': u[11] / 1000.0, 'accel_z': u[12] / 1000.0,
202 'gyro_x': u[13] / 100.0, 'gyro_y': u[14] / 100.0, 'gyro_z': u[15] / 100.0,
203 'altitude': u[16] / 10.0, 'vario': u[17] / 100.0,
204 'pitch': u[18] / 10.0, 'roll': u[19] / 10.0, 'yaw': u[20] / 10.0,
205 }
206

208def parse_slow_packet(data):
209 if len(data) < 30:
210 return None
211 fmt = '<BBBBIiiHBBhhBBHH'
212 u = struct.unpack(fmt, data[:30])
213 return {
214 'type': 'slow',
215 'frame_id': u[1], 'slot_id': u[2], 'seq': u[3], 'time_ms': u[4],
216 'latitude': u[5] / 1e7, 'longitude': u[6] / 1e7,
217 'gps_altitude': u[7] / 10.0, 'gps_lock': u[8], 'satellites': u[9],
218 'pressure': (u[10] / 10.0) + 1000.0, 'temperature': u[11] / 10.0,
219 'battery': u[12], 'sd_status': u[13], 'free_heap': u[14] * 10,
220 }

278class ConnectionManager:
279 """
280 Manages WebSocket connections to browser clients.
281
282 Supports multiple simultaneous connections and broadcasts telemetry
283 data to all connected clients efficiently.
284
285 @note Automatically removes disconnected clients on broadcast errors.
286 """
287
288 def __init__(self):
289 """Initialize with empty connection set."""
290 self.active = set() #: Set of active WebSocket connections
291
292 async def connect(self, websocket: WebSocket):
293 """
294 Accept and register a new WebSocket connection.
295
296 @param websocket: FastAPI WebSocket instance
297 """
298 await websocket.accept()
299 self.active.add(websocket)
300
301 def disconnect(self, websocket: WebSocket):
302 """
303 Remove a WebSocket from the active set.
304
305 @param websocket: WebSocket to disconnect
306 """
307 self.active.discard(websocket)
308
309 async def broadcast(self, message: dict):
310 """
311 Send a message to all connected WebSocket clients.
312
313 @param message: Dictionary to send (JSON serialized)
314 @note Failed sends result in automatic client disconnect
315 """
316 if not self.active:
317 return
318 data = json.dumps(message)
319 dead = []
320 for ws in self.active:
321 try:
322 await ws.send_text(data)
323 except Exception:
324 dead.append(ws)
325 for ws in dead:
326 self.disconnect(ws)

333class SerialReader:
334 """
335 Threaded serial port reader for Arduino Ground Station communication.
336
337 Handles:
338 - Asynchronous serial reading in background thread
339 - Binary frame detection and parsing
340 - Command transmission to GS
341 - Error recovery
342
343 The reader uses a framed protocol with sync bytes (0xAA 0x55) to
344 detect packet boundaries reliably.
345
346 @note Runs in a daemon thread to avoid blocking the async event loop.
347 """
348
349 def __init__(self, loop, queue):
350 """
351 Initialize the serial reader.
352
353 @param loop: asyncio event loop for cross-thread communication
354 @param queue: asyncio.Queue for parsed packets
355 """
356 self.loop = loop #: Asyncio event loop reference
357 self.queue = queue #: Queue for parsed packets
358 self.serial = None #: pyserial Serial instance
359 self.running = False #: Thread running flag
360 self.rx_buffer = bytearray() #: Receive buffer for frame assembly
361 self.thread = None #: Background reader thread
362 self.cmd_queue = deque() #: Pending commands to send
363 self.last_rx_time = None #: Timestamp of last received data
364 self.last_error = None #: Last error message
365
366 def connect(self, port, baudrate=115200):
367 if self.running:
368 self.disconnect()
369 self.serial = serial.Serial(port, baudrate, timeout=0.05)
370 self.running = True
371 self.last_rx_time = time.time()
372 self.thread = threading.Thread(target=self.run, daemon=True)
373 self.thread.start()
374
375 def disconnect(self):
376 self.running = False
377 if self.serial:
378 try:
379 self.serial.close()
380 except Exception:
381 pass
382 self.serial = None
383 if self.thread and self.thread.is_alive():
384 self.thread.join(timeout=1.0)
385 self.thread = None
386 self.rx_buffer = bytearray()
387
388 def send_command(self, cmd):
389 if cmd:
390 self.cmd_queue.append(cmd)
391
392 def run(self):
393 while self.running:
394 if not self.serial or not self.serial.is_open:
395 time.sleep(0.1)
396 continue
397 try:
398 while self.cmd_queue:
399 cmd = self.cmd_queue.popleft()
400 self.serial.write(cmd.encode() if isinstance(cmd, str) else cmd)
401 data = self.serial.read(256)
402 if data:
403 self.last_rx_time = time.time()
404 self.rx_buffer.extend(data)
405 self.process_buffer()
406 except Exception:
407 self.last_error = "serial read error"
408 time.sleep(0.1)
409
410 def process_buffer(self):
411 while len(self.rx_buffer) >= 6:
412 idx = 0
413 while idx < len(self.rx_buffer) - 1:
414 if self.rx_buffer[idx] == FRAME_SYNC_1 and self.rx_buffer[idx + 1] == FRAME_SYNC_2:
415 break
416 idx += 1
417 if idx > 0:
418 self.rx_buffer = self.rx_buffer[idx:]
419 if len(self.rx_buffer) < 6:
420 break
421 length = (self.rx_buffer[2] << 8) | self.rx_buffer[3]
422 total_len = 4 + length + 2
423 if length > 256:
424 self.rx_buffer = self.rx_buffer[2:]
425 continue
426 if len(self.rx_buffer) < total_len:
427 break
428 frame = self.rx_buffer[:total_len]
429 self.rx_buffer = self.rx_buffer[total_len:]
430 msg_type = frame[4]
431 payload = bytes(frame[5:-2])
432 self.parse_message(msg_type, payload)
433
434 def parse_message(self, msg_type, payload):
435 pkt = None
436 if msg_type == MSG_TYPE_FC_FAST:
437 pkt = parse_fast_packet(payload)
438 elif msg_type == MSG_TYPE_FC_SLOW:
439 pkt = parse_slow_packet(payload)
440 elif msg_type == MSG_TYPE_FC_EVENT:
441 pkt = parse_event_packet(payload)
442 elif msg_type == MSG_TYPE_GS_STATUS:
443 pkt = parse_gs_status(payload)
444 elif msg_type == MSG_TYPE_GS_PONG:
445 pkt = parse_gs_pong(payload)
446 elif msg_type == MSG_TYPE_GS_STATS:
447 pkt = parse_gs_stats(payload)
448 elif msg_type == MSG_TYPE_GS_LOG:
449 pkt = {'type': 'log', 'message': payload.decode('utf-8', errors='replace')}
450 if pkt:
451 self.loop.call_soon_threadsafe(self.queue.put_nowait, pkt)

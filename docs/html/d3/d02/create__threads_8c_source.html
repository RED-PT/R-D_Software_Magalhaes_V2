221 // Terminate CubeMX default task to free resources

226 // FSM events queue: 5 events * ~60 bytes = ~300 bytes

232 queue_cmd_to_fsm = xQueueCreate(8, sizeof(fsm_cmd_msg_t));

242 printf("Queues created successfully\r\n");

244 printf("Creating timers...\r\n");

246 xBaroTimer = xTimerCreate("BaroTimer", pdMS_TO_TICKS(BARO_UPDATE_RATE_MS), pdTRUE, NULL, vBaroTimerCallback);

250 printf("ERROR: Failed to create timers\r\n");

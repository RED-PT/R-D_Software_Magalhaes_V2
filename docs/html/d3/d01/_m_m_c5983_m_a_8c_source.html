17bool MMC5983MA_Init(MMC5983MA_t *dev, SPI_HandleTypeDef *hspi) {
18 uint8_t product_id;
19
20 if (!dev || !hspi) {
21 return false;
22 }
23
24 dev->hspi = hspi;
25 dev->data_ready = 0;
26 memset(dev->read_buffer, 0, sizeof(dev->read_buffer));
27
28 HAL_Delay(10);
29
30 // Read and verify Product ID
31 product_id = 0x00;
32 if (!MMC5983MA_ReadRegister(dev, MMC5983MA_REG_PRODUCT_ID, &product_id)) {
33 printf("[MAG] Failed to read product ID\r\n");
34 return false;
35 }
36 printf("[MAG] Product ID = 0x%02X (expected 0x30)\r\n", product_id);
37 if (product_id != 0x30) { // Product ID should be 0x30
38 printf("[MAG] Product ID mismatch!\r\n");
39 return false;
40 }
41
42 // Perform software reset via SET operation
43 if (!MMC5983MA_WriteRegister(dev, MMC5983MA_REG_CTRL0, MMC5983MA_CTRL0_SET)) {
44 return false;
45 }
46
47 HAL_Delay(2);
48
49 return true;

52bool MMC5983MA_Configure(MMC5983MA_t *dev) {
53 if (!dev) {
54 return false;
55 }
56
57 // Reset to default
58 if (!MMC5983MA_WriteRegister(dev, MMC5983MA_REG_CTRL1, 0x00)) {
59 return false;
60 }
61 osDelay(1);
62
63 // Set BW[1:0] = 11 (800Hz, 0.5ms measurement time)
64 if (!MMC5983MA_WriteRegister(dev, MMC5983MA_REG_CTRL1,
65 (MMC5983MA_CTRL1_BW0 | MMC5983MA_CTRL1_BW1))) {
66 return false;
67 }
68 osDelay(1);
69
70 // Enable continuous measurement at 1000 Hz
71 if (!MMC5983MA_WriteRegister(dev, MMC5983MA_REG_CTRL2,
72 (0x04 | 0x07))) { // Cmm_en | CM_Freq=1000Hz
73 return false;
74 }
75 osDelay(1);
76
77 // Enable interrupt and auto set/reset
78 if (!MMC5983MA_WriteRegister(dev, MMC5983MA_REG_CTRL0,
79 (MMC5983MA_CTRL0_INT_EN | MMC5983MA_CTRL0_AUTO_SR))) {
80 return false;
81 }
82 osDelay(1);
83
84 return true;
85}

87bool MMC5983MA_StartReadDMA(MMC5983MA_t *dev) {
88 if (!dev || !dev->hspi) {
89 return false;
90 }
91
92 // Prepare TX buffer: [COMMAND][DUMMIES]
93 // Read from XOUT0 (0x00) through TOUT (0x07) = 8 bytes of data
94 uint8_t tx_buffer[9];
95 tx_buffer[0] = (1 << 7) | (MMC5983MA_REG_XOUT0 << 1); // Read from XOUT0
96 memset(&tx_buffer[1], 0x00, 8);
97
98 // Single DMA TransmitReceive
99 if (HAL_SPI_TransmitReceive_DMA(dev->hspi, tx_buffer, dev->read_buffer, 9) != HAL_OK) {
100 return false;
101 }
102
103 return true;
104}

106void MMC5983MA_ParseDMABuffer(MMC5983MA_t *dev) {
107 if (!dev) {
108 return;
109 }
110
111 // Parse 18-bit magnetic field data (skip first byte which is command echo)
112 // Bytes: [CMD][XOUT0][XOUT1][YOUT0][YOUT1][ZOUT0][ZOUT1][XYZout2][TOUT]
113
114 dev->mag_x_raw = ((int32_t)dev->read_buffer[1] << 10) |
115 ((int32_t)dev->read_buffer[2] << 2) |
116 ((int32_t)(dev->read_buffer[7] >> 6) & 0x03);
117
118 dev->mag_y_raw = ((int32_t)dev->read_buffer[3] << 10) |
119 ((int32_t)dev->read_buffer[4] << 2) |
120 ((int32_t)(dev->read_buffer[7] >> 4) & 0x03);
121
122 dev->mag_z_raw = ((int32_t)dev->read_buffer[5] << 10) |
123 ((int32_t)dev->read_buffer[6] << 2) |
124 ((int32_t)(dev->read_buffer[7] >> 2) & 0x03);
125
126 // Parse temperature (TOUT register at byte 8)
127 dev->temp_raw = dev->read_buffer[8];
128
129 dev->data_ready = 1;
130}

132bool MMC5983MA_ProcessData(MMC5983MA_t *dev, MAG_t *output) {
133 if (!dev || !output || !dev->data_ready) {
134 return false;
135 }
136
137 // Convert raw counts to Gauss (18-bit: 16384 Counts/G)
138 output->mag_x = (float)dev->mag_x_raw / MMC5983MA_SENSITIVITY_18BIT;
139 output->mag_y = (float)dev->mag_y_raw / MMC5983MA_SENSITIVITY_18BIT;
140 output->mag_z = (float)dev->mag_z_raw / MMC5983MA_SENSITIVITY_18BIT;
141
142 // Convert temperature: (TOUT_raw * 0.8) - 75
143 // Range: -75~125°C at 0.8°C/LSB
144 output->temperature_c = (float)dev->temp_raw * MMC5983MA_TEMP_SCALE + MMC5983MA_TEMP_OFFSET;
145 output->timestamp_ms = HAL_GetTick();
146
147 dev->data_ready = 0;
148
149 return true;
150}

155 uint8_t tx_buf[2];

167 HAL_GPIO_WritePin(CS_MAG_PORT, CS_MAG_PIN, GPIO_PIN_SET);
168 return false;
169 }
170 HAL_GPIO_WritePin(CS_MAG_PORT, CS_MAG_PIN, GPIO_PIN_SET);
171
172 *data = rx_buf[1];
173 return true;
174}

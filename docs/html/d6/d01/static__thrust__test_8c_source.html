57bool StaticTest_Start(uint8_t max_throttle_percent) {
58 // Validate parameters
59 if (max_throttle_percent < 1 || max_throttle_percent > 100) {
60 test_ctx.error_msg = "Invalid throttle percentage (1-100)";
61 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
62 return false;
63 }
64
65 // Check if test already running
66 if (test_ctx.state != STATIC_TEST_IDLE &&
67 test_ctx.state != STATIC_TEST_COMPLETE &&
68 test_ctx.state != STATIC_TEST_FAILED) {
69 test_ctx.error_msg = "Test already in progress";
70 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
71 return false;
72 }
73
74 // Reset context
75 memset(&test_ctx, 0, sizeof(test_ctx));
76
77 // Configure test
78 test_ctx.config.max_throttle_percent = max_throttle_percent;
79 test_ctx.config.ramp_duration_ms = STATIC_TEST_RAMP_DURATION_MS;
80 test_ctx.config.hold_duration_ms = STATIC_TEST_HOLD_DURATION_MS;
81 test_ctx.config.sample_interval_ms = 1000 / STATIC_TEST_SAMPLE_RATE_HZ;
82 test_ctx.max_samples = STATIC_TEST_MAX_SAMPLES;
83
84 // Generate filename
85 snprintf(test_filename, sizeof(test_filename), "static_test_pwm_%02d.csv", max_throttle_percent);
86
87 printf("[STATIC_TEST] Starting test: max=%d%%, file=%s\r\n",
88 max_throttle_percent, test_filename);
89
90 // Pause SD card to avoid EMI interference from motor
91 sd_card_pause();
92
93 // Start initialization phase
94 test_ctx.state = STATIC_TEST_INIT;
95 test_ctx.test_start_tick = HAL_GetTick();
96 test_ctx.phase_start_tick = HAL_GetTick();
97
98 return true;
99}

101static_test_state_t StaticTest_Update(void) {
102 uint32_t now = HAL_GetTick();
103 uint32_t phase_elapsed = now - test_ctx.phase_start_tick;
104
105 switch (test_ctx.state) {
106 case STATIC_TEST_IDLE:
107 case STATIC_TEST_COMPLETE:
108 case STATIC_TEST_FAILED:
109 // Nothing to do
110 break;
111
112 case STATIC_TEST_INIT: {
113 // Initialize load cell using I2C2 (I2C_LOADCELL from config.h)
114 printf("[STATIC_TEST] Initializing load cell on I2C2...\r\n");
115
116 // First check if load cell is responding
117 if (HAL_I2C_IsDeviceReady(&hi2c2, (FX29_ADDR_0 << 1), 3, 100) != HAL_OK) {
118 test_ctx.error_msg = "Load cell not connected (I2C2 addr 0x28)";
119 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
120 test_ctx.state = STATIC_TEST_FAILED;
121 sd_card_resume(); // Resume SD on failure
122 break;
123 }
124 printf("[STATIC_TEST] Load cell found on I2C2\r\n");
125
126 if (!FX29_Init(&test_ctx.loadcell, &hi2c2, FX29_ADDR_0, FX29_RANGE_500N)) {
127 test_ctx.error_msg = "Load cell init failed";
128 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
129 test_ctx.state = STATIC_TEST_FAILED;
130 sd_card_resume(); // Resume SD on failure
131 break;
132 }
133 test_ctx.loadcell_initialized = true;
134
135 // Apply calibration scale factor
136 FX29_SetScaleFactor(&test_ctx.loadcell, LOADCELL_SCALE_FACTOR);
137 printf("[STATIC_TEST] Calibration applied: scale=%.4f\r\n", LOADCELL_SCALE_FACTOR);
138
139 // Tare the load cell
140 printf("[STATIC_TEST] Taring load cell...\r\n");
141 if (!FX29_Tare(&test_ctx.loadcell)) {
142 test_ctx.error_msg = "Load cell tare failed";
143 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
144 test_ctx.state = STATIC_TEST_FAILED;
145 sd_card_resume(); // Resume SD on failure
146 break;
147 }
148
149 // Ensure motor is at 0%
150 PWM_SetThrottle(0.0f);
151
152 // Move to settle phase
153 printf("[STATIC_TEST] Init complete, settling...\r\n");
154 test_ctx.state = STATIC_TEST_SETTLE;
155 test_ctx.phase_start_tick = now;
156 break;
157 }
158
159 case STATIC_TEST_SETTLE: {
160 if (phase_elapsed >= STATIC_TEST_SETTLE_DELAY_MS) {
161 // STEP: Jump directly to max throttle (no ramp)
162 printf("[STATIC_TEST] STEP to %d%%\r\n",
163 test_ctx.config.max_throttle_percent);
164 test_ctx.current_throttle = (float)test_ctx.config.max_throttle_percent;
165 PWM_SetThrottle(test_ctx.current_throttle);
166
167 test_ctx.state = STATIC_TEST_HOLD;
168 test_ctx.phase_start_tick = now;
169 test_ctx.last_sample_tick = now;
170 }
171 break;
172 }
173
174 case STATIC_TEST_HOLD: {
175 // Hold at max throttle
176 test_ctx.current_throttle = (float)test_ctx.config.max_throttle_percent;
177 PWM_SetThrottle(test_ctx.current_throttle);
178
179 // Continue sampling
180 if ((now - test_ctx.last_sample_tick) >= test_ctx.config.sample_interval_ms) {
181 take_sample();
182 test_ctx.last_sample_tick = now;
183 }
184
185 // Check if hold complete
186 if (phase_elapsed >= test_ctx.config.hold_duration_ms) {
187 // STEP: Jump directly to 0% (no ramp)
188 printf("[STATIC_TEST] Hold complete, STEP to 0%%\r\n");
189 PWM_SetThrottle(0.0f);
190 test_ctx.current_throttle = 0.0f;
191 test_ctx.state = STATIC_TEST_SAVING;
192 test_ctx.phase_start_tick = now;
193 }
194 break;
195 }
196
197 case STATIC_TEST_SAVING: {
198 // Save data to SD card
199 if (save_test_data_to_sd()) {
200 printf("[STATIC_TEST] TEST COMPLETE! %d samples saved to %s\r\n",
201 test_ctx.data_count, test_filename);
202 printf("[STATIC_TEST] Max thrust: %.2f N at %.1f%% PWM\r\n",
203 test_ctx.max_thrust_n, test_ctx.max_thrust_pwm);
204 test_ctx.state = STATIC_TEST_COMPLETE;
205 sd_card_resume(); // Resume SD after test
206 } else {
207 test_ctx.error_msg = "Failed to save data";
208 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
209 test_ctx.state = STATIC_TEST_FAILED;
210 sd_card_resume(); // Resume SD after test
211 }
212 break;
213 }
214 }
215
216 return test_ctx.state;
217}

219static void take_sample(void) {
220 if (test_ctx.data_count >= test_ctx.max_samples) {
221 return; // Buffer full
222 }
223
224 LOADCELL_t reading;
225 if (FX29_ReadWithPWM(&test_ctx.loadcell, &reading, (uint16_t)(test_ctx.current_throttle * 10))) {
226 thrust_data_point_t *dp = &test_ctx.data[test_ctx.data_count];
227 dp->timestamp_ms = HAL_GetTick() - test_ctx.test_start_tick;
228 dp->thrust_n = reading.force_n;
229 dp->pwm_percent = test_ctx.current_throttle;
230 dp->raw_counts = reading.raw_counts;
231
232 test_ctx.current_thrust = reading.force_n;
233 test_ctx.data_count++;
234
235 // Track max thrust
236 if (reading.force_n > test_ctx.max_thrust_n) {
237 test_ctx.max_thrust_n = reading.force_n;
238 test_ctx.max_thrust_pwm = test_ctx.current_throttle;
239 }
240 }

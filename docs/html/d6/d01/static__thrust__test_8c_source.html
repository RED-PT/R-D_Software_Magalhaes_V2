38bool StaticTest_Start(uint8_t max_throttle_percent) {
39 // Validate parameters
40 if (max_throttle_percent < 1 || max_throttle_percent > 100) {
41 test_ctx.error_msg = "Invalid throttle percentage (1-100)";
42 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
43 return false;
44 }
45
46 // Check if test already running
47 if (test_ctx.state != STATIC_TEST_IDLE &&
48 test_ctx.state != STATIC_TEST_COMPLETE &&
49 test_ctx.state != STATIC_TEST_FAILED) {
50 test_ctx.error_msg = "Test already in progress";
51 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
52 return false;
53 }
54
55 // Reset context
56 memset(&test_ctx, 0, sizeof(test_ctx));
57
58 // Configure test
59 test_ctx.config.max_throttle_percent = max_throttle_percent;
60 test_ctx.config.ramp_duration_ms = STATIC_TEST_RAMP_DURATION_MS;
61 test_ctx.config.hold_duration_ms = STATIC_TEST_HOLD_DURATION_MS;
62 test_ctx.config.sample_interval_ms = 1000 / STATIC_TEST_SAMPLE_RATE_HZ;
63 test_ctx.max_samples = STATIC_TEST_MAX_SAMPLES;
64
65 // Generate filename
66 snprintf(test_filename, sizeof(test_filename), "static_test_pwm_%02d.csv", max_throttle_percent);
67
68 printf("[STATIC_TEST] Starting test: max=%d%%, file=%s\r\n",
69 max_throttle_percent, test_filename);
70
71 // Pause SD card to avoid EMI interference from motor
72 sd_card_pause();
73
74 // Start initialization phase
75 test_ctx.state = STATIC_TEST_INIT;
76 test_ctx.test_start_tick = HAL_GetTick();
77 test_ctx.phase_start_tick = HAL_GetTick();
78
79 return true;
80}

82static_test_state_t StaticTest_Update(void) {
83 uint32_t now = HAL_GetTick();
84 uint32_t phase_elapsed = now - test_ctx.phase_start_tick;
85
86 switch (test_ctx.state) {
87 case STATIC_TEST_IDLE:
88 case STATIC_TEST_COMPLETE:
89 case STATIC_TEST_FAILED:
90 // Nothing to do
91 break;
92
93 case STATIC_TEST_INIT: {
94 // Initialize load cell using I2C2 (I2C_LOADCELL from config.h)
95 printf("[STATIC_TEST] Initializing load cell on I2C2...\r\n");
96
97 // First check if load cell is responding
98 if (HAL_I2C_IsDeviceReady(&hi2c2, (FX29_ADDR_0 << 1), 3, 100) != HAL_OK) {
99 test_ctx.error_msg = "Load cell not connected (I2C2 addr 0x28)";
100 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
101 test_ctx.state = STATIC_TEST_FAILED;
102 sd_card_resume(); // Resume SD on failure
103 break;
104 }
105 printf("[STATIC_TEST] Load cell found on I2C2\r\n");
106
107 if (!FX29_Init(&test_ctx.loadcell, &hi2c2, FX29_ADDR_0, FX29_RANGE_500N)) {
108 test_ctx.error_msg = "Load cell init failed";
109 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
110 test_ctx.state = STATIC_TEST_FAILED;
111 sd_card_resume(); // Resume SD on failure
112 break;
113 }
114 test_ctx.loadcell_initialized = true;
115
116 // Tare the load cell
117 printf("[STATIC_TEST] Taring load cell...\r\n");
118 if (!FX29_Tare(&test_ctx.loadcell)) {
119 test_ctx.error_msg = "Load cell tare failed";
120 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
121 test_ctx.state = STATIC_TEST_FAILED;
122 sd_card_resume(); // Resume SD on failure
123 break;
124 }
125
126 // Ensure motor is at 0%
127 PWM_SetThrottle(0.0f);
128
129 // Move to settle phase
130 printf("[STATIC_TEST] Init complete, settling...\r\n");
131 test_ctx.state = STATIC_TEST_SETTLE;
132 test_ctx.phase_start_tick = now;
133 break;
134 }
135
136 case STATIC_TEST_SETTLE: {
137 if (phase_elapsed >= STATIC_TEST_SETTLE_DELAY_MS) {
138 printf("[STATIC_TEST] Starting ramp up to %d%%\r\n",
139 test_ctx.config.max_throttle_percent);
140 test_ctx.state = STATIC_TEST_RAMP_UP;
141 test_ctx.phase_start_tick = now;
142 test_ctx.last_sample_tick = now;
143 test_ctx.current_throttle = 0.0f;
144 }
145 break;
146 }
147
148 case STATIC_TEST_RAMP_UP: {
149 // Calculate current throttle based on ramp progress
150 float ramp_progress = (float)phase_elapsed / (float)test_ctx.config.ramp_duration_ms;
151 if (ramp_progress > 1.0f) ramp_progress = 1.0f;
152
153 test_ctx.current_throttle = ramp_progress * (float)test_ctx.config.max_throttle_percent;
154 PWM_SetThrottle(test_ctx.current_throttle);
155
156 // Take sample at regular intervals
157 if ((now - test_ctx.last_sample_tick) >= test_ctx.config.sample_interval_ms) {
158 take_sample();
159 test_ctx.last_sample_tick = now;
160 }
161
162 // Check if ramp complete
163 if (phase_elapsed >= test_ctx.config.ramp_duration_ms) {
164 printf("[STATIC_TEST] Ramp complete, holding at %d%%\r\n",
165 test_ctx.config.max_throttle_percent);
166 test_ctx.state = STATIC_TEST_HOLD;
167 test_ctx.phase_start_tick = now;
168 }
169 break;
170 }
171
172 case STATIC_TEST_HOLD: {
173 // Hold at max throttle
174 test_ctx.current_throttle = (float)test_ctx.config.max_throttle_percent;
175 PWM_SetThrottle(test_ctx.current_throttle);
176
177 // Continue sampling
178 if ((now - test_ctx.last_sample_tick) >= test_ctx.config.sample_interval_ms) {
179 take_sample();
180 test_ctx.last_sample_tick = now;
181 }
182
183 // Check if hold complete
184 if (phase_elapsed >= test_ctx.config.hold_duration_ms) {
185 printf("[STATIC_TEST] Hold complete, ramping down\r\n");
186 test_ctx.state = STATIC_TEST_RAMP_DOWN;
187 test_ctx.phase_start_tick = now;
188 }
189 break;
190 }
191
192 case STATIC_TEST_RAMP_DOWN: {
193 // Calculate current throttle based on ramp down progress
194 float ramp_progress = (float)phase_elapsed / (float)test_ctx.config.ramp_duration_ms;
195 if (ramp_progress > 1.0f) ramp_progress = 1.0f;
196
197 test_ctx.current_throttle = (1.0f - ramp_progress) * (float)test_ctx.config.max_throttle_percent;
198 PWM_SetThrottle(test_ctx.current_throttle);
199
200 // Take sample at regular intervals
201 if ((now - test_ctx.last_sample_tick) >= test_ctx.config.sample_interval_ms) {
202 take_sample();
203 test_ctx.last_sample_tick = now;
204 }
205
206 // Check if ramp down complete
207 if (phase_elapsed >= test_ctx.config.ramp_duration_ms) {
208 printf("[STATIC_TEST] Ramp down complete, saving data\r\n");
209 PWM_SetThrottle(0.0f);
210 test_ctx.current_throttle = 0.0f;
211 test_ctx.state = STATIC_TEST_SAVING;
212 test_ctx.phase_start_tick = now;
213 }
214 break;
215 }
216
217 case STATIC_TEST_SAVING: {
218 // Save data to SD card
219 if (save_test_data_to_sd()) {
220 printf("[STATIC_TEST] TEST COMPLETE! %d samples saved to %s\r\n",
221 test_ctx.data_count, test_filename);
222 printf("[STATIC_TEST] Max thrust: %.2f N at %.1f%% PWM\r\n",
223 test_ctx.max_thrust_n, test_ctx.max_thrust_pwm);
224 test_ctx.state = STATIC_TEST_COMPLETE;
225 sd_card_resume(); // Resume SD after test
226 } else {
227 test_ctx.error_msg = "Failed to save data";
228 printf("[STATIC_TEST] ERROR: %s\r\n", test_ctx.error_msg);
229 test_ctx.state = STATIC_TEST_FAILED;
230 sd_card_resume(); // Resume SD after test
231 }
232 break;
233 }
234 }
235
236 return test_ctx.state;
237}

239static void take_sample(void) {
240 if (test_ctx.data_count >= test_ctx.max_samples) {
241 return; // Buffer full
242 }
243
244 LOADCELL_t reading;
245 if (FX29_ReadWithPWM(&test_ctx.loadcell, &reading, (uint16_t)(test_ctx.current_throttle * 10))) {
246 thrust_data_point_t *dp = &test_ctx.data[test_ctx.data_count];
247 dp->timestamp_ms = HAL_GetTick() - test_ctx.test_start_tick;
248 dp->thrust_n = reading.force_n;
249 dp->pwm_percent = test_ctx.current_throttle;
250 dp->raw_counts = reading.raw_counts;
251
252 test_ctx.current_thrust = reading.force_n;
253 test_ctx.data_count++;
254
255 // Track max thrust
256 if (reading.force_n > test_ctx.max_thrust_n) {
257 test_ctx.max_thrust_n = reading.force_n;
258 test_ctx.max_thrust_pwm = test_ctx.current_throttle;
259 }
260 }

277void StaticTest_Cancel(void) {
278 if (test_ctx.state != STATIC_TEST_IDLE &&
279 test_ctx.state != STATIC_TEST_COMPLETE &&
280 test_ctx.state != STATIC_TEST_FAILED) {
281
282 printf("[STATIC_TEST] Test CANCELLED\r\n");
283
284 // Safety: stop motor
285 PWM_SetThrottle(0.0f);
286 test_ctx.current_throttle = 0.0f;
287
288 test_ctx.error_msg = "Test cancelled by user";
289 test_ctx.state = STATIC_TEST_FAILED;
290 sd_card_resume(); // Resume SD after cancel
291 }

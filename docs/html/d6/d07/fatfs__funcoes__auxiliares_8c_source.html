90FRESULT Scan_SD(char *pat) {
91 if (sd_not_mounted) {
92 return_sd_card_not_mounted();
93 return INVALID;
94 }
95
96 DIR dir;
97 UINT i;
98 char buf[128];
99 char fname[32];
100 char path[64];
101 snprintf(path, sizeof(path), "%s", pat);
102
103 fresult = f_opendir(&dir, path); /* Open the directory */
104 if (fresult == FR_OK) {
105 for (;;) {
106 fresult = f_readdir(&dir, &fno); /* Read a directory item */
107 if (fresult != FR_OK || fno.fname[0] == 0)
108 break; /* Break on error or end of dir */
109 if (fno.fattrib & AM_DIR) /* It is a directory */
110 {
111 if (!(strcmp("SYSTEM~1", fno.fname)))
112 continue;
113 strncpy(fname, fno.fname, sizeof(fname) - 1);
114 snprintf(buf, sizeof(buf), "Dir: %s\r\n", fname);
115 Send_Uart(buf);
116 i = strlen(path);
117 snprintf(&path[i], sizeof(path) - i, "/%s", fno.fname);
118 fresult = Scan_SD(path); /* Enter the directory */
119 if (fresult != FR_OK)
120 break;
121 path[i] = 0;
122 } else { /* It is a file. */
123 strncpy(fname, fno.fname, sizeof(fname) - 1);
124 snprintf(buf, sizeof(buf), "File: %s/%s\r\n", path, fname);
125 Send_Uart(buf);
126 }
127 }
128 f_closedir(&dir);
129 }
130 return fresult;
131}

496void Update_File(int index, void *data, size_t len) {
497 if (sd_not_mounted) {

500 }
501
502 custom_open_file *cof = get_open_file(index);
503
504 executionTime = time_in_millis - cof->time_opened_ms;

506 {
507 // save data to sd card
508 f_sync(&cof->open_file);
509
510 // debugging
511#if (DEBUG_SD_CARD)
512 printf("sd card data: %ld MB, %ld KB\r\n", data_size/1000000, data_size/1000);
513
514 printf("sd card freq: %d Hz\r\n", msg_counter);
515
516// printf("execution time = %d s\r\n", executionTime);
517
518 printf("sd card datarate: %f KB/s\r\n\n", (float)msg_counter*(float)len/(float)executionTime);
519 #endif
520
521 // reset counters
522 msg_counter = 0;
523 cof->time_opened_ms = time_in_millis;
524 }
525
526 // write data to the file and refresh variables
527 Write_File(index, data, INVALID, len);
528 msg_counter++;
529 data_size += len;
530}
